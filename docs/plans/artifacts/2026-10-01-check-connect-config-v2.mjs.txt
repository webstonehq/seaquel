#!/usr/bin/env node
// Phase 5a, Task 1: check the v2 connect-config fixtures against v1.
//
//   node check-connect-config-v2.mjs     (from the repo root)
//
// This reference copy lives in docs/plans/artifacts/ as .mjs.txt; copy it
// anywhere as .mjs to run it. It reads
// crates/seaquel-workspace/tests/fixtures/connect-config/ (v1: the nine
// phase 4 groups and form-add/form-test) and connect-config-v2/, and checks:
//
//   - every v1 case is in v2 exactly once, under the same id, and nothing
//     else is: `target` is `saved` for the phase 4 cases and `form` for the
//     form cases, and `op` is `connect` or `test`;
//   - a case with an empty `changedBy` expects exactly v1's output;
//   - a case with a non-empty `changedBy` expects something else, and names
//     only Decision 6 rows (1-6, 7a-7d, 8-11, and 5-wizard for the owner's
//     settled choice I: the wizard's SSL mode defaults to unset);
//   - `needsLiveCheck` cases carry a note.
//
// "v1's output" is what the phase 4 replay test compares: for a saved case
// `case[case.gui]`'s config and tunnel, or CREDENTIALS_REQUIRED when
// `gui` is `form`, plus autoReconnect's `secretsRead`; for a form case the
// recorded config, tunnel and error. `tunnelClosed` is left out of both.
// It prints the counts per row and the needsLiveCheck cases, and exits 1 on
// any failure.

import fs from "node:fs";
import path from "node:path";

const DIR = path.join(process.cwd(), "crates/seaquel-workspace/tests/fixtures");
const V1 = path.join(DIR, "connect-config");
const V2 = path.join(DIR, "connect-config-v2");
const GROUPS = ["postgres", "mysql", "mariadb", "mssql", "sqlite", "duckdb", "ssh", "secrets", "shared"];
const FORMS = ["form-add", "form-test"];
const ROWS = ["1", "2", "3", "4", "5", "5-wizard", "6", "7a", "7b", "7c", "7d", "8", "9", "10", "11"];

const read = (dir, f) => JSON.parse(fs.readFileSync(path.join(dir, f + ".json"), "utf8"));

/** Deep equality for JSON values; object key order doesn't matter. */
function same(a, b) {
  if (a === b) return true;
  if (typeof a !== "object" || typeof b !== "object" || a === null || b === null) return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a)) return a.length === b.length && a.every((x, i) => same(x, b[i]));
  const ka = Object.keys(a).sort();
  const kb = Object.keys(b).sort();
  return same(ka, kb) && ka.every((k) => same(a[k], b[k]));
}

const failures = [];
const fail = (m) => failures.push(m);

// v1 outputs by id.
const v1 = new Map();
for (const g of GROUPS) {
  for (const c of read(V1, g)) {
    const out = { secretsRead: c.autoReconnect.secretsRead };
    if (c.gui === "form") out.error = "CREDENTIALS_REQUIRED";
    else {
      if (c[c.gui].tunnel !== undefined) out.tunnel = c[c.gui].tunnel;
      out.config = c[c.gui].config;
    }
    v1.set(c.name, { target: "saved", op: "connect", out });
  }
}
for (const f of FORMS) {
  for (const c of read(V1, f)) {
    const out = {};
    if (c.tunnel !== undefined) out.tunnel = c.tunnel;
    if (c.config !== undefined) out.config = c.config;
    if (c.error !== undefined) out.error = c.error;
    v1.set(c.name, { target: "form", op: c.op === "add" ? "connect" : "test", out });
  }
}

const seen = new Set();
const perRow = Object.fromEntries(ROWS.map((r) => [r, 0]));
const live = [];
let unchanged = 0;
let changed = 0;
const byTarget = { saved: 0, form: 0 };
for (const g of GROUPS) {
  for (const c of read(V2, g)) {
    if (seen.has(c.name)) fail(`${c.name}: twice in v2`);
    seen.add(c.name);
    const old = v1.get(c.name);
    if (!old) {
      fail(`${c.name}: not a v1 case`);
      continue;
    }
    if (c.target !== old.target) fail(`${c.name}: target ${c.target}, expected ${old.target}`);
    if (c.op !== old.op) fail(`${c.name}: op ${c.op}, expected ${old.op}`);
    byTarget[c.target] = (byTarget[c.target] ?? 0) + 1;
    if (!Array.isArray(c.changedBy)) {
      fail(`${c.name}: changedBy isn't a list`);
      continue;
    }
    for (const r of c.changedBy) {
      if (!ROWS.includes(r)) fail(`${c.name}: unknown row ${JSON.stringify(r)}`);
      else perRow[r] += 1;
    }
    const equal = same(c.expected, old.out);
    if (c.changedBy.length === 0) {
      unchanged += 1;
      if (!equal) fail(`${c.name}: changedBy is empty but the output differs from v1\n  v1 ${JSON.stringify(old.out)}\n  v2 ${JSON.stringify(c.expected)}`);
    } else {
      changed += 1;
      if (equal) fail(`${c.name}: changedBy ${c.changedBy.join(",")} but the output equals v1`);
    }
    if (c.needsLiveCheck) {
      live.push(c.name);
      if (!c.notes) fail(`${c.name}: needsLiveCheck without a note`);
    }
  }
}
for (const name of v1.keys()) if (!seen.has(name)) fail(`${name}: missing from v2`);

console.log(`v2: ${seen.size} cases (saved ${byTarget.saved}, form ${byTarget.form}); ${unchanged} equal v1, ${changed} differ`);
console.log("cases per Decision 6 row (a case can name several):");
for (const r of ROWS) console.log(`  ${r.padEnd(3)} ${perRow[r]}`);
console.log(`needsLiveCheck: ${live.length}`);
for (const n of live) console.log(`  ${n}`);
if (failures.length) {
  console.log(`\n${failures.length} failure(s):`);
  for (const f of failures) console.log(`- ${f}`);
  process.exit(1);
}
console.log("OK");
