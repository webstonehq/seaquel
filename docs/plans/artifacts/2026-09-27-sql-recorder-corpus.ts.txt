export interface CorpusEntry {
  id: string;
  kind: "tutorial" | "roundtrip" | "editor";
  sql: string;
  /** Engine whose dialect the visual AST uses. */
  dialect: Dialect;
  /** tutorial: the challenge this query should pass. */
  challengeId?: string;
  /** "tutorial" → the tutorial's table list, null → accept every table. */

  /** roundtrip: which table-name resolver generated the SQL. */
  tableRef?: TableRefKind;
  /** roundtrip: the builder state, so the harness can regenerate SQL. */
  state?: BuilderState;
}

/** SQL after "Write:" in a hint, up to the first blank line. */
function hintSql(hint: string | undefined): string | null {
  if (!hint) return null;
  const m = /Write:\s*([\s\S]*?)(\n\n|$)/.exec(hint);
  return m ? m[1].trim() : null;
}

export function tutorialEntries(): CorpusEntry[] {
  const out: CorpusEntry[] = [];
  for (const lesson of Object.values(LESSONS)) {
    for (const ch of lesson.challenges) {
      const sql = SOLUTIONS[ch.id];
      if (!sql) throw new Error(`no solution for ${ch.id}`);
      out.push({
        id: `tut:${ch.id}`,
        kind: "tutorial",
        sql,
        dialect: "postgres",
        challengeId: ch.id,
        validTables: "tutorial",
      });
      const hinted = hintSql(ch.hint);
      if (hinted && hinted !== sql) {
        out.push({
          id: `tut:${ch.id}:hint`,
          kind: "tutorial",
          sql: hinted,
          dialect: "postgres",
          challengeId: ch.id,
          validTables: "tutorial",
        });
      }
    }
  }
  return out;
}

/** Serializable builder state (Sets/Maps as arrays). */
export interface BuilderState {
  tables: {
    tableName: string;
    cols: string[];
    aggs?: [string, ColumnAggregate][];
    cte?: boolean;
  }[];
  joins: Omit<CanvasJoin, "id">[];
  filters: (Omit<FilterCondition, "id" | "subqueryId"> & { subquery?: number })[];
  groupBy: string[];
  having: Omit<HavingCondition, "id">[];
  orderBy: Omit<SortCondition, "id">[];
  limit: number | null;
  selectAggregates: Omit<SelectAggregate, "id">[];
  subqueries: { role: "where" | "from" | "select"; alias?: string; inner: BuilderState }[];
  ctes?: { name: string; inner: BuilderState }[];
}

function st(p: Partial<BuilderState>): BuilderState {
  return {
    tables: [],
    joins: [],
    filters: [],
    groupBy: [],
    having: [],
    orderBy: [],
    limit: null,
    selectAggregates: [],
    subqueries: [],
    ...p,
  };
}
const t = (tableName: string, cols: string[] = [], aggs?: [string, ColumnAggregate][]) => ({
  tableName,
  cols,
  aggs,
});
const j = (
  sourceTable: string,
  sourceColumn: string,
  targetTable: string,
  targetColumn: string,
  joinType: JoinType = "INNER",
) => ({ sourceTable, sourceColumn, targetTable, targetColumn, joinType });
const f = (
  column: string,
  operator: FilterOperator,
  value = "",
  connector: "AND" | "OR" = "AND",
  subquery?: number,
) => ({ column, operator, value, connector, subquery });
const h = (
  aggregateFunction: AggregateFunction,
  column: string,
  operator: HavingOperator,
  value: string,
  connector: "AND" | "OR" = "AND",
) => ({ aggregateFunction, column, operator, value, connector });

export type TableRefKind = "bare" | "postgres" | "mysql" | "mssql" | "sqlite" | "duckdb";

/** What editorQualifiedTable produces per engine for these tables. */
export const TABLE_REFS: Record<Exclude<TableRefKind, "bare">, TableRef> = {
  postgres: (n) => (n === "customers" ? `"Sales"."customers"` : `public.${n}`),
  mysql: (n) => (n === "customers" ? "`Sales`.`customers`" : `shop.${n}`),
  mssql: (n) => (n === "customers" ? "[Sales].[customers]" : `dbo.${n}`),
  sqlite: (n) => `main.${n}`,
  duckdb: (n) => (n === "customers" ? `"Sales".customers` : `main.${n}`),
};

export function tableRefFor(kind: TableRefKind): TableRef | undefined {
  return kind === "bare" ? undefined : TABLE_REFS[kind];
}

let idSeq = 0;
const nid = (p: string) => `${p}-${++idSeq}`;

function toInner(s: BuilderState): SubqueryInnerState & { ctes: CanvasCTE[] } {
  const subqueries: CanvasSubquery[] = s.subqueries.map((sq) => {
    const inner = toInner(sq.inner);
    return {
      id: nid("sq"),
      position: { x: 0, y: 0 },
      size: { width: 300, height: 200 },
      role: sq.role,
      alias: sq.alias,
      innerQuery: inner,
    };
  });
  const tables: CanvasTable[] = s.tables.map((tb) => ({
    id: nid("t"),
    tableName: tb.tableName,
    position: { x: 0, y: 0 },
    selectedColumns: new Set(tb.cols),
    columnAggregates: new Map(tb.aggs ?? []),
    ...(tb.cte ? { cteId: nid("cte") } : {}),
  }));
  const filters: FilterCondition[] = s.filters.map((fl) => {
    const out: FilterCondition = {
      id: nid("f"),
      column: fl.column,
      operator: fl.operator,
      value: fl.value,
      connector: fl.connector,
    };
    if (fl.subquery !== undefined) {
      out.subqueryId = subqueries[fl.subquery].id;
      subqueries[fl.subquery].linkedFilterId = out.id;
    }
    return out;
  });
  return {
    tables,
    joins: s.joins.map((jn) => ({ id: nid("j"), ...jn })),
    filters,
    groupBy: s.groupBy.map((c): GroupByCondition => ({ id: nid("g"), column: c })),
    having: s.having.map((hv) => ({ id: nid("h"), ...hv })),
    orderBy: s.orderBy.map((o) => ({ id: nid("o"), ...o })),
    limit: s.limit,
    selectAggregates: s.selectAggregates.map((a) => ({ id: nid("a"), ...a })),
    subqueries,
    ctes: (s.ctes ?? []).map((c) => ({
      id: nid("cte"),
      name: c.name,
      position: { x: 0, y: 0 },
      size: { width: 300, height: 200 },
      innerQuery: toInner(c.inner),
    })),
  };
}

export function sqlForState(s: BuilderState, ref: TableRefKind): string {
  const i = toInner(s);
  return buildSql(
    i.tables,
    i.joins,
    i.filters,
    i.groupBy,
    i.having,
    i.orderBy,
    i.limit,
    i.selectAggregates,
    i.subqueries,
    i.ctes,
    tableRefFor(ref),
  );
}

export function roundtripEntries(): CorpusEntry[] {
  const out: CorpusEntry[] = [];
  for (const [name, state] of Object.entries(BUILDER_STATES)) {
    out.push({
      id: `rt:${name}`,
      kind: "roundtrip",
      sql: sqlForState(state, "bare"),
      dialect: "postgres",
      validTables: "tutorial",
      tableRef: "bare",
      state,
    });
  }
  for (const name of QUALIFIED_STATES) {
    for (const ref of Object.keys(TABLE_REFS) as Exclude<TableRefKind, "bare">[]) {
      out.push({
        id: `rt:${name}:${ref}`,
        kind: "roundtrip",
        sql: sqlForState(BUILDER_STATES[name], ref),
        dialect: ref,
        validTables: null,
        tableRef: ref,
        state: BUILDER_STATES[name],
      });
    }
  }
  return out;
}
