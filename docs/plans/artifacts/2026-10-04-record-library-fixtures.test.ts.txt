/**
 * Phase 5d-1, Task 2: records what today's TypeScript does with the library
 * (connections, projects, custom labels, saved queries and their versions)
 * into the library parity fixtures,
 * `crates/seaquel-workspace/tests/fixtures/library/*.json`.
 *
 * This reference copy is
 * `docs/plans/artifacts/2026-10-04-record-library-fixtures.test.ts.txt`.
 * To run it, copy it into the tree, run it with vitest, then delete the copy:
 *
 *   cp docs/plans/artifacts/2026-10-04-record-library-fixtures.test.ts.txt \
 *      src/lib/hooks/database/record-library.test.ts
 *   FREEZE_LIBRARY=1 SEAQUEL_WASM_PREBUILT=1 \
 *     mise exec -- npx vitest run src/lib/hooks/database/record-library.test.ts
 *   rm src/lib/hooks/database/record-library.test.ts
 *
 * Without FREEZE_LIBRARY it records every case and writes nothing.
 * FREEZE_LIBRARY_OUT points the output at another directory (to check that
 * two runs agree).
 *
 * What runs unmodified:
 * - `ConnectionManager` (`add`, `update`, `reconnect`, `autoReconnect`,
 *   `remove`, `toggleLocalOnly`, `importConnections`,
 *   `initializePersistedConnections`), `LabelManager`, `ProjectManager`
 *   (`initialize`, `add`, `update`, `setGitRepoPath`, `remove`, the custom
 *   label calls, `importFromGitRepo`, `importSharedConnections`,
 *   `importSingleSharedConnection`), `SavedQueryManager`,
 *   `StateRestorationManager` and `PersistenceManager`, after Task 1's fixes;
 * - the demo's storage client (`createSqljsStorageClient`) on an in-memory
 *   sql.js file bootstrapped by `bootstrapSqljsDatabase`. The frozen repo
 *   fixtures pin those repositories to the Rust ones;
 * - the desktop keyring (`TauriKeyringService`), whose `callSecret` is
 *   answered by a recording in-memory store: the secret calls are the `secret`
 *   requests desktop sends Core.
 *
 * What is stubbed:
 * - the providers (`connect` answers a Core id, `disconnect` and `test` do
 *   nothing) and `getEngineClient` (`schemaTables` answers `[]`): no database
 *   is reached;
 * - the shared repos: `SharedRepoManager`'s methods the library calls, and the
 *   saved queries' file projection, record their calls (`files`) and touch no

type Json = unknown;
type Row = Record<string, unknown>;
type Seed = Partial<Record<string, Row[]>>;
type Library = { method: string; params?: Json } | { method: string; params?: Json }[] | null;

type Tab = Awaited<ReturnType<typeof openTab>>;

interface Step {
  /** The TS call, as `manager.method`. */
  op: string;
  /** Its arguments, as recorded. */
  args?: Json;
  /** The Core call the Rust replay sends for it (`library` group), or null for none. */
  library: Library | ((ids: string[]) => Library);
  run: (t: Tab, ids: string[]) => Promise<Json>;
  /** A failure the recorder forces, named in the fixture. */
  inject?: string;
}

interface Case {
  name: string;
  note?: string;
  /** The file the case starts from: a fresh one, or one that started on v2026.4.5-beta.1. */
  file?: "current" | "v2026.4.5-beta.1";
  /** Rows inserted before the managers load, in table order; the standard seed when absent. */
  seed?: Seed;
  /** Entries in the secret store before the case. */
  secrets?: Record<string, string>;
  /** Shared-repo state in memory before the case (not stored). */
  sharedRepo?: boolean;
  /** Whether the managers load before the first step (default true). */
  load?: boolean;
  web?: boolean;
  steps: Step[];
}

type Db = InstanceType<typeof WebSqliteDatabase>;

async function insertRows(db: Db, seed: Seed): Promise<void> {
  for (const { table } of TABLES) {
    for (const row of seed[table] ?? []) {
      const cols = Object.keys(row);
      await db.execute(
        `INSERT INTO ${table} (${cols.join(", ")}) VALUES (${cols.map(() => "?").join(", ")})`,
        cols.map((c) => row[c]),
      );
    }
  }
}

/**
 * A file that started on v2026.4.5-beta.1, as the baseline upgrade leaves it:
 * `saved_queries.project_id` and `dashboards.project_id` are plain nullable
 * columns with no foreign key, so a project's delete doesn't cascade to them.
 */
async function makeBetaEra(db: Db): Promise<void> {
  await db.execute("PRAGMA foreign_keys=OFF");
  const rebuild: [string, string, string][] = [
    [
      "saved_queries",
      `CREATE TABLE saved_queries_beta (id TEXT PRIMARY KEY, name TEXT NOT NULL, query TEXT NOT NULL,
        parameters TEXT, created_at TEXT NOT NULL, updated_at TEXT NOT NULL,
        starred INTEGER NOT NULL DEFAULT 0, shared INTEGER NOT NULL DEFAULT 0, description TEXT,
        database_type TEXT, tags TEXT, folder TEXT, project_id TEXT)`,
      "CREATE INDEX idx_saved_queries_project ON saved_queries(project_id)",
    ],
    [
      "dashboards",
      `CREATE TABLE dashboards_beta (id TEXT PRIMARY KEY, name TEXT NOT NULL,
        viewport TEXT NOT NULL DEFAULT '{"x":0,"y":0,"zoom":1}', widgets TEXT NOT NULL DEFAULT '[]',
        date_filter TEXT, created_at TEXT NOT NULL, updated_at TEXT NOT NULL, starred INTEGER DEFAULT 0,
        shared INTEGER NOT NULL DEFAULT 0, description TEXT, project_id TEXT)`,
      "CREATE INDEX idx_dashboards_project ON dashboards(project_id)",
    ],
  ];
  for (const [table, create, index] of rebuild) {
    await db.execute(create);
    await db.execute(`DROP TABLE ${table}`);
    await db.execute(`ALTER TABLE ${table}_beta RENAME TO ${table}`);
    await db.execute(`DROP INDEX IF EXISTS idx_${table}_project`);
    await db.execute(index);
  }
  await db.execute("PRAGMA foreign_keys=ON");
}

async function dump(db: Db): Promise<Record<string, Row[]>> {
  const out: Record<string, Row[]> = {};
  for (const { table, order, columns } of TABLES) {
    const rows = await db.query<Row>(`SELECT ${columns ?? "*"} FROM ${table} ORDER BY ${order}`);
    if (rows.length) out[table] = rows;
  }
  return out;
}

/** One window: its managers over the case's storage, wired as `UseDatabase` wires them. */
async function openTab(client: StorageClient, load: boolean) {
  rec.storage = client;
  const state = new DatabaseState();
  const persistence = new PersistenceManager(state);
  const restoration = new StateRestorationManager(state, persistence);
  const projects = new ProjectManager(state, persistence, restoration);
  const labels = new LabelManager(state, persistence);
  const savedQueries = new SavedQueryManager(
    state,
    (id) => persistence.scheduleProject(id),
    persistence,
  );
  savedQueries.setFileProjection({
    writeQueryFile: async (q) => {
      rec.files.push({
        call: "writeQueryFile",
        id: q.id,
        path: `${q.folder ?? ""}/${queryNameToFilename(q.name)}`,
      });
    },
    deleteQueryFile: async (q) => {
      rec.files.push({
        call: "deleteQueryFile",
        id: q.id,
        path: `${q.folder ?? ""}/${queryNameToFilename(q.name)}`,
      });
    },
  });
  let coreSeq = 0;
  const provider = {
    connect: async () => `core-${++coreSeq}`,
    disconnect: async () => {},
    test: async () => {},
  };
  const providers = { getForType: async () => provider };
  const connections = new ConnectionManager(
    state,
    persistence,
    restoration,
    {} as never,
    providers as never,
    async () => {},
    () => {},
  );
  projects.setRemoveConnectionCallback((id, options) => connections.remove(id, options));
  const repos = sharedRepoStub(state);
  projects.setSharedRepoManager(repos as never);
  connections.setSharedRepoManager(repos as never);
  if (load) {
    await projects.initialize();
    await connections.initializePersistedConnections();
  }
  return { state, persistence, restoration, projects, labels, savedQueries, connections };
}

/** `UseDatabase.setConnectionAIModel`, copied: `UseDatabase` isn't built here. */
async function setConnectionAIModel(
  t: Tab,
  connectionId: string,
  providerId: string,
  model: string,
) {
  const c = t.state.connections.find((x) => x.id === connectionId);
  if (!c) return;
  const updated = { ...c, activeAIProviderId: providerId, activeAIModel: model };
  await t.persistence.persistConnection(updated);
  t.state.connections = t.state.connections.map((x) => (x.id === connectionId ? updated : x));
}

/** Lets fire-and-forget writes finish and the 500 ms project save fire. */
async function settle(): Promise<void> {
  for (let i = 0; i < 6; i++) {
    await new Promise((r) => setImmediate(r));
    await vi.advanceTimersByTimeAsync(600);
  }
}

/** What the window shows: the lists a TS replay compares. */
function view(t: Tab): Json {
  const s = t.state;
  const nonEmpty = <T>(o: Record<string, T[]>) =>
    Object.fromEntries(
      Object.entries(o)
        .filter(([, v]) => v.length > 0)
        .sort(([a], [b]) => a.localeCompare(b)),
    );
  return {
    activeProjectId: s.activeProjectId,
    projects: s.projects.map((p) => ({
      id: p.id,
      name: p.name,
      description: p.description ?? null,
      gitRepoPath: p.gitRepoPath ?? null,
      customLabels: p.customLabels.map((l) => ({ id: l.id, name: l.name, color: l.color })),
    })),
    connections: s.connections.map((c) => ({
      id: c.id,
      projectId: c.projectId,
      name: c.name,
      labelIds: c.labelIds,
      connected: !!c.providerConnectionId,
    })),
    connectionOrder: nonEmpty(s.connectionOrderByProject),
    savedQueries: nonEmpty(
      Object.fromEntries(
        Object.entries(s.queriesByProject).map(([p, qs]) => [
          p,
          qs.map((q) => ({ id: q.id, name: q.name, starred: !!q.starred, shared: !!q.shared })),
        ]),
      ),
    ),
    versions: nonEmpty(
      Object.fromEntries(
        Object.entries(s.queryVersionsByProject).map(([p, vs]) => [
          p,
          vs.map((v) => `${v.queryId}#${v.version}`),
        ]),
      ),
    ),
  };
}

function errorText(e: unknown): string {
  return e instanceof Error ? e.message : String(e);
}

async function record(c: Case): Promise<Json> {
  uuidSeq = 0;
  rec.env.web = !!c.web;
  rec.toasts.length = 0;
  rec.secretCalls.length = 0;
  rec.files.length = 0;
  rec.secrets = new Map(Object.entries(c.secrets ?? {}));
  rec.secretFailure = null;
  rec.vaultCancel = false;
  vi.useFakeTimers({ toFake: ["setTimeout", "clearTimeout", "Date"], now: FIXED });
  try {
    const db = new WebSqliteDatabase(new SQL.Database());
    await bootstrapSqljsDatabase(db);
    if (c.file === "v2026.4.5-beta.1") await makeBetaEra(db);
    const seed = c.seed ?? STANDARD_SEED;
    await insertRows(db, seed);
    const client = createSqljsStorageClient(db);
    const t = await openTab(client, c.load !== false);
    if (c.sharedRepo) seedSharedRepo(t.state);
    await settle();
    rec.toasts.length = 0;
    rec.secretCalls.length = 0;
    rec.files.length = 0;
    const before = await dump(db);

    const ids: string[] = [];
    const steps: Json[] = [];
    for (const step of c.steps) {
      let outcome: Json;
      try {
        const value = await step.run(t, ids);
        outcome = { ok: true, value: value ?? null };
      } catch (e) {
        outcome = { ok: false, error: errorText(e) };
      }
      // After the run, so a step can name the ids it made.
      const library = typeof step.library === "function" ? step.library(ids) : step.library;
      await settle();
      steps.push({
        op: step.op,
        ...(step.args !== undefined ? { args: step.args } : {}),
        ...(step.inject ? { inject: step.inject } : {}),
        library,
        outcome,
        rows: await dump(db),
        secretCalls: [...rec.secretCalls],
        secretStore: Object.fromEntries([...rec.secrets].sort(([a], [b]) => a.localeCompare(b))),
        files: [...rec.files],
        toasts: [...rec.toasts],
        view: view(t),
      });
      rec.toasts.length = 0;
      rec.secretCalls.length = 0;
      rec.files.length = 0;
      rec.secretFailure = null;
    }
    return {
      name: c.name,
      ...(c.note ? { note: c.note } : {}),
      file: c.file ?? "current",
      seed,
      ...(c.secrets ? { secrets: c.secrets } : {}),
      ...(c.sharedRepo ? { sharedRepo: true } : {}),
      ...(c.web ? { target: "web" } : {}),
      before,
      steps,
    };
  } finally {
    vi.clearAllTimers();
    vi.useRealTimers();
  }
}

/**
 * Ids made during the case become `<id:n>` (numbered in order of first
 * appearance in the case's JSON), and times from the pinned clock `<now>`.
 * Seeded ids and times (2024) stay.
 */
function normalise(caseJson: Json): Json {
  let text = JSON.stringify(caseJson);
  const seen = new Map<string, string>();
  text = text.replace(/00000000-0000-4000-8000-\d{12}/g, (m) => {
    if (!seen.has(m)) seen.set(m, `<id:${seen.size + 1}>`);
    return seen.get(m)!;
  });
  text = text.replace(/2030-01-01T\d\d:\d\d:\d\d\.\d{3}Z/g, "<now>");
  return JSON.parse(text);
}

/** The `connectionCreate` a GUI on Core sends for an `add` of `input` in `projectId`. */
function createOf(
  input: Record<string, unknown>,
  projectId: string,
  extra: Record<string, unknown> = {},
): Library {
  const draft: Record<string, unknown> = {
    projectId,
    name: input.name,
    type: input.type,
    host: input.host ?? "",
    port: input.port ?? 0,
    databaseName: input.databaseName ?? "",
    username: input.username ?? "",
  };
  if (input.sslMode !== undefined) draft.sslMode = input.sslMode;
  if (input.connectionString !== undefined) draft.connectionString = input.connectionString;
  if (input.sshTunnel !== undefined) draft.sshTunnel = input.sshTunnel;
  draft.savePassword = !!input.savePassword;
  draft.saveSshPassword = !!input.saveSshPassword;
  draft.saveSshKeyPassphrase = !!input.saveSshKeyPassphrase;
  draft.labelIds = input.labelIds ?? [];
  draft.isLocalOnly = input.isLocalOnly ?? true;
  if (input.aiShareSchema !== undefined) draft.aiShareSchema = input.aiShareSchema;
  if (input.aiShareData !== undefined) draft.aiShareData = input.aiShareData;
  draft.connected = true;
  Object.assign(draft, extra);
  const secrets: Record<string, string> = {};
  if (input.savePassword && input.password) secrets.db = input.password as string;
  if (input.saveSshPassword && input.sshPassword) secrets.ssh = input.sshPassword as string;
  if (input.saveSshKeyPassphrase && input.sshKeyPassphrase)
    secrets.sshKey = input.sshKeyPassphrase as string;
  return {
    method: "connectionCreate",
    params: { connection: draft, ...(Object.keys(secrets).length ? { secrets } : {}) },
  };
}

function addStep(input: Record<string, unknown>, projectId = "p1"): Step {
  return {
    op: "connections.add",
    args: input,
    library: createOf(input, projectId),
    run: async (t, ids) => {
      const id = await t.connections.add(input as never);
      ids.push(id);
      return id;
    },
  };
}

/** The stored row of `c1` as the form shows it, for `update`/`reconnect` inputs. */
const c1Form = (over: Record<string, unknown> = {}) => ({
  name: "Local",
  type: "postgres",
  host: "localhost",
  port: 5432,
  databaseName: "app",
  username: "me",
  password: "",
  savePassword: false,
  saveSshPassword: false,
  saveSshKeyPassphrase: false,
  ...over,
});

function updateStep(
  input: Record<string, unknown>,
  patch: Record<string, unknown>,
  secrets?: Record<string, unknown>,
  id = "c1",
): Step {
  return {
    op: "connections.update",
    args: { id, input },
    library: {
      method: "connectionUpdate",
      params: { id, patch, ...(secrets ? { secrets } : {}) },
    },
    run: async (t) => {
      await t.connections.update(id, input as never);
      return null;
    },
  };
}

const tabWith = (t: Tab, tabId: string, queryId?: string) => {
  t.state.queryTabsByProject = {
    ...t.state.queryTabsByProject,
    p1: [
      ...(t.state.queryTabsByProject.p1 ?? []),
      { id: tabId, name: "Tab", query: "", ...(queryId ? { queryId } : {}) } as never,
    ],
  };
};

/** `saveQuery` from a tab linked to `q1` (the save dialog on a linked tab): an update. */
function saveLinked(
  name: string,
  query: string,
  patch: Record<string, unknown>,
  params?: unknown,
): Step {
  return {
    op: "savedQueries.saveQuery",
    args: { name, query, tabId: "t1", ...(params !== undefined ? { parameters: params } : {}) },
    library: { method: "savedQueryUpdate", params: { id: "q1", patch } },
    run: async (t) => {
      if (!(t.state.queryTabsByProject.p1 ?? []).some((x) => x.id === "t1")) tabWith(t, "t1", "q1");
      return t.savedQueries.saveQuery(name, query, "t1", params as never);
    },
  };
}

const q1Seed = (extra: Seed = {}, q1: Row = {}) =>
  withStandard({ ...extra, saved_queries: [savedQuery(q1), ...(extra.saved_queries ?? [])] });

/** The prune setting, added to the standard seed's `app_state`. */
function limit(n: string): Row[] {
  return [{ key: "query_version_limit", value: n }];
}

function importDraft(i: Record<string, unknown>): Record<string, unknown> {
  const d: Record<string, unknown> = {
    projectId: "p1",
    name: i.name,
    type: i.type,
    host: i.host,
    port: i.port,
    databaseName: i.databaseName,
    username: i.username,
  };
  if (i.sslMode !== undefined) d.sslMode = i.sslMode;
  if (i.sshTunnel !== undefined) d.sshTunnel = i.sshTunnel;
  d.labelIds = [];
  d.isLocalOnly = true;
  d.renameIfTaken = true;
  return d;
}

/**
 * `importConnections`. `sent` lists the drafts a GUI on Core creates: not the
 * ones it skips as already saved, nor the injected failure. Each is sent with
 * its own name and `renameIfTaken`; Core picks the free name.
 */
function importStep(
  drafts: Record<string, unknown>[],
  sent: Record<string, unknown>[],
  failName?: string,
): Step {
  return {
    op: "connections.importConnections",
    args: { drafts },
    ...(failName ? { inject: `the save of "${failName}" fails: STORAGE_ERROR: disk full` } : {}),
    library: sent.map((d) => ({
      method: "connectionCreate",
      params: { connection: importDraft(d) },
    })),
    run: async (t) => {
      if (failName) {
        const storage = rec.storage as StorageClient;
        const save = storage.connections.save;
        storage.connections.save = async (c) => {
          if (c.name === failName) throw new Error("STORAGE_ERROR: disk full");
          return save(c);
        };
        try {
          return await t.connections.importConnections(drafts as never);
        } finally {
          storage.connections.save = save;
        }
      }
      return await t.connections.importConnections(drafts as never);
    },
  };
}

/** The `connectionCreate` for a shared template imported into `projectId`. */
function sharedCreate(projectId: string, which: "prod" | "bastion"): Library {
  const base = {
    projectId,
    type: "postgres",
    host: "db.internal",
    port: 5432,
    databaseName: "app",
    username: "",
    labelIds: [],
    renameIfTaken: true,
  };
  return {
    method: "connectionCreate",
    params: {
      connection:
        which === "prod"
          ? { ...base, name: "Prod", sslMode: "require", sharedConnectionId: "repo-1:prod" }
          : {
              ...base,
              name: "Behind bastion",
              sshTunnel: {
                enabled: true,
                host: "bastion",
                port: 22,
                username: "",
                authMethod: "key",
              },
              sharedConnectionId: "repo-1:bastion",
            },
    },
  };
}
