// ---------------------------------------------------------------- the mock provider

interface Script {
  /** Answered by the fetch stub without a request: `fetch` throws `TypeError("fetch failed")`. */
  networkError?: boolean;
  status?: number;
  contentType?: string;
  body: string;
  /** Write the body, then keep the response open until the client closes it. */
  stall?: boolean;
}

const mock = {
  server: null as Server | null,
  port: 0,
  queue: [] as Script[],
  /** Per response written: whether the client closed it before the server ended it. */
  closedEarly: [] as boolean[],
  unscripted: 0,
};

function handle(_req: IncomingMessage, res: ServerResponse) {
  const script = mock.queue.shift();
  if (!script) {
    mock.unscripted++;
    res.writeHead(599, { "content-type": "text/plain" });
    res.end("unscripted request");
    return;
  }
  const index = mock.closedEarly.length;
  mock.closedEarly.push(false);
  let ended = false;
  res.on("close", () => {
    if (!ended) mock.closedEarly[index] = true;
  });
  res.writeHead(script.status ?? 200, {
    "content-type": script.contentType ?? "text/event-stream",
    "cache-control": "no-cache",
  });
  const bytes = Buffer.from(script.body, "utf8");
  let at = 0;
  const step = () => {
    if (res.destroyed) return;
    if (at < bytes.length) {
      res.write(bytes.subarray(at, at + PIECE));
      at += PIECE;
      setImmediate(step);
      return;
    }
    if (script.stall) return;
    ended = true;
    res.end();
  };
  step();
}

const realFetch = globalThis.fetch;
let requests: Rec[] = [];

function normHeaders(h: unknown): Record<string, string> {
  const out: Record<string, string> = {};
  const entries = h instanceof Headers ? [...h.entries()] : Object.entries((h ?? {}) as Rec);
  for (const [k, v] of entries) out[k.toLowerCase()] = String(v).split(KEY).join("<key>");
  return Object.fromEntries(Object.entries(out).sort(([a], [b]) => (a < b ? -1 : 1)));
}

async function stubFetch(input: unknown, init?: RequestInit): Promise<Response> {
  const url = String(input);
  const body = typeof init?.body === "string" ? (JSON.parse(init.body) as Json) : null;
  requests.push({
    method: init?.method ?? "GET",
    url,
    headers: normHeaders(init?.headers),
    body,
  });
  if (mock.queue[0]?.networkError) {
    mock.queue.shift();
    throw new TypeError("fetch failed");
  }
  const target = `http://127.0.0.1:${mock.port}/`;
  if (!target.startsWith("http://127.0.0.1:")) throw new Error("refusing a non-local request");
  return realFetch(target, {
    method: init?.method ?? "GET",
    headers: init?.headers,
    body: init?.body,
    signal: init?.signal,
  });
}

// ---------------------------------------------------------------- scripted streams

const sseEvent = (event: string | null, data: unknown) =>
  (event ? `event: ${event}\n` : "") +
  `data: ${typeof data === "string" ? data : JSON.stringify(data)}\n\n`;

/** Pieces of `s` of `n` characters (tool input JSON arrives in pieces). */
const pieces = (s: string, n: number) => {
  const out: string[] = [];
  for (let i = 0; i < s.length; i += n) out.push(s.slice(i, i + n));
  return out;
};

/** A tool input as the model streams it: JSON, or `{ $raw }` sent as is (malformed input). */
const rawJson = (input: unknown) =>
  typeof input === "object" && input !== null && "$raw" in input
    ? String((input as { $raw: string }).$raw)
    : JSON.stringify(input);

type Block = { text: string } | { tool: { id: string; name: string; input: unknown } };

/** An Anthropic Messages stream, as the API documents it. */
function anthropic(
  blocks: Block[],
  stop: string,
  opts: { before?: string; after?: string; cut?: boolean } = {},
): string {
  let s = sseEvent("message_start", {
    type: "message_start",
    message: {
      id: "msg_01",
      type: "message",
      role: "assistant",
      model: "model-1",
      content: [],
      stop_reason: null,
      stop_sequence: null,
      usage: { input_tokens: 25, output_tokens: 1 },
    },
  });
  s += sseEvent("ping", { type: "ping" });
  blocks.forEach((b, index) => {
    if ("text" in b) {
      s += sseEvent("content_block_start", {
        type: "content_block_start",
        index,
        content_block: { type: "text", text: "" },
      });
      for (const t of b.text.split(/(?<= )/)) {
        s += sseEvent("content_block_delta", {
          type: "content_block_delta",
          index,
          delta: { type: "text_delta", text: t },
        });
      }
    } else {
      s += sseEvent("content_block_start", {
        type: "content_block_start",
        index,
        content_block: { type: "tool_use", id: b.tool.id, name: b.tool.name, input: {} },
      });
      for (const p of pieces(rawJson(b.tool.input), 7)) {
        s += sseEvent("content_block_delta", {
          type: "content_block_delta",
          index,
          delta: { type: "input_json_delta", partial_json: p },
        });
      }
    }
    s += sseEvent("content_block_stop", { type: "content_block_stop", index });
  });
  if (opts.cut) return s;
  if (opts.before) s += opts.before;
  s += sseEvent("message_delta", {
    type: "message_delta",
    delta: { stop_reason: stop, stop_sequence: null },
    usage: { output_tokens: 42 },
  });
  s += sseEvent("message_stop", { type: "message_stop" });
  if (opts.after) s += opts.after;
  return s;
}

const oaiChunk = (delta: unknown, finish: string | null = null) => ({
  id: "chatcmpl-1",
  object: "chat.completion.chunk",
  created: 1700000000,
  model: "model-1",
  choices: [{ index: 0, delta, finish_reason: finish }],
});

/**
 * An OpenAI Chat Completions stream. Tool calls' arguments arrive in pieces,
 * interleaved across calls when there are several.
 */
function openai(
  text: string,
  calls: { id: string; name: string; input: unknown }[],
  finish: string,
  opts: { before?: string; afterDone?: string; cut?: boolean; earlyDone?: boolean } = {},
): string {
  let s = sseEvent(null, oaiChunk({ role: "assistant", content: "" }));
  for (const t of text ? text.split(/(?<= )/) : []) s += sseEvent(null, oaiChunk({ content: t }));
  if (opts.earlyDone) s += sseEvent(null, "[DONE]");
  calls.forEach((c, index) => {
    s += sseEvent(
      null,
      oaiChunk({
        tool_calls: [{ index, id: c.id, type: "function", function: { name: c.name, arguments: "" } }],
      }),
    );
  });
  const parts = calls.map((c) => pieces(rawJson(c.input), 7));
  const longest = Math.max(0, ...parts.map((p) => p.length));
  for (let i = 0; i < longest; i++) {
    parts.forEach((p, index) => {
      if (i < p.length) {
        s += sseEvent(null, oaiChunk({ tool_calls: [{ index, function: { arguments: p[i] } }] }));
      }
    });
  }
  if (opts.cut) return s;
  if (opts.before) s += opts.before;
  s += sseEvent(null, oaiChunk({}, finish));
  s += sseEvent(null, "[DONE]");
  if (opts.afterDone) s += opts.afterDone;
  return s;
}

/**
 * The large schema (the README gives the same rule): tables `t_0001` …
 * `t_3000` in schema `big`, each with columns `c1` … `c6` of type `integer`,
 * `c<j>` nullable when j is even, and no indexes.
 */
const LARGE = { tables: 3000, schema: "big", columns: 6 };
function largeSchema(): SchemaTable[] {
  const out: SchemaTable[] = [];
  for (let i = 1; i <= LARGE.tables; i++) {
    const columns = [];
    for (let j = 1; j <= LARGE.columns; j++) columns.push(col(`c${j}`, "integer", j % 2 === 0));
    out.push({
      name: `t_${String(i).padStart(4, "0")}`,
      schema: LARGE.schema,
      type: "table",
      columns,
      indexes: [],
    });
  }
  return out;
}

// ---------------------------------------------------------------- query answers

/** A read-only stream's answer: one final batch in the wire format, or an error. */
type Answer =
  | { columns: string[]; rows: Json[][]; truncated?: boolean }
  | { error: string }
  | { cancel: true };

const DEFAULT_ANSWER: Answer = { columns: ["n"], rows: [[1]] };

/** Runs `answer` through the real `collectReadOnly`, as `CoreProvider.selectReadOnly` does. */
async function answerQuery(answer: Answer, signal?: AbortSignal) {
  return collectReadOnly(async (onBatch) => {
    if ("error" in answer) return { aborted: false, error: answer.error };
    if ("cancel" in answer) {
      await new Promise<void>((resolve) => {
        if (signal?.aborted) resolve();
        signal?.addEventListener("abort", () => resolve(), { once: true });
      });
      return { aborted: true };
    }
    const rows = structuredClone(answer.rows).map((r) => r.map((v) => decodeCell(v)));
    onBatch({ columns: answer.columns, rows, isFinal: true, truncated: answer.truncated });
    return { aborted: false };
  }, signal);
}

/** JSON with sorted keys, for comparing values. */
function canon(v: unknown): string {
  return JSON.stringify(v, (_k, x: unknown) =>
    x && typeof x === "object" && !Array.isArray(x)
      ? Object.fromEntries(Object.entries(x as Rec).sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)))
      : x,
  );
}

/**
 * One changes.json entry. With `record`, a field whose expected value equals
 * the recording is dropped (so every entry names a real difference), unless
 * listed in `always`; nothing left, no entry.
 */
function change(
  key: string,
  expected: Rec,
  record?: Rec,
  opts: { notes?: string[]; always?: string[]; decisions?: string[] } = {},
) {
  if (changes[key]) throw new Error(`two changes for ${key}`);
  let exp = expected;
  if (record) {
    exp = Object.fromEntries(
      Object.entries(expected).filter(([k, v]) => opts.always?.includes(k) || canon(v) !== canon(record[k])),
    );
  }
  if (Object.keys(exp).length === 0) return;
  const decisions = new Set<string>(opts.decisions ?? []);
  const why: string[] = [...(opts.notes ?? [])];
  for (const k of Object.keys(exp)) {
    const r = REASONS[k];
    if (!r) throw new Error(`no reason for ${k}`);
    for (const d of r[0]) decisions.add(d);
    why.push(r[1]);
  }
  changes[key] = { decision: [...decisions].join(", "), why: why.join(" "), expected: exp };
}

const utf8 = (s: string) => Buffer.byteLength(s, "utf8");

/** Bug 7, Decision 13: schema-qualified names, cut at whole tables at 128 KiB. */
function rustSchemaContext(tables: SchemaTable[]): { text: string; kept: number; left: number } {
  if (tables.length === 0) return { text: "", kept: 0, left: 0 };
  let text = "Database schema:";
  let kept = 0;
  for (const table of tables) {
    const lines = [`\nTable: ${table.schema}.${table.name}`];
    for (const c of table.columns) lines.push(`  ${c.name} ${c.type}${c.nullable ? "" : " NOT NULL"}`);
    for (const idx of table.indexes ?? []) lines.push(`  INDEX ${idx.name} (${idx.columns.join(", ")})`);
    const block = "\n" + lines.join("\n");
    if (utf8(text + block) > SCHEMA_CAP) break;
    text += block;
    kept++;
  }
  const left = tables.length - kept;
  if (left > 0) text += `\n\n${schemaNote(left)}`;
  return { text, kept, left };
}

const MAX_CELL_BYTES = 64 * 1024;
const MAX_FETCH_BYTES = 8 * 1024 * 1024;
const CALL_TIMEOUT_MS = 60_000;
/** Decision 23: a stored tool result is cut at 16 KiB. */
const PART_RESULT_BYTES = 16 * 1024;
/** Decision 29: the history budget. */

const ASSISTANT_RESULT_BYTES = 256 * 1024;
/** Decision 10: the provider's message, cut at 1 KiB. */
const PROVIDER_MESSAGE_BYTES = 1024;

/** A prefix of `s` of at most `max` UTF-8 bytes, cut at a character boundary. */
function utf8Prefix(s: string, max: number): string {
  const b = Buffer.from(s, "utf8");
  if (b.length <= max) return s;
  let end = max;
  while (end > 0 && (b[end] & 0xc0) === 0x80) end--;
  return b.subarray(0, end).toString("utf8");
}

/** JSON with every object's keys sorted, as serde_json's map writes it. */
const sortedJson = (v: unknown) => canon(v);

/** A query as Core runs it for a tool (Decision 4, bug 14). */
const rustQuery = (sql: string, maxRows = RUN_QUERY_DEFAULT_ROWS): Rec => ({
  sql,
  maxRows,
  maxBytes: MAX_FETCH_BYTES,
  timeoutMs: CALL_TIMEOUT_MS,
});

/** MCP's cell rules (`format.rs`): plain JSON stays, everything else is `cellText`; text past 64 KB is cut. */
function rustCell(v: Json): { cell: Json; cut: boolean } {
  if (v === null || typeof v === "boolean" || typeof v === "number") return { cell: v, cut: false };
  const text = typeof v === "string" ? v : cellText(decodeCell(v));
  const bytes = utf8(text);
  if (bytes > MAX_CELL_BYTES) {
    return { cell: { bytes, text: utf8Prefix(text, MAX_CELL_BYTES), truncated: true }, cut: true };
  }
  return { cell: text, cut: false };
}

/**
 * Q4: `run_query`'s result in the assistant profile: MCP's JSON (keys
 * sorted, as serde_json's map writes them), `maxRows` rows (100 unless the
 * model passes `max_rows`), 256 KB, the notes worded in KB.
 */
function rustRunQuery(
  a: { columns: string[]; rows: Json[][]; truncated?: boolean },
  maxRows = RUN_QUERY_DEFAULT_ROWS,
): string {
  const fetched = a.rows.slice(0, maxRows);
  const truncated = a.rows.length > maxRows || a.truncated === true;
  let bytes = utf8(JSON.stringify(a.columns));
  const rows: Json[] = [];
  let tooBig = false;
  let cutCells = 0;
  for (const row of fetched) {
    let cut = 0;
    const cells = row.map((v) => {
      const c = rustCell(v);
      if (c.cut) cut++;
      return c.cell;
    });
    const size = utf8(JSON.stringify(cells)) + 1;
    if (bytes + size > ASSISTANT_RESULT_BYTES) {
      tooBig = true;
      break;
    }
    bytes += size;
    cutCells += cut;
    rows.push(cells);
  }
  const notes: string[] = [];
  if (tooBig) {
    notes.push(
      rows.length === 0
        ? "No rows are shown: the first row alone would take the result past the 256 KB limit. Select fewer or narrower columns."
        : `Only the first ${rows.length} rows are shown: the next would take the result past the 256 KB limit. Select fewer or narrower columns, or fewer rows.`,
    );
  } else if (truncated && rows.length < maxRows) {
    notes.push(
      `Only the first ${rows.length} rows are shown: the rows fetched reached the 8 MB limit on what a query may read. Select fewer or narrower columns, or fewer rows.`,
    );
  } else if (truncated) {
    notes.push(
      `Only the first ${maxRows} rows are shown; the query returned more. Narrow it or raise max_rows (at most 1000).`,
    );
  }
  if (cutCells > 0) {
    notes.push(
      `${cutCells} cell(s) were longer than 64 KB and are cut: each is an object {"truncated": true, "bytes": <full length>, "text": <its start>}. Select a substring to see more.`,
    );
  }
  const out: Rec = { columns: a.columns };
  if (notes.length > 0) out.message = notes.join(" ");
  out.rowCount = rows.length;
  out.rows = rows;
  out.truncated = truncated || tooBig;
  if (cutCells > 0) out.truncatedCells = cutCells;
  const text = JSON.stringify(out);
  if (utf8(text) > ASSISTANT_RESULT_BYTES) throw new Error("model: a result past the assistant budget");
  return text;
}

const expectedList = (names: string[]) =>
  names.length === 0
    ? "there are no fields"
    : names.length === 1
      ? `expected \`${names[0]}\``
      : names.length === 2
        ? `expected \`${names[0]}\` or \`${names[1]}\``
        : `expected one of ${names.map((n) => `\`${n}\``).join(", ")}`;

/** serde's name for an unexpected JSON value. */
function unexpected(v: unknown): string {
  if (typeof v === "string") return `string ${JSON.stringify(v)}`;
  if (typeof v === "number") return Number.isInteger(v) ? `integer \`${v}\`` : `floating point \`${v}\``;
  if (typeof v === "boolean") return `boolean \`${v}\``;
  if (Array.isArray(v)) return "sequence";
  if (v !== null && typeof v === "object") return "map";
  throw new Error("model: null arguments aren't modelled");
}

const isMap = (v: unknown): v is Rec => v !== null && typeof v === "object" && !Array.isArray(v);

/**
 * Decision 24: serde's message for the first problem, keys visited in
 * sorted order (serde_json's map), missing fields checked after in
 * declaration order; `<path>: ` before it unless the problem is the root
 * object's own (a missing or unknown field).
 */
function checkArgs(v: unknown, t: Ty, path: string[] = []): string | null {
  const at = (msg: string) => (path.length > 0 ? `${path.join(".")}: ${msg}` : msg);
  switch (t.k) {
    case "any":
      return null;
    case "str":
      return typeof v === "string" ? null : at(`invalid type: ${unexpected(v)}, expected a string`);
    case "f64":
      return typeof v === "number" ? null : at(`invalid type: ${unexpected(v)}, expected f64`);
    case "u32":
      return typeof v === "number" && Number.isInteger(v) && v >= 0 && v <= 4294967295
        ? null
        : at(`invalid type: ${unexpected(v)}, expected u32`);
    case "enum":
      if (typeof v !== "string") throw new Error("model: a non-string enum value");
      return t.v.includes(v) ? null : at(`unknown variant \`${v}\`, ${expectedList(t.v)}`);
    case "arr": {
      if (!Array.isArray(v)) return at(`invalid type: ${unexpected(v)}, expected a sequence`);
      for (let i = 0; i < v.length; i++) {
        const e = checkArgs(v[i], t.of, [...path, String(i)]);
        if (e) return e;
      }
      return null;
    }
    case "map": {
      if (!isMap(v)) throw new Error("model: a non-object map");
      for (const k of Object.keys(v).sort()) {
        const e = checkArgs(v[k], t.of, [...path, k]);
        if (e) return e;
      }
      return null;
    }
    case "obj": {
      if (!isMap(v)) throw new Error("model: a non-object struct");
      const names = t.fields.map((f) => f[0]);
      for (const k of Object.keys(v).sort()) {
        const f = t.fields.find((x) => x[0] === k);
        if (!f) return at(`unknown field \`${k}\`, ${expectedList(names)}`);
        if (v[k] === null && !f[2]) continue;
        const e = checkArgs(v[k], f[1], [...path, k]);
        if (e) return e;
      }
      for (const [n, , required] of t.fields) {
        if (required && !(n in v)) return at(`missing field \`${n}\``);
      }
      return null;
    }
  }
}

/** MCP's EXPLAIN text (`format::explain_text`) for the nodes the cases use. */
function explainText(r: Rec): string {
  const fixed2 = (n: number) => n.toFixed(2);
  let out = "";
  const node = (n: Rec, depth: number) => {
    const indent = "  ".repeat(depth);
    out += `${indent}${String(n.nodeType)}`;
    if (n.relationName) {
      out += ` on ${String(n.relationName)}`;
      if (n.alias && n.alias !== n.relationName) out += ` ${String(n.alias)}`;
    }
    if (n.indexName) out += ` using ${String(n.indexName)}`;
    const est: string[] = [];
    if (typeof n.startupCost === "number" && typeof n.totalCost === "number") {
      est.push(`cost=${fixed2(n.startupCost)}..${fixed2(n.totalCost)}`);
    } else if (typeof n.totalCost === "number") est.push(`cost=${fixed2(n.totalCost)}`);
    if (typeof n.planRows === "number") est.push(`rows=${n.planRows}`);
    if (typeof n.planWidth === "number") est.push(`width=${n.planWidth}`);
    if (est.length > 0) out += ` (${est.join(" ")})`;
    out += "\n";
    for (const [label, key] of [
      ["Join type", "joinType"],
      ["Index cond", "indexCond"],
      ["Hash cond", "hashCond"],
      ["Filter", "filter"],
    ]) {
      if (n[key]) out += `${indent}    ${label}: ${String(n[key])}\n`;
    }
    for (const child of (n.children ?? []) as Rec[]) node(child, depth + 1);
  };
  node(r.plan as Rec, 0);
  if ((r.planningTime as number) > 0) out += `Planning time: ${r.planningTime as number} ms\n`;
  return out.trimEnd();
}

function rustToolResult(name: string, input: Rec, ctx: ToolCtx): RustResult {
  const isClient = DASHBOARD_TOOLS.includes(name);
  const known = DATA_TOOLS.includes(name) || SCHEMA_TOOLS.includes(name) || (ctx.clientTools && isClient);
  if (!known) return rustError("INVALID_ARGUMENT", `Unknown tool: ${name}`);
  if (DATA_TOOLS.includes(name) && !ctx.sharing.data) return dataOff(ctx.connectionName);
  if (SCHEMA_TOOLS.includes(name) && !ctx.sharing.schema) return schemaOff(ctx.connectionName);
  const bad = checkArgs(input, ARGS[name]);
  if (bad) return rustError("INVALID_ARGUMENT", bad);
  const runRows = (sql: string, maxRows: number): RustResult => {
    ctx.queries.push(rustQuery(sql, maxRows));
    const a = ctx.answers?.[sql] ?? DEFAULT_ANSWER;
    if ("error" in a) {
      const i = a.error.indexOf(": ");
      return rustError(a.error.slice(0, i), a.error.slice(i + 2));
    }
    if ("cancel" in a) throw new Error("model: a cancelled query in a turn");
    return { text: rustRunQuery(a, maxRows), isError: false };
  };
  const tables = ctx.tables ?? SCHEMA;
  switch (name) {
    case "run_query": {
      const maxRows = (input.max_rows as number | undefined) ?? RUN_QUERY_DEFAULT_ROWS;
      if (maxRows < 1 || maxRows > 1000) {
        return rustError("INVALID_ARGUMENT", `max_rows must be between 1 and 1000, got ${maxRows}`);
      }
      const sql = String(input.sql);
      if (readOnlyError(sql, ctx.engine)) return rustError("READ_ONLY", READ_ONLY);
      if (!ctx.approve(sql)) return DENIED;
      return runRows(sql, maxRows);
    }
    case "explain_query": {
      const sql = String(input.sql);
      if (readOnlyError(sql, ctx.engine)) return rustError("READ_ONLY", READ_ONLY);
      if (!ctx.approve(sql)) return DENIED;
      if (!ctx.rust?.explain) throw new Error("model: explain needs its plan");
      const plan = explainText(ctx.rust.explain);
      if (utf8(plan) > ASSISTANT_RESULT_BYTES) {
        // Decision 30: MCP's cut, at the assistant's 256 KB, worded in KB.
        return {
          text: sortedJson({ plan: utf8Prefix(plan, ASSISTANT_RESULT_BYTES), truncated: true, message: "The plan is cut at 256 KB." }),
          isError: false,
        };
      }
      return { text: sortedJson({ plan }), isError: false };
    }
    case "list_schemas":
      return { text: sortedJson({ schemas: ctx.rust?.schemas ?? [] }), isError: false };
    case "list_tables": {
      const want = input.schema as string | undefined;
      const list = tables
        .filter((t) => want === undefined || t.schema === want)
        .map((t) => ({ schema: t.schema, name: t.name, type: t.type }));
      return { text: sortedJson({ tables: list }), isError: false };
    }
    case "describe_table": {
      const want = input.schema as string | undefined;
      const tname = String(input.table);
      const matches = tables.filter((t) => t.name === tname && (want === undefined || t.schema === want));
      if (matches.length === 0) {
        return rustError(
          "TABLE_NOT_FOUND",
          want === undefined
            ? `No table ${JSON.stringify(tname)}`
            : `No table ${JSON.stringify(tname)} in schema ${JSON.stringify(want)}`,
        );
      }
      if (matches.length > 1) {
        return rustError(
          "AMBIGUOUS_TABLE",
          `Tables named ${JSON.stringify(tname)} exist in several schemas (${matches.map((t) => JSON.stringify(t.schema)).join(", ")}); pass \`schema\``,
        );
      }
      const t = matches[0];
      return {
        text: sortedJson({
          schema: t.schema,
          table: tname,
          type: t.type,
          columns: t.columns.map((c) => ({
            name: c.name,
            type: c.type,
            nullable: c.nullable,
            default: c.defaultValue ?? null,
            primaryKey: c.isPrimaryKey,
          })),
          indexes: (t.indexes ?? []).map((i) => ({ name: i.name, columns: i.columns, unique: i.unique, type: i.type })),
          foreignKeys: t.columns
            .filter((c) => c.foreignKeyRef)
            .map((c) => ({
              column: c.name,
              referencedSchema: c.foreignKeyRef!.referencedSchema,
              referencedTable: c.foreignKeyRef!.referencedTable,
              referencedColumn: c.foreignKeyRef!.referencedColumn,
            })),
        }),
        isError: false,
      };
    }
    case "list_saved_queries": {
      const list = (ctx.saved ?? SAVED).map((q) => ({
        id: q.id,
        name: q.name,
        project: ctx.rust?.project ?? "Main",
        connections: [ctx.connectionName],
        parameters: [],
      }));
      return { text: sortedJson({ savedQueries: list }), isError: false };
    }
    case "run_saved_query": {
      const want = String(input.saved_query);
      const all = ctx.saved ?? SAVED;
      const byId = all.filter((q) => q.id === want);
      const found = byId.length > 0 ? byId : all.filter((q) => q.name === want);
      if (found.length === 0) {
        return rustError(
          "SAVED_QUERY_NOT_FOUND",
          `No saved query ${JSON.stringify(want)} in project ${JSON.stringify(ctx.rust?.project ?? "Main")}, the project of connection ${JSON.stringify(ctx.connectionName)}`,
        );
      }
      const maxRows = (input.max_rows as number | undefined) ?? RUN_QUERY_DEFAULT_ROWS;
      if (maxRows < 1 || maxRows > 1000) {
        return rustError("INVALID_ARGUMENT", `max_rows must be between 1 and 1000, got ${maxRows}`);
      }
      const sql = found[0].query;
      if (!ctx.approve(sql)) return DENIED;
      return runRows(sql, maxRows);
    }
  }
  // A client tool: Core checks a widget's query, then the page answers (Decision 6).
  if ((name === "add_widget" || name === "update_widget") && typeof input.query === "string") {
    if (input.query.trim() && readOnlyError(input.query, ctx.engine)) return rustError("READ_ONLY", READ_ONLY);
  }
  ctx.onForward?.();
  if (ctx.pageAnswer === undefined) throw new Error(`model: no page answer for ${name}`);
  const parsed = JSON.parse(ctx.pageAnswer) as Rec;
  const isError = isMap(parsed) && "error" in parsed;
  if (name === "get_dashboard" && !ctx.sharing.schema && !isError) {
    // Decision 26 (e): Core strips the widget queries from the page's answer.
    for (const w of (parsed.widgets ?? []) as Rec[]) delete w.query;
    return { text: sortedJson(parsed), isError };
  }
  return { text: ctx.pageAnswer, isError };
}

// ---- parts and history (Q7, Decisions 12, 23, 29) ----

type Part =
  | { round: number; type: "text"; text: string }
  | {
      round: number;
      type: "tool";
      callId: string;
      name: string;
      input: Json;
      ok: boolean;
      result: string;
      resultBytes?: number;
    };
type ToolPart = Extract<Part, { type: "tool" }>;

function toolPart(round: number, callId: string, name: string, input: Json, res: RustResult): ToolPart {
  const bytes = utf8(res.text);
  const p: ToolPart = {
    round,
    type: "tool",
    callId,
    name,
    input,
    ok: !res.isError,
    result: bytes > PART_RESULT_BYTES ? utf8Prefix(res.text, PART_RESULT_BYTES) : res.text,
  };
  if (bytes > PART_RESULT_BYTES) p.resultBytes = bytes;
  return p;
}

/** A stored tool result as history sends it back (Decision 23). */
const partResultText = (p: ToolPart) =>
  p.resultBytes === undefined ? p.result : `${p.result}\n(cut at 16 KB of ${p.resultBytes} bytes)`;

const roundsOf = (parts: Part[]) => {
  const rounds = new Map<number, Part[]>();
  for (const p of parts) rounds.set(p.round, [...(rounds.get(p.round) ?? []), p]);
  return [...rounds.entries()].sort(([a], [b]) => a - b);
};

/** One round's messages in the provider's shape. */
function roundMessages(text: string, tools: ToolPart[], shape: "anthropic" | "openai"): Rec[] {
  if (tools.length === 0) return text ? [{ role: "assistant", content: text }] : [];
  if (shape === "anthropic") {
    return [
      {
        role: "assistant",
        content: [
          ...(text ? [{ type: "text", text }] : []),
          ...tools.map((t) => ({ type: "tool_use", id: t.callId, name: t.name, input: t.input })),
        ],
      },
      {
        role: "user",
        content: tools.map((t) => ({
          type: "tool_result",
          tool_use_id: t.callId,
          content: partResultText(t),
          ...(t.ok ? {} : { is_error: true }),
        })),
      },
    ];
  }
  return [
    {
      role: "assistant",
      ...(text ? { content: text } : {}),
      tool_calls: tools.map((t) => ({
        id: t.callId,
        type: "function",
        function: { name: t.name, arguments: sortedJson(t.input) },
      })),
    },
    ...tools.map((t) => ({
      role: "tool",
      tool_call_id: t.callId,
      content: t.ok ? partResultText(t) : `Error: ${partResultText(t)}`,
    })),
  ];
}

/** Stored rows rendered back as history (Decision 23); `keep` limits a reply to some rounds (Decision 29). */
function renderHistory(rows: Rec[], shape: "anthropic" | "openai"): Rec[] {
  const out: Rec[] = [];
  for (const row of rows) {
    const parts = row.parts as Part[] | undefined;
    if (!parts) {
      out.push({ role: row.role, content: row.content });
      continue;
    }
    for (const [, items] of roundsOf(parts)) {
      const text = items
        .filter((p): p is Extract<Part, { type: "text" }> => p.type === "text")
        .map((p) => p.text)
        .join("");
      out.push(...roundMessages(text, items.filter((p): p is ToolPart => p.type === "tool"), shape));
    }
  }
  return out;
}

/** Decision 29: what a stored row costs against the history budget. */
function partsCost(items: Part[]): number {
  let n = 0;
  for (const p of items) n += p.type === "text" ? utf8(p.text) : utf8(sortedJson(p.input)) + utf8(partResultText(p));
  return n;
}

/**
 * Decision 29: turns newest first, whole while they fit; in the first turn
 * that doesn't, its user message and its reply's latest whole rounds that
 * fit, then nothing older. A round is never split, so a tool call is never
 * sent without its result.
 */
function historyBudget(rows: Rec[], budget = HISTORY_BYTES): { kept: Rec[]; bytes: number } {
  const turns: Rec[][] = [];
  for (const row of rows) {
    if (row.role === "user" || turns.length === 0) turns.push([row]);
    else turns[turns.length - 1].push(row);
  }
  const rowCost = (r: Rec) => (r.parts ? partsCost(r.parts as Part[]) : utf8(String(r.content)));
  let total = 0;
  const kept: Rec[][] = [];
  for (let t = turns.length - 1; t >= 0; t--) {
    const turn = turns[t];
    const cost = turn.reduce((n, r) => n + rowCost(r), 0);
    if (total + cost <= budget) {
      kept.unshift(turn.map((r) => ({ id: r.id })));
      total += cost;
      continue;
    }
    const [user, ...replies] = turn;
    const reply = replies.at(-1);
    if (user.role === "user" && replies.length === 1 && reply?.parts) {
      let used = total + rowCost(user);
      const rounds = roundsOf(reply.parts as Part[]);
      const keptRounds: number[] = [];
      for (let i = rounds.length - 1; i >= 0; i--) {
        const c = partsCost(rounds[i][1]);
        if (used + c > budget) break;
        used += c;
        keptRounds.unshift(rounds[i][0]);
      }
      if (keptRounds.length > 0) {
        kept.unshift([{ id: user.id }, { id: reply.id, rounds: keptRounds }]);
        total = used;
      }
    }
    break;
  }
  return { kept: kept.flat(), bytes: total };
}

// ---- decoding the scripted responses (what Task 2's decoder does with them) ----

interface Decoded {
  text: string;
  calls: { id: string; name: string; input: Json }[];
  stop: "end" | "maxTokens" | "toolUse";
  error?: { code: string; message: string };
}

function providerMessage(body: string): string {
  let message = body.trim();
  try {
    const j = JSON.parse(body) as Rec;
    const e = (j.error ?? {}) as Rec;
    if (typeof e.message === "string") message = e.message;
  } catch {
    /* not JSON: the text */
  }
  return utf8Prefix(message, PROVIDER_MESSAGE_BYTES);
}

const MALFORMED_EVENT = "The provider sent an event that isn't valid JSON.";
const MALFORMED_INPUT = "The provider sent a tool call whose input isn't a JSON object.";
const INCOMPLETE = "The provider's stream ended before the reply did.";
const UNREACHABLE = "Could not reach the provider.";

function decodeScript(script: Script, shape: "anthropic" | "openai"): Decoded {
  const out: Decoded = { text: "", calls: [], stop: "end" };
  if (script.networkError) return { ...out, error: { code: "PROVIDER_ERROR", message: UNREACHABLE } };
  const status = script.status ?? 200;
  if (status < 200 || status > 299) {
    return {
      ...out,
      error: { code: status === 429 ? "RATE_LIMITED" : "PROVIDER_ERROR", message: providerMessage(script.body) },
    };
  }
  const open: { id: string; name: string; json: string }[] = [];
  let stopReason: string | null = null;
  let ended = false;
  const finishCalls = () => {
    for (const c of open) {
      let input: unknown;
      try {
        input = JSON.parse(c.json || "{}");
      } catch {
        return MALFORMED_INPUT;
      }
      if (!isMap(input)) return MALFORMED_INPUT;
      out.calls.push({ id: c.id, name: c.name, input: input as Json });
    }
    return null;
  };
  for (const block of script.body.split("\n\n")) {
    const data = block
      .split("\n")
      .filter((l) => l.startsWith("data:"))
      .map((l) => l.slice(5).replace(/^ /, ""))
      .join("\n");
    if (!data) continue;
    if (data === "[DONE]") {
      ended = true;
      break;
    }
    let ev: Rec;
    try {
      ev = JSON.parse(data) as Rec;
    } catch {
      return { ...out, error: { code: "PROVIDER_ERROR", message: MALFORMED_EVENT } };
    }
    if (shape === "anthropic") {
      if (ev.type === "error") {
        return { ...out, error: { code: "PROVIDER_ERROR", message: utf8Prefix(String((ev.error as Rec).message), PROVIDER_MESSAGE_BYTES) } };
      }
      if (ev.type === "content_block_start" && (ev.content_block as Rec).type === "tool_use") {
        const b = ev.content_block as Rec;
        open.push({ id: String(b.id), name: String(b.name), json: "" });
      }
      if (ev.type === "content_block_delta") {
        const d = ev.delta as Rec;
        if (d.type === "text_delta") out.text += String(d.text);
        if (d.type === "input_json_delta") open[open.length - 1].json += String(d.partial_json);
      }
      if (ev.type === "message_delta") stopReason = String((ev.delta as Rec).stop_reason);
      if (ev.type === "message_stop") {
        ended = true;
        break;
      }
    } else {
      if (ev.error) {
        return { ...out, error: { code: "PROVIDER_ERROR", message: utf8Prefix(String((ev.error as Rec).message), PROVIDER_MESSAGE_BYTES) } };
      }
      const choice = ((ev.choices ?? []) as Rec[])[0];
      const d = (choice?.delta ?? {}) as Rec;
      if (typeof d.content === "string") out.text += d.content;
      for (const tc of (d.tool_calls ?? []) as Rec[]) {
        const i = tc.index as number;
        open[i] ??= { id: "", name: "", json: "" };
        if (tc.id) open[i].id = String(tc.id);
        const fn = (tc.function ?? {}) as Rec;
        if (fn.name) open[i].name = String(fn.name);
        if (fn.arguments) open[i].json += String(fn.arguments);
      }
      if (choice?.finish_reason) stopReason = String(choice.finish_reason);
    }
  }
  if (!ended && !(shape === "anthropic" && stopReason)) {
    return { ...out, error: { code: "PROVIDER_ERROR", message: INCOMPLETE } };
  }
  const bad = finishCalls();
  if (bad) return { ...out, calls: [], error: { code: "PROVIDER_ERROR", message: bad } };
  out.stop =
    stopReason === "max_tokens" || stopReason === "length"
      ? "maxTokens"
      : out.calls.length > 0
        ? "toolUse"
        : "end";
  return out;
}

// ---- one turn as Core runs it ----

interface SimInput {
  shape: "anthropic" | "openai";
  url: string;
  headers: Record<string, string>;
  engine: DatabaseType;
  tables: SchemaTable[];
  /** Sharing at the turn's start (prompt and tools). */
  sharing: Sharing;
  /** Sharing as read before the n-th tool call (0-based). */
  sharingAt?: (n: number) => Sharing;
  clientTools: boolean;
  connectionName: string;
  history: Rec[];
  /** The user message as the first request carries it (mentions and the dashboard-id line). */
  userContent: string;
  scripts: Script[];
  answers?: Record<string, Answer>;
  approve: (sql: string) => boolean;
  pageAnswers: Map<string, string>;
  /** Stop at the first streamed text: the text Rust had then. */
  stopText?: string;
}

interface SimOutput {
  requests: Rec[];
  text: string;
  parts: Part[];
  results: { callId: string; name: string; input: Json; result: RustResult }[];
  queries: Rec[];
  outcome: Rec;
  consumed: number;
}

function simulateTurn(s: SimInput): SimOutput {
  const system = rustSystem(s.engine, s.tables, s.sharing, s.clientTools);
  const names = rustToolNames(s.sharing, s.clientTools);
  const out: SimOutput = { requests: [], text: "", parts: [], results: [], queries: [], outcome: {}, consumed: 0 };
  const roundMsgs: Rec[] = [];
  let calls = 0;
  for (let round = 0; ; round++) {
    const messages = [...s.history, { role: "user", content: s.userContent }, ...roundMsgs];
    const body: Rec =
      s.shape === "anthropic"
        ? { model: "model-1", max_tokens: 4096, system, messages, stream: true }
        : { model: "model-1", messages: [{ role: "system", content: system }, ...messages], stream: true };
    if (names.length > 0) body.tools = { $toolSchemas: { profile: "assistant", shape: s.shape, names } };
    out.requests.push({ method: "POST", url: s.url, headers: s.headers, body });
    const script = s.scripts[round];
    if (!script) throw new Error("model: a round without a script");
    out.consumed = round + 1;
    if (s.stopText !== undefined) {
      out.text += s.stopText;
      out.outcome = { kind: "cancelled" };
      return out;
    }
    const d = decodeScript(script, s.shape);
    // Rule 1: the replay's model sends `run_query`'s argument as `sql`.
    for (const c of d.calls) c.input = renameInput(c.name, c.input) as Json;
    out.text += d.text;
    if (d.text) out.parts.push({ round, type: "text", text: d.text });
    if (d.error) {
      out.outcome = { kind: "error", ...d.error };
      return out;
    }
    if (d.calls.length === 0) {
      out.outcome = { kind: "done", stop: d.stop === "maxTokens" ? "maxTokens" : "end" };
      return out;
    }
    const tools: ToolPart[] = [];
    for (const c of d.calls) {
      calls++;
      if (calls > 20) {
        out.outcome = { kind: "error", code: "TOOL_LIMIT", message: "The model asked for more than 20 tool calls in this turn." };
        return out;
      }
      let result: RustResult;
      try {
        result = rustToolResult(c.name, c.input as Rec, {
          engine: s.engine,
          sharing: s.sharingAt ? s.sharingAt(calls - 1) : s.sharing,
          clientTools: s.clientTools,
          connectionName: s.connectionName,
          answers: s.answers,
          approve: s.approve,
          pageAnswer: s.pageAnswers.get(c.id),
          queries: out.queries,
          tables: s.tables,
        });
      } catch (e) {
        if (!(e instanceof Stopped)) throw e;
        out.outcome = { kind: "cancelled" };
        return out;
      }
      out.results.push({ callId: c.id, name: c.name, input: c.input, result });
      const p = toolPart(round, c.id, c.name, c.input, result);
      out.parts.push(p);
      tools.push(p);
    }
    // The request carries each result whole; only the stored copy is cut.
    roundMsgs.push(
      ...roundMessages(
        d.text,
        tools.map((t, i) => ({ ...t, result: out.results[out.results.length - tools.length + i].result.text, resultBytes: undefined })),
        s.shape,
      ),
    );
  }
}

/** The page's answers to client tools, read from today's requests (its `tool_result`s). */
function pageAnswersOf(reqs: Rec[]): Map<string, string> {
  const m = new Map<string, string>();
  for (const r of reqs) {
    const body = r.body as Rec | null;
    for (const msg of ((body?.messages ?? []) as Rec[])) {
      if (msg.role === "user" && Array.isArray(msg.content)) {
        for (const b of msg.content as Rec[]) if (b.type === "tool_result") m.set(String(b.tool_use_id), String(b.content));
      }
      if (msg.role === "tool") m.set(String(msg.tool_call_id), String(msg.content));
    }
  }
  return m;
}

/** The current user message of a recorded request: the last user message with text content. */
function userContentOf(req: Rec): string {
  const msgs = ((req.body as Rec).messages ?? []) as Rec[];
  for (let i = msgs.length - 1; i >= 0; i--) {
    if (msgs[i].role === "user" && typeof msgs[i].content === "string") return String(msgs[i].content);
  }
  throw new Error("model: no user message");
}

/** R2: a forbidden string shorter than this can't be told from ordinary log text. */
const MIN_FORBIDDEN = 8;
const shortForbidden = new Set<string>();

/** Decision 16 (I3): strings no log line may hold for a case. */
function forbidden(extra: (string | undefined | null)[], reqs: Rec[]): Rec {
  // `$TEST_KEY` stands for the test key, which no fixture may hold.
  const set = new Set<string>(["$TEST_KEY"]);
  for (const s of extra) if (s && s.trim()) set.add(s);
  for (const s of set) if (s.length < MIN_FORBIDDEN) shortForbidden.add(s);
  for (const r of reqs) {
    try {
      const u = new URL(String(r.url));
      set.add(u.pathname);
    } catch {
      /* none */
    }
  }
  return {
    $forbidden: [...set],
    $allowed: "Decision 16's lines: activity, ids, provider kind, model id, HTTP status, codes, round and tool-call counts, durations and token counts",
  };
}

/** The scripted model's `run_query` input as the Rust replay sends it (Decision 3: `query` is `sql`). */
function renameInput(name: string, input: unknown): unknown {
  if (name !== "run_query" || typeof input !== "object" || input === null || Array.isArray(input)) {
    return input;
  }
  const out: Rec = {};
  for (const [k, v] of Object.entries(input)) out[k === "query" ? "sql" : k] = v;
  return out;
}

/**
 * Rewrites recorded provider requests into what the Rust port sends: its
 * system prompt, its tools (a reference into Task 3's `tool-schemas.json`),
 * `run_query` inputs with `sql`, and each tool result as `results` gives it
 * (by tool call id), with `is_error` on Anthropic and `Error: ` on OpenAI.
 */
function rustRequests(
  recorded: Rec[],
  ctx: {
    engine: DatabaseType;
    tables: SchemaTable[] | null;
    sharing: Sharing;
    clientTools: boolean;
    results: Map<string, RustResult>;
    userContent?: (content: string) => string;
    /** False for `ai.generate`, which offers no tools. */
    tools?: boolean;
  },
): Rec[] {
  const system = rustSystem(ctx.engine, ctx.tables, ctx.sharing, ctx.clientTools, ctx.tools ?? true);
  const names = rustToolNames(ctx.sharing, ctx.clientTools);
  return recorded.map((r) => {
    const body = structuredClone(r.body) as Rec | null;
    const url = String(r.url);
    if (!body || !Array.isArray(body.messages)) return { ...r, body };
    const isAnthropic = url.startsWith("https://api.anthropic.com/");
    const streaming = body.stream === true;
    const messages = body.messages as Rec[];
    if (isAnthropic) {
      body.system = system;
    } else {
      messages[0] = { ...messages[0], content: system };
    }
    if (streaming) {
      if (names.length > 0) {
        body.tools = {
          $toolSchemas: { profile: "assistant", shape: isAnthropic ? "anthropic" : "openai", names },
        };
      } else {
        delete body.tools;
      }
    }
    for (const m of messages) {
      if (m.role === "user" && typeof m.content === "string" && ctx.userContent) {
        m.content = ctx.userContent(m.content);
      }
      if (isAnthropic && m.role === "assistant" && Array.isArray(m.content)) {
        for (const b of m.content as Rec[]) {
          if (b.type === "tool_use") b.input = renameInput(String(b.name), b.input) as Json;
        }
      }
      if (isAnthropic && m.role === "user" && Array.isArray(m.content)) {
        for (const b of m.content as Rec[]) {
          if (b.type !== "tool_result") continue;
          const res = ctx.results.get(String(b.tool_use_id));
          if (!res) throw new Error(`model: no result for ${String(b.tool_use_id)}`);
          b.content = res.text;
          if (res.isError) b.is_error = true;
        }
      }
      if (!isAnthropic && m.role === "assistant" && Array.isArray(m.tool_calls)) {
        for (const tc of m.tool_calls as Rec[]) {
          const fn = tc.function as Rec;
          fn.arguments = JSON.stringify(renameInput(String(fn.name), JSON.parse(String(fn.arguments))));
        }
      }
      if (!isAnthropic && m.role === "tool") {
        const res = ctx.results.get(String(m.tool_call_id));
        if (!res) throw new Error(`model: no result for ${String(m.tool_call_id)}`);
        m.content = res.isError ? `Error: ${res.text}` : res.text;
      }
    }
    return { ...r, body };
  });
}

/** Fails the recorder when the key appears anywhere in `value`. */

async function until(cond: () => boolean, what: string, ms = 5000) {
  const end = Date.now() + ms;
  while (!cond()) {
    if (Date.now() > end) throw new Error(`timed out waiting for ${what}`);
    await new Promise((r) => setTimeout(r, 2));
  }
}

async function runTurn(c: TurnCase): Promise<Rec> {
  resetCase();

  const controller = new AbortController();
  const chunks: string[] = [];
  const queries: Rec[] = [];
  const approvals: string[] = [];
  const dashboardCalls: Rec[] = [];
  const decisions = [...(c.approvals ?? [])];
  let outcome: Rec | null = null;
  const messages: AIMessage[] = [
    ...(c.history ?? []),
    { role: "user" as const, content: c.message ?? "How many users signed up this month?" },
  ].map((m, i) => ({
    id: `m${i + 1}`,
    chatId: "chat-1",
    role: m.role,
    content: m.content,
    timestamp: new Date(0),
  }));
  const params: SendAIMessageParams = {
    providerId: "prov-1",
    model: "model-1",
    messages,
    schema: c.schema ?? SCHEMA,
    shareSchema: c.shareSchema ?? true,
    shareData: c.shareData ?? true,
    connection: { id: "conn-1", type: engine, name: "Local" },
    runQuery: async (sql, signal, maxRows) => {

    },
    activeConnection: () =>
      c.activeConnection === undefined ? { id: "conn-1", name: "Local" } : c.activeConnection,
    aiAllowAllQueries: c.allowAll ?? true,
    onApprovalRequired: (query, _conn, approve, deny) => {
      approvals.push(query);
      const d = decisions.shift() ?? "allow";
      queueMicrotask(() => (d === "allow" ? approve() : d === "deny" ? deny() : controller.abort()));
    },
    signal: controller.signal,
    onChunk: (delta) => {
      chunks.push(delta);
      if (c.stopAtFirstChunk && !controller.signal.aborted) controller.abort();
    },
    onDone: () => {
      outcome ??= { kind: "done" };
    },
    onError: (error) => {
      outcome ??= { kind: "error", error };
    },
  };
  if (c.dashboards) {
    Object.assign(params, {
      onCreateDashboard: async (name: string) => {
        dashboardCalls.push({ call: "create", name });
        return { dashboardId: "dash-new" };
      },
      onAddWidget: async (dashboardId: string, widget: unknown) => {
        dashboardCalls.push({ call: "addWidget", dashboardId, widget: widget as Json });
        return { widgetId: "w-new" };
      },
      onGetDashboard: (dashboardId: string) => {
        dashboardCalls.push({ call: "get", dashboardId });
        return dashboardId === "dash-1"
          ? { id: "dash-1", name: "Sales", widgets: [] }
          : null;
      },
      onUpdateWidget: async (dashboardId: string, widgetId: string, updates: unknown) => {
        dashboardCalls.push({ call: "updateWidget", dashboardId, widgetId, updates: updates as Json });
      },
      onRemoveWidget: async (dashboardId: string, widgetId: string) => {
        dashboardCalls.push({ call: "removeWidget", dashboardId, widgetId });
      },
      onDashboardCreated: (id: string) => {
        dashboardCalls.push({ call: "created", dashboardId: id });
      },
    });
  }
  try {
    await sendAIMessage(params);
    outcome ??= controller.signal.aborted ? { kind: "cancelled" } : { kind: "returned" };
  } catch (err) {
    outcome ??= { kind: "threw", message: err instanceof Error ? err.message : String(err) };
  }
  await settle();
  const stalled = c.responses.some((r) => r.stall);
  if (stalled) await until(() => mock.closedEarly.some(Boolean), "the mock to see the close");
  if (mock.queue.length > 0) throw new Error(`${c.name}: ${mock.queue.length} responses left`);
  const record: Rec = {
    name: c.name,
    note: c.note,
    input: {
      provider: c.provider ? PROVIDERS[c.provider] : null,
      key: c.key ?? true,
      model: "model-1",
      engine,
      messages: messages.map((m) => ({ role: m.role, content: m.content })),
      schema: c.schema === undefined ? "SCHEMA" : c.schema,
      shareSchema: c.shareSchema ?? true,
      shareData: c.shareData ?? true,
      dashboards: c.dashboards ?? false,
      allowAll: c.allowAll ?? true,
      ...(c.approvals ? { approvals: c.approvals } : {}),
      ...(c.answers ? { answers: c.answers } : {}),
      ...(c.stopAtFirstChunk ? { stopAtFirstChunk: true } : {}),
      ...(c.activeConnection !== undefined ? { activeConnection: c.activeConnection } : {}),
    },
    responses: c.responses,
    requests,
    chunks,
    text: chunks.join(""),
    queries,
    approvals,
    ...(c.dashboards ? { dashboardCalls } : {}),
    outcome,
    ...(stalled ? { providerClosed: mock.closedEarly } : {}),
    logs: [...hoisted.logs],
  };
  assertNoKey(c.name, record);
  return record;
}

// ---------------------------------------------------------------- page (UIStateManager)

interface PageCase {
  name: string;
  note: string;
  connections: Rec[];
  active: string;
  global?: { shareSchemaGlobally: boolean; shareDataGlobally: boolean };
  history?: Partial<AIMessage>[];
  steps: (SendStep | { activate: string })[];
  responses: Script[];
  answers?: Record<string, Answer>;
  /** Make the chat's connection disappear before sending. */
  removeConnection?: boolean;
  /** The provider (Anthropic when absent). */
  provider?: "anthropic" | "openai";
  /** Turn the connection's data sharing off once its first query ran (Decision 22). */
  dataOffAfterFirstQuery?: boolean;
  /** The provider has no API key (Core refuses with `NO_API_KEY`). */
  noKey?: boolean;
}

interface SendStep {
  send: string;
  /** Answers to approval cards in order; `stop` presses Stop while the card waits. */
  approvals?: ("allow" | "deny" | "allowAll" | "stop")[];
  /** Press Stop once the reply shows text. */
  stopAtFirstChunk?: boolean;
}

const conn = (id: string, extra: Rec = {}): Rec => ({
  id,
  name: id === "conn-1" ? "Local" : "Other",
  type: "postgres",
  activeAIProviderId: "prov-1",
  activeAIModel: "model-1",
  ...extra,
});

async function runPage(c: PageCase): Promise<Rec> {
  resetCase();
  setProvider(c.provider ?? "anthropic", !c.noKey);
  if (c.global) Object.assign(aiSettingsStore.settings, c.global);
  const side: PageTurn[] = [];
  pageSide.set(c.name, side);
  mock.queue = [...c.responses];
  const library = new RecordingLibrary();
  for (const k of c.connections) library.seedConnection(String(k.id));
  setLibrary(library);
  const state = new DatabaseState();
  state.activeProjectId = "p1";
  state.connections = c.connections as never;
  state.activeConnectionIdByProject = { p1: c.active };
  state.schemas = Object.fromEntries(c.connections.map((k) => [String(k.id), SCHEMA]));
  state.queriesByProject = { p1: SAVED };
  state.dashboardsByProject = { p1: DASHBOARDS };
  const queries: Rec[] = [];
  const approvals: Rec[] = [];
  const stored: Rec[][] = [];
  let ui!: InstanceType<typeof UIStateManager>;
  const chats = new AIChatManager(state, async () => {}, (id) => ui.abortStreamFor(id));
  const view = (m: AIMessage): Rec => ({
    role: m.role,
    content: m.content,
    ...(m.dashboardId ? { dashboardId: m.dashboardId } : {}),
    ...(m.query ? { query: m.query } : {}),
    ...(m.pendingModelSelection ? { pendingModelSelection: m.pendingModelSelection } : {}),
    ...(m.pendingApproval ? { pendingApproval: m.pendingApproval.query } : {}),
  });
  ui = new UIStateManager(
    state,
    () => {},
    async (connectionId, sql, signal, _name, maxRows) => {
      queries.push({ connectionId, sql, maxRows: maxRows ?? null });
      if (c.dataOffAfterFirstQuery) {
        state.connections = state.connections.map((k) =>
          k.id === connectionId ? ({ ...k, aiShareData: false } as never) : k,
        );
      }
      return answerQuery(c.answers?.[sql] ?? DEFAULT_ANSWER, signal);
    },
    chats,
    async (chatId) => {
      stored.push((state.aiMessagesByChat[chatId] ?? []).map(view));
    },
    {
      createDashboard: async (name: string) => ({ id: "dash-new", name }),
      addWidget: async () => {},
      executeWidget: async () => {},
      getDashboard: () => null,
      updateWidget: async () => {},
      removeWidget: async () => {},
    } as never,
    { add: () => {} } as never,
  );
  if (c.history) {
    const chatId = await chats.ensureActiveChat();
    state.aiMessagesByChat = {
      ...state.aiMessagesByChat,
      [chatId!]: c.history.map((m, i) => ({
        id: `h${i + 1}`,
        chatId: chatId!,
        timestamp: new Date(0),
        ...m,
      })) as AIMessage[],
    };
  }
  if (c.removeConnection) {
    await chats.ensureActiveChat();
    state.connections = [];
  }
  for (const [index, step] of c.steps.entries()) {
    if ("activate" in step) {
      state.activeConnectionIdByProject = { p1: step.activate };
      continue;
    }
    const decisions = [...(step.approvals ?? [])];
    const turn: PageTurn = {
      step: index,
      connectionId: state.activeConnectionId ?? "",
      reqStart: requests.length,
      reqEnd: requests.length,
    };
    side.push(turn);
    const sent = await ui.sendAIMessage(step.send);
    if (!sent) continue;
    await settle();
    const end = Date.now() + 5000;
    while (state.isAIStreaming) {
      if (Date.now() > end) throw new Error(`${c.name}: the turn didn't end`);
      const chatId = state.aiStreamingChatId;
      const streaming = chatId ? (state.aiMessagesByChat[chatId] ?? []).at(-1) : undefined;
      if (step.stopAtFirstChunk && streaming?.content) {
        turn.stopText = streaming.content;
        ui.cancelAIStream();
        break;
      }
      const pending = chatId
        ? (state.aiMessagesByChat[chatId] ?? []).find((m) => m.pendingApproval)?.pendingApproval
        : undefined;
      if (pending && !approvals.some((a) => a.id === pending.id)) {
        approvals.push({ id: pending.id, query: pending.query, connectionName: pending.connectionName });
        const d = decisions.shift() ?? "allow";
        if (d === "stop") {
          ui.cancelAIStream();
          break;
        }
        if (d === "allowAll") ui.setAIAllowAll();
        if (d === "deny") pending.deny();
        else pending.approve();
      }
      await new Promise((r) => setTimeout(r, 2));
    }
    await settle();
    if (c.responses.some((r) => r.stall)) await until(() => mock.closedEarly.some(Boolean), "the mock to see the close");
    turn.reqEnd = requests.length;
  }
  if (mock.queue.length > 0) throw new Error(`${c.name}: ${mock.queue.length} responses left`);
  const chatId = state.activeAIChatId;
  const record: Rec = {
    name: c.name,
    note: c.note,
    input: {
      connections: c.connections,
      active: c.active,
      global: c.global ?? { shareSchemaGlobally: true, shareDataGlobally: false },
      ...(c.history ? { history: c.history } : {}),
      steps: c.steps,
      ...(c.answers ? { answers: c.answers } : {}),
      ...(c.removeConnection ? { removeConnection: true } : {}),
      ...(c.provider ? { provider: c.provider } : {}),
      ...(c.dataOffAfterFirstQuery ? { dataOffAfterFirstQuery: true } : {}),
      ...(c.noKey ? { noKey: true } : {}),
      schema: "SCHEMA",
      savedQueries: "SAVED",
      dashboards: "DASHBOARDS",
    },
    responses: c.responses,
    requests,
    queries,
    approvals: approvals.map(({ query, connectionName }) => ({ query, connectionName })),
    messages: chatId ? (state.aiMessagesByChat[chatId] ?? []).map(view) : [],
    stored: stored.at(-1) ?? null,
    storeCalls: stored.length,
    allowAllAfter: ui.aiAllowAllQueries,
    logs: [...hoisted.logs],
  };
  assertNoKey(c.name, record);
  return record;
}

/** The SQL and saved-query names a call carries, for the forbidden list. */
function callStrings(r: { input: Json; result: RustResult }): string[] {
  const i = (r.input ?? {}) as Rec;
  return [i.sql, i.query, i.saved_query, r.result.text].filter((s): s is string => typeof s === "string");
}

const providerMessages = (outcome: Rec) =>
  outcome.kind === "error" && typeof outcome.message === "string" && !CORE_WORDED.has(outcome.message)
    ? [outcome.message]
    : [];

function turnChanges(c: TurnCase, rec: Rec) {
  const base = c.name.split("/")[1];
  const reqs = rec.requests as Rec[];
  const expected: Rec = {};
  const extras: string[] = [c.message ?? "How many users signed up this month?", ...(c.markers ?? [])];
  if (base === "no-key") {
    expected.outcome = { kind: "error", code: "NO_API_KEY", message: NO_KEY };
  } else if (base === "no-provider") {
    expected.outcome = { kind: "error", code: "NO_PROVIDER", message: PROVIDER_GONE };
  } else {
    const decisions = [...(c.approvals ?? [])];
    const approvals: string[] = [];
    const sim = simulateTurn({
      shape: c.provider === "anthropic" ? "anthropic" : "openai",
      url: String(reqs[0].url),
      headers: reqs[0].headers as Record<string, string>,
      engine: c.engine ?? "postgres",
      tables: c.schema ?? SCHEMA,
      sharing: { schema: c.shareSchema ?? true, data: c.shareData ?? true },
      clientTools: c.dashboards ?? false,
      connectionName: "Local",
      history: (c.history ?? []).map((m) => ({ role: m.role, content: m.content })),
      userContent: userContentOf(reqs[0]),
      scripts: c.responses,
      answers: c.answers,
      approve: (sql) => {
        if (c.allowAll ?? true) return true;
        approvals.push(sql);
        const d = decisions.shift() ?? "allow";
        if (d === "stop") throw new Stopped();
        return d !== "deny";
      },
      pageAnswers: pageAnswersOf(reqs),
      stopText: c.stopAtFirstChunk ? String(rec.text) : undefined,
    });
    expected.requests = sim.requests;
    expected.queries = sim.queries;
    expected.approvals = approvals;
    expected.text = sim.text;
    expected.outcome = sim.outcome;
    for (const r of sim.results) extras.push(...callStrings(r));
    extras.push(...providerMessages(sim.outcome));
    extras.push(sim.text);
  }
  expected.logs = forbidden(extras, reqs);
  const note = CASE_NOTES[base];
  change(`turns/${c.name}`, expected, rec, { notes: note ? [note] : [], always: ["logs"] });
}

/** What runPage saw of each send step, kept out of the record. */
interface PageTurn {
  step: number;
  connectionId: string;
  reqStart: number;
  reqEnd: number;
  stopText?: string;
}
const pageSide = new Map<string, PageTurn[]>();

function pageChanges(c: PageCase, rec: Rec) {
  const shape = (c.provider ?? "anthropic") === "anthropic" ? "anthropic" : "openai";
  const reqs = rec.requests as Rec[];
  const responses = rec.responses as Script[];
  const global = c.global ?? { shareSchemaGlobally: true, shareDataGlobally: false };
  // Decision 31 (R4): a send the page stopped (a pending-model message and the user
  // message before it) never reached Core, so only the rest of the seed is stored.
  const seed = c.history ?? [];
  const seedRows: Rec[] = seed
    .filter((m, i) => !m.pendingModelSelection && !seed[i + 1]?.pendingModelSelection)
    .map((m) => ({ role: m.role, content: m.content, ...(m.dashboardId ? { dashboardId: m.dashboardId } : {}) }));
  // R1: one chat per connection; the seed is the first connection's chat.
  const chats = new Map<string, Rec[]>([[c.active, seedRows]]);
  let rows: Rec[] = seedRows;
  const approvals: Rec[] = [];
  const queries: Rec[] = [];
  const requestsOut: Rec[] = [];
  const extras: string[] = [];
  const allowAll = new Set<string>();
  let storeCalls = 0;
  let error: Rec | null = null;
  for (const t of pageSide.get(c.name) ?? []) {
    const step = c.steps[t.step] as SendStep;
    extras.push(step.send);
    if (!chats.has(t.connectionId)) chats.set(t.connectionId, []);
    rows = chats.get(t.connectionId)!;
    if (c.noKey) {
      // Decision 31 (R3): Core refuses before it stores anything.
      error = { code: "NO_API_KEY", message: NO_KEY };
      continue;
    }
    if (t.reqEnd === t.reqStart) continue; // the page stopped it (no model, connection removed)
    const row = c.connections.find((k) => k.id === t.connectionId)!;
    const sharing = {
      schema: (row.aiShareSchema as boolean | undefined) ?? global.shareSchemaGlobally,
      data: (row.aiShareData as boolean | undefined) ?? global.shareDataGlobally,
    };
    const decisions = [...(step.approvals ?? [])];
    const turnReqs = reqs.slice(t.reqStart, t.reqEnd);
    const sim = simulateTurn({
      shape,
      url: String(turnReqs[0].url),
      headers: turnReqs[0].headers as Record<string, string>,
      engine: "postgres",
      tables: SCHEMA,
      sharing,
      sharingAt: c.dataOffAfterFirstQuery ? (n) => (n === 0 ? sharing : { ...sharing, data: false }) : undefined,
      clientTools: true,
      connectionName: String(row.name),
      history: renderHistory(rows, shape),
      userContent: c.name === "mention-schema-off" ? step.send : userContentOf(turnReqs[0]),
      scripts: responses.slice(t.reqStart),
      answers: c.answers,
      approve: (sql) => {
        if (allowAll.has(String(row.id))) return true;
        approvals.push({ query: sql, connectionName: row.name });
        const d = decisions.shift() ?? "allow";
        if (d === "stop") throw new Stopped();
        if (d === "allowAll") {
          allowAll.add(String(row.id));
          return true;
        }
        return d !== "deny";
      },
      pageAnswers: pageAnswersOf(turnReqs),
      stopText: t.stopText,
    });
    requestsOut.push(...sim.requests);
    queries.push(...sim.queries.map((q) => ({ connectionId: row.id, ...q })));
    rows.push({ role: "user", content: step.send });
    const reply: Rec = { role: "assistant", content: sim.text };
    const created = sim.results.find((r) => r.name === "create_dashboard" && !r.result.isError);
    if (created) reply.dashboardId = (JSON.parse(created.result.text) as Rec).dashboard_id;
    if (sim.parts.some((p) => p.type === "tool")) reply.parts = sim.parts;
    rows.push(reply);
    storeCalls += 2;
    if (sim.outcome.kind === "error") error = { code: sim.outcome.code, message: sim.outcome.message };
    for (const r of sim.results) extras.push(...callStrings(r));
    extras.push(...providerMessages(sim.outcome));
    extras.push(sim.text);
  }
  const expected: Rec = {
    requests: requestsOut,
    queries,
    approvals,
    stored: rows,
    storeCalls,
    ...(error ? { error, messages: { $absent: "Task 7 words a failed turn; the page view isn't pinned here" } } : {}),
    logs: forbidden(extras, reqs),
  };
  const note = CASE_NOTES[c.name];
  change(`page/${c.name}`, expected, rec, { notes: note ? [note] : [], always: ["logs", "storeCalls"] });
}

/** Decision 29's cases: stored rows only Rust reads (no TypeScript counterpart). */
function historyCases(): Rec[] {
  const turn = (tag: string, user: string, rounds: number, resultBytes: number, finalText: string): Rec[] => {
    const parts: Part[] = [];
    let content = "";
    for (let r = 0; r < rounds; r++) {
      const text = `Round ${r}. `;
      content += text;
      parts.push({ round: r, type: "text", text });
      parts.push({
        round: r,
        type: "tool",
        callId: `${tag}_${r}`,
        name: "run_query",
        input: { sql: `SELECT ${r}` },
        ok: true,
        result: "x".repeat(resultBytes),
      });
    }
    content += finalText;
    parts.push({ round: rounds, type: "text", text: finalText });
    return [
      { id: `${tag}-user`, role: "user", content: user },
      { id: `${tag}-reply`, role: "assistant", content, parts },
    ];
  };
  const specs: { name: string; note: string; turns: [string, string, number, number, string][] }[] = [
    {
      name: "budget-whole-turns",
      note: "Two small turns: both kept whole.",
      turns: [
        ["a", "First question", 2, 100, "Done one."],
        ["b", "Second question", 1, 100, "Done two."],
      ],
    },
    {
      name: "budget-partial-turn",
      note: "Two turns of 20 rounds with 15,000-byte results: the newer whole, the older's question and its latest rounds that fit.",
      turns: [
        ["a", "First question", 20, 15000, "Done one."],
        ["b", "Second question", 20, 15000, "Done two."],
      ],
    },
    {
      name: "budget-drops-older-turns",
      note: "Three turns where the newest alone nearly fills the budget: the turn that doesn't fit keeps only whole rounds, and nothing older is kept.",
      turns: [
        ["a", "First question", 1, 100, "Done one."],
        ["b", "Second question", 2, 15000, "Done two."],
        ["c", "Third question", 34, 15000, "Done three."],
      ],
    },
  ];
  return specs.map((s) => {
    const rows = s.turns.flatMap((t) => turn(...t));
    const result = historyBudget(rows);
    const rec: Rec = {
      name: s.name,
      note: s.note,
      rustOnly: true,
      input: {
        generate: s.turns.map(([tag, user, rounds, resultBytes, finalText]) => ({ tag, user, rounds, resultBytes, finalText })),
      },
    };
    change(`history/${s.name}`, { kept: result.kept, bytes: result.bytes }, undefined, {
      notes: ["Rust only: today's history is text only and has no budget (item 4)."],
    });
    return rec;
  });
}

function promptCases(): Rec[] {
  const out: Rec[] = [];
  const ctx = buildSchemaContext(SCHEMA);
  for (const engine of ENGINES) {
    for (const schema of [false, true]) {
      for (const dashboards of [false, true]) {
        const name = `system/${engine}/${schema ? "schema" : "no-schema"}/${dashboards ? "dashboards" : "no-dashboards"}`;
        const prompt = buildSystemPrompt(schema ? ctx : "", engine, dashboards);
        out.push({ name, input: { engine, schema: schema ? "SCHEMA" : null, dashboards }, prompt });
        const rust = rustSystem(engine, schema ? SCHEMA : null, { schema, data: false }, dashboards);
        if (rust !== prompt) {
          change(`prompts/${name}`, { prompt: rust });
        }
      }
    }
  }
  const contexts: [string, string, SchemaTable[]][] = [
    ["empty", "No tables: no context at all.", []],
    ["full", "Indexes, nullable columns, a view, two tables named `users` in two schemas, a table without columns.", SCHEMA],
    ["no-indexes-field", "A table whose `indexes` is missing.", [{ name: "t", schema: "main", type: "table", columns: [col("a", "TEXT", true)] } as never]],
  ];
  for (const [name, note, tables] of contexts) {
    const context = buildSchemaContext(tables);
    out.push({ name: `schema-context/${name}`, note, input: { tables }, context });
    const rust = rustSchemaContext(tables).text;
    if (rust !== context) {
      change(`prompts/schema-context/${name}`, { context: rust });
    }
  }
  const big = largeSchema();
  const ctxBig = buildSchemaContext(big);
  const digest = (s: string) => ({
    bytes: utf8(s),
    sha256: createHash("sha256").update(s, "utf8").digest("hex"),
    head: s.slice(0, 200),
    tail: s.slice(-200),
  });
  out.push({
    name: "schema-context/large",
    note: "3,000 tables (generated, see the README): today all of it goes into every round.",
    input: { generate: LARGE },
    context: digest(ctxBig),
  });
  const r = rustSchemaContext(big);
  change("prompts/schema-context/large", { context: digest(r.text) }, undefined, {
    notes: [`The cut keeps ${r.kept} tables and names the ${r.left} left out.`],
  });
  return out;
}

    id: "conn-1",

    activeAIModel: g.model === null ? null : (g.model ?? "model-1"),

  };

    getMonacoRef: () => ({ insertText: (t: string) => inserted.push(t) }),
  } as never;
  const prompt = createAIInlinePrompt(ctx, { onExecute: () => executed++ });

  const rec: Rec = {
    name: g.name,
    note: g.note,

    requests,
    inserted,
    executed,
    error: prompt.error?.message ?? null,
    toasts: [...hoisted.toasts],
    logs: [...hoisted.logs],
  };
  assertNoKey(g.name, rec);
  return rec;
}

/** The reply a non-streaming answer carries. */
function generateReplyText(script: Script): string {
  try {
    const j = JSON.parse(script.body) as Rec;
    const content = (j.content ?? []) as Rec[];
    if (Array.isArray(j.content)) return content.map((b) => String(b.text ?? "")).join("");
    return String((((j.choices ?? []) as Rec[])[0]?.message as Rec | undefined)?.content ?? "");
  } catch {
    return "";
  }
}

function generateChanges(g: GenCase, rec: Rec) {
  const reqs = rec.requests as Rec[];
  const expected: Rec = {};
  const extras: string[] = [g.request, g.existing];
  if (reqs.length > 0) {
    expected.requests = rustRequests(reqs, {
      engine: g.engine ?? "postgres",
      tables: SCHEMA,
      sharing: { schema: g.shareSchema ?? true, data: false },
      clientTools: false,
      results: new Map(),
      tools: false,
    });
  }
  if ((rec.inserted as string[]).length > 0) {
    expected.executed = 0;
    expected.notice = "Inserted. Run it with ⌘↵.";
    extras.push(...(rec.inserted as string[]));
  }
  const fail = (code: string, message: string) => {
    expected.code = code;
    expected.message = message;
  };
  const base = g.name.split("/")[1];
  const shape = g.provider === "anthropic" ? "anthropic" : "openai";
  const d = g.responses[0] ? decodeScript(g.responses[0], shape) : null;
  if (g.responses[0] && !g.responses[0].networkError && (g.responses[0].status ?? 200) === 200) {
    extras.push(generateReplyText(g.responses[0]));
  }
  if (base === "http-429" || base === "http-500" || base === "network-error") {
    fail(d!.error!.code, d!.error!.message);
    if (base !== "http-429") expected.toasts = [d!.error!.message];
    if (!CORE_WORDED.has(d!.error!.message)) extras.push(d!.error!.message);
  }
  if (base === "no-key") fail("NO_API_KEY", NO_KEY);
  if (g.name === "none/no-model") fail("NO_MODEL", NO_MODEL);
  if (g.name === "none/no-providers") fail("NO_PROVIDER", NO_PROVIDER_YET);
  expected.logs = forbidden(extras, reqs);
  const notes: string[] = [];
  if (g.name.startsWith("none/")) {
    notes.push("Decision 27 (f), M3: with no model chosen `NO_MODEL`; with no provider configured at all `NO_PROVIDER`, \"No AI provider is configured.\".");
  }
  change(`generate/${g.name}`, expected, rec, { notes, always: ["logs"] });
}

/** Decision 28 (I1): `models` answers the list or an `RpcError`; `test` answers `null` or an `RpcError`. */
function modelsChanges(r: Rec) {
  const name = String(r.name);
  const [op, kind, which] = name.split("/");
  const isTest = op === "test";
  const reqs = r.requests as Rec[];
  const expected: Rec = {};
  const notes: string[] = [];
  const err = (code: string, message: string) => {
    expected.result = { $absent: "the call fails with an RpcError" };
    expected.error = { code, message };
  };
  const ok = (list: string[]) => {
    expected.result = isTest ? null : list;
  };
  const extras: string[] = [];
  switch (which) {
    case "ok":
    case "keyless":
      ok(["model-1", "model-2"]);
      break;
    case "no-key":
      err("NO_API_KEY", NO_KEY);
      notes.push("Decisions 7 and 15: `NO_API_KEY` before any request.");
      break;
    case "401":
      err("PROVIDER_ERROR", "invalid x-api-key");
      extras.push("invalid x-api-key");
      break;
    case "default-url":
    case "empty-url":
      expected.requests = [
        { method: "GET", url: "https://api.openai.com/v1/models", headers: { authorization: "Bearer <key>" }, body: null },
      ];
      ok(["model-1", "model-2"]);
      expected.unusedResponses = 0;
      notes.push("Decision 10 (Task 2's wire): without a base URL the model list and the test use the chat's default, `https://api.openai.com/v1`; today they ask nothing.");
      break;
    case "not-json":
      if (isTest) {
        ok([]);
        notes.push("Decision 28 (g): the test checks reachability and auth, so a 200 whose body isn't JSON passes.");
      } else {
        err("PROVIDER_ERROR", "The provider's answer isn't valid JSON.");
      }
      break;
    case "network-error":
      err("PROVIDER_ERROR", UNREACHABLE);
      break;
    default:
      throw new Error(`model: no rule for ${name}`);
  }
  void kind;
  notes.push("Decision 28 (I1): `models` answers the list and `test` answers `null`; a failure is an `RpcError` with Decision 15's code.");
  expected.logs = forbidden(extras, (expected.requests as Rec[] | undefined) ?? reqs);
  change(`models/${name}`, expected, r, { notes, always: ["logs"], decisions: ["28"] });
}
