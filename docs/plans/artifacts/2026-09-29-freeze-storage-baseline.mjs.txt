/** JSON with every value that fits in ~100 columns kept on one line. */
function compactJson(value, indent = "") {
  const flat = JSON.stringify(value);
  if (flat === undefined) return "null";
  if (flat.length + indent.length <= 100 || value === null || typeof value !== "object") return flat;
  const inner = indent + "  ";
  if (Array.isArray(value)) {
    return `[\n${value.map((v) => inner + compactJson(v, inner)).join(",\n")}\n${indent}]`;
  }
  const entries = Object.entries(value).filter(([, v]) => v !== undefined);
  return `{\n${entries.map(([k, v]) => `${inner}${JSON.stringify(k)}: ${compactJson(v, inner)}`).join(",\n")}\n${indent}}`;
}
const writeJson = (file, value) => {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, compactJson(value) + "\n");
};

function materialize(tree) {
  if (tree === "current") return REPO;
  const dir = path.join(WORK, "trees", tree);
  fs.mkdirSync(dir, { recursive: true });
  const tar = execFileSync("git", ["archive", tree, "src/lib"], { cwd: REPO, maxBuffer: 1 << 30 });
  execFileSync("tar", ["-x", "-C", dir], { input: tar });
  return dir;
}

async function bundle(tree, root, extraExports = "") {
  const lib = path.join(root, "src/lib");
  const shimDir = path.join(WORK, "shims", tree);
  fs.mkdirSync(shimDir, { recursive: true });
  const storageShim = path.join(shimDir, "storage.ts");
  fs.writeFileSync(
    storageShim,
    `export * from ${JSON.stringify(path.join(lib, "storage/repository.ts"))};\n` +
      `export * from ${JSON.stringify(path.join(lib, "storage/schema.ts"))};\n` +
      `export async function getDatabase() { return (globalThis as any).__FREEZE_DB__; }\n`,
  );
  const loggerShim = path.join(shimDir, "logger.ts");
  fs.writeFileSync(
    loggerShim,
    `const noop = () => Promise.resolve();\nexport const log = { info: noop, warn: noop, error: noop, debug: noop, trace: noop };\n`,
  );
  const entry = path.join(shimDir, "entry.ts");
  fs.writeFileSync(
    entry,
    `export * as schema from ${JSON.stringify(path.join(lib, "storage/schema.ts"))};\n` +
      `export * as repos from ${JSON.stringify(path.join(lib, "storage/repository.ts"))};\n` +
      `export { MigrationManager } from ${JSON.stringify(path.join(lib, "hooks/database/migration.svelte.ts"))};\n` +
      extraExports,
  );
  const sqlWasm = path.join(REPO, "node_modules/sql.js/dist/sql-wasm.wasm");
  const outfile = path.join(WORK, `bundle-${tree}.mjs`);
  await esbuild.build({
    entryPoints: [entry],
    outfile,
    bundle: true,
    format: "esm",
    platform: "node",
    target: "node20",
    external: ["better-sqlite3", "sql.js"],
    logLevel: "error",
    plugins: [
      {
        name: "seaquel-freeze",
        setup(build) {
          build.onResolve({ filter: /^\$lib(\/.*)?$/ }, async (args) => {
            if (args.path === "$lib/storage") return { path: storageShim };
            if (args.path === "$lib/utils/logger") return { path: loggerShim };
            const rest = args.path.slice("$lib".length) || "/index";
            return build.resolve("." + rest, { resolveDir: lib, kind: args.kind });
          });
          build.onResolve({ filter: /\?url$/ }, (args) => ({ path: args.path, namespace: "url" }));
          build.onLoad({ filter: /.*/, namespace: "url" }, () => ({
            contents: `export default ${JSON.stringify(sqlWasm)};`,
            loader: "js",
          }));
        },
      },
    ],
  });
  return import(pathToFileURL(outfile).href);
}

// ---------------------------------------------------------------------------
// The SqliteDatabase adapter, copied from src/lib/server/storage.ts (it isn't
// exported). Only used for the schema runs; the repository cases use the real
// one through openUserDb.
// ---------------------------------------------------------------------------

function adapter(db) {
  return {
    async execute(sql, params = []) {
      return db.prepare(sql).run(...params).changes;
    },
    async query(sql, params = []) {
      return db.prepare(sql).all(...params);
    },
    async transaction(statements) {
      const tx = db.transaction((stmts) => {
        for (const { sql, params = [] } of stmts) db.prepare(sql).run(...params);
      });
      tx(statements);
    },
    async close() {},
  };
}

/** The persistence methods MigrationManager calls, as PersistenceManager implements them. */
function fakePersistence(a, mod) {
  return {
    async getStorageVersion() {
      try {
        const rows = await a.query("SELECT version FROM schema_version ORDER BY version DESC LIMIT 1");
        return rows.length > 0 ? rows[0].version : 0;
      } catch {
        return 0;
      }
    },
    async setStorageVersion(version) {
      await a.execute("INSERT INTO schema_version (version) VALUES (?)", [version]);
    },
    async loadPersistedConnections() {
      try {
        return await mod.repos.connectionsRepo.loadAll(a);
      } catch {
        return [];
      }
    },
    // No legacy JSON files exist in these runs.
    async loadLegacyConnectionState() {
      return null;
    },
    async removeLegacyConnectionState() {},
  };
}

function dumpSchema(db, header) {
  const objs = db
    .prepare(
      "SELECT type, name, tbl_name, sql FROM sqlite_master WHERE sql IS NOT NULL AND name NOT LIKE 'sqlite_%' ORDER BY rowid",
    )
    .all();
  const lines = [header, ""];
  for (const o of objs) lines.push(`${o.sql};`, "");
  const versions = db.prepare("SELECT version FROM schema_version ORDER BY rowid").all();
  for (const v of versions) {
    lines.push(
      `INSERT INTO schema_version (version, migrated_at) VALUES (${v.version}, '${FIXED_MIGRATED_AT}');`,
    );
  }
  return lines.join("\n") + "\n";
}

const norm = (s) => (s ?? "").replace(/\s+/g, " ").replace(/\( /g, "(").replace(/ \)/g, ")").trim();

/** Structural description of a database's schema, for comparing two files. */
function describeSchema(db) {
  const out = { tables: {}, indexes: {} };
  const tables = db
    .prepare("SELECT name, sql FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name")
    .all();
  for (const t of tables) {
    out.tables[t.name] = {
      columns: db
        .prepare(`SELECT cid, name, type, "notnull", dflt_value, pk, hidden FROM pragma_table_xinfo(?)`)
        .all(t.name),
      foreignKeys: db
        .prepare(`SELECT id, seq, "table", "from", "to", on_update, on_delete, "match" FROM pragma_foreign_key_list(?)`)
        .all(t.name),
      indexes: db
        .prepare(`SELECT name, "unique", origin, partial FROM pragma_index_list(?) ORDER BY name`)
        .all(t.name)
        .map((i) => ({
          ...i,
          columns: db.prepare(`SELECT seqno, name FROM pragma_index_xinfo(?) WHERE key = 1`).all(i.name),
          desc: db.prepare(`SELECT name, "desc" FROM pragma_index_xinfo(?) WHERE key = 1`).all(i.name),
        })),
      sql: norm(t.sql),
    };
  }
  for (const i of db
    .prepare("SELECT name, tbl_name, sql FROM sqlite_master WHERE type='index' AND sql IS NOT NULL ORDER BY name")
    .all()) {
    out.indexes[i.name] = { table: i.tbl_name, sql: norm(i.sql) };
  }
  return out;
}

function diffSchemas(fresh, upgraded) {
  const diffs = [];
  const tn = new Set([...Object.keys(fresh.tables), ...Object.keys(upgraded.tables)]);
  for (const name of [...tn].sort()) {
    const f = fresh.tables[name];
    const u = upgraded.tables[name];
    if (!f) {
      diffs.push({ table: name, kind: "extra-table", upgraded: u.sql });
      continue;
    }
    if (!u) {
      diffs.push({ table: name, kind: "missing-table", fresh: f.sql });
      continue;
    }
    const fcols = f.columns.map((c) => c.name);
    const ucols = u.columns.map((c) => c.name);
    if (JSON.stringify(fcols) !== JSON.stringify(ucols)) {
      const sameSet = JSON.stringify([...fcols].sort()) === JSON.stringify([...ucols].sort());
      diffs.push({
        table: name,
        kind: sameSet ? "column-order" : "column-set",
        fresh: fcols,
        upgraded: ucols,
      });
    }
    for (const fc of f.columns) {
      const uc = u.columns.find((c) => c.name === fc.name);
      if (!uc) continue;
      for (const k of ["type", "notnull", "dflt_value", "pk", "hidden"]) {
        if (fc[k] !== uc[k]) {
          diffs.push({ table: name, column: fc.name, kind: `column-${k}`, fresh: fc[k], upgraded: uc[k] });
        }
      }
    }
    const fk = (l) => l.map(({ id, ...r }) => JSON.stringify(r)).sort();
    if (JSON.stringify(fk(f.foreignKeys)) !== JSON.stringify(fk(u.foreignKeys))) {
      diffs.push({
        table: name,
        kind: "foreign-keys",
        fresh: f.foreignKeys.map(({ id, seq, ...r }) => r),
        upgraded: u.foreignKeys.map(({ id, seq, ...r }) => r),
      });
    }
    if (JSON.stringify(f.indexes) !== JSON.stringify(u.indexes)) {
      diffs.push({ table: name, kind: "indexes", fresh: f.indexes, upgraded: u.indexes });
    }
    if (f.sql !== u.sql) diffs.push({ table: name, kind: "create-sql-text", fresh: f.sql, upgraded: u.sql });
  }
  const inames = new Set([...Object.keys(fresh.indexes), ...Object.keys(upgraded.indexes)]);
  for (const name of [...inames].sort()) {
    const f = fresh.indexes[name];
    const u = upgraded.indexes[name];
    if (!f) diffs.push({ index: name, kind: "extra-index", upgraded: u.sql });
    else if (!u) diffs.push({ index: name, kind: "missing-index", fresh: f.sql });
    else if (f.sql !== u.sql) diffs.push({ index: name, kind: "index-sql", fresh: f.sql, upgraded: u.sql });
  }
  return diffs;
}

/** What today's db.ts does to an existing file (desktop and demo), minus the legacy-JSON re-import. */
async function upgradeWithCurrent(file, mod) {
  const db = new Database(file);
  const a = adapter(db);
  globalThis.__FREEZE_DB__ = a;
  try {
    db.pragma("journal_mode = WAL");
    db.pragma("busy_timeout = 5000");
    db.pragma("foreign_keys = ON");
    const fresh = await mod.schema.initializeSchema(a);
    if (fresh) throw new Error("expected an existing schema");
    await new mod.MigrationManager(fakePersistence(a, mod)).migrateIfNeeded();
    return { db };
  } catch (e) {
    return { db, error: String(e?.message ?? e) };
  }
}

async function freezeSchemas(mods) {
  const schemaDir = path.join(OUT, "schemas");
  fs.mkdirSync(path.join(schemaDir, "upgraded"), { recursive: true });
  const fresh = {};
  for (const tree of [...TAGS, "current"]) {
    const mod = mods[tree];
    const dbText = tree === "current"
      ? fs.readFileSync(path.join(REPO, "src/lib/storage/db.ts"), "utf8")
      : execFileSync("git", ["show", `${tree}:src/lib/storage/db.ts`], { cwd: REPO, encoding: "utf8" });
    const constName = FRESH_VERSION_CONST[tree];
    if (!new RegExp(`INSERT INTO schema_version \\(version\\) VALUES \\(\\?\\)",\\s*\\[\\s*${constName}`).test(dbText)) {
      throw new Error(`${tree}: db.ts doesn't insert ${constName}`);
    }
    const file = path.join(WORK, `fresh-${tree}.db`);
    const db = new Database(file);
    const a = adapter(db);
    globalThis.__FREEZE_DB__ = a;
    db.pragma("journal_mode = WAL");
    db.pragma("foreign_keys = ON");
    const isFresh = await mod.schema.initializeSchema(a);
    if (!isFresh) throw new Error(`${tree}: initializeSchema didn't report a fresh file`);
    await a.execute("INSERT INTO schema_version (version) VALUES (?)", [mod.schema[constName]]);
    await new mod.MigrationManager(fakePersistence(a, mod)).migrateIfNeeded();
    // Data the first launch wrote (beta.1's MigrationManager creates the default
    // project). Timestamps are "now", so they're masked.
    const firstLaunchRows = {};
    for (const t of ["projects", "app_state", "project_state"]) {
      const rows = db.prepare(`SELECT * FROM ${t}`).all().map((r) => {
        for (const k of ["created_at", "updated_at"]) if (k in r) r[k] = "<first launch time>";
        return r;
      });
      if (rows.length) firstLaunchRows[t] = rows;
    }
    const header =
      `-- Frozen: the schema a fresh metadata database had after its first launch on ${tree === "current" ? `today's tree (${HEAD})` : tree}.\n` +
      `-- Made by docs/plans/artifacts/2026-09-29-freeze-storage-baseline.mjs.txt from that release's schema.ts, db.ts and\n` +
      `-- MigrationManager; migrated_at is fixed. See ../README.md.`;
    fs.writeFileSync(path.join(schemaDir, `${tree}.sql`), dumpSchema(db, header));
    fresh[tree] = { describe: describeSchema(db), firstLaunchRows };
    db.close();
    log(`schema ${tree}: ${Object.keys(fresh[tree].describe.tables).length} tables`);
  }

  // Today's web path (openUserDb) must agree with the desktop/demo path.
  const current = mods.current;
  process.env.DATA_DIR = path.join(WORK, "web-schema-check");
  const raw = await current.openUserDb("schema-check");
  const webDescribe = describeSchema(raw);
  const webDiffs = diffSchemas(fresh.current.describe, webDescribe);
  if (webDiffs.length) throw new Error("openUserDb schema differs: " + JSON.stringify(webDiffs));
  current.closeAllUserDbs();

  // Upgrade each old dump with today's code. beta.1 also goes through
  // v2026.4.5's code first, the path its users took if they updated then.
  const report = {};
  for (const [tag, via] of [...TAGS.map((t) => [t, null]), ["v2026.4.5-beta.1", "v2026.4.5"]]) {
    const label = via ? `${tag}-via-${via}` : tag;
    const file = path.join(WORK, `upgrade-${label}.db`);
    const seed = new Database(file);
    seed.exec(fs.readFileSync(path.join(schemaDir, `${tag}.sql`), "utf8"));
    seed.close();
    let viaError = null;
    if (via) {
      const r = await upgradeWithCurrent(file, mods[via]);
      viaError = r.error ?? null;
      r.db.close();
    }
    const first = await upgradeWithCurrent(file, current);
    const after1 = first.db ? dumpSchema(first.db, "") : null;
    first.db.close();
    const second = await upgradeWithCurrent(file, current);
    const after2 = dumpSchema(second.db, "");
    const describe = describeSchema(second.db);
    const header =
      `-- Frozen: ${tag}.sql after ${via ? `${via}'s upgrade path and then ` : ""}today's upgrade path (initializeSchema's upgradeSchema, then MigrationManager).\n` +
      (first.error ? `-- Today's upgrade FAILS on this file (${first.error}); this is the state it leaves behind.\n` : "") +
      `-- Made by docs/plans/artifacts/2026-09-29-freeze-storage-baseline.mjs.txt; migrated_at is fixed. See ../../README.md.`;
    fs.writeFileSync(path.join(schemaDir, "upgraded", `${label}.sql`), dumpSchema(second.db, header));
    second.db.close();
    report[label] = {
      viaError,
      firstRunError: first.error ?? null,
      secondRunError: second.error ?? null,
      secondRunChangedSchema: after1 !== after2,
      differencesFromFresh: diffSchemas(fresh.current.describe, describe),
    };
    log(
      `upgrade ${label}: ${report[label].differencesFromFresh.length} differences` +
        (first.error ? `, first run failed: ${first.error}` : "") +
        (second.error ? `, second run failed: ${second.error}` : ""),
    );
  }
  writeJson(path.join(schemaDir, "upgrade-differences.json"), {
    note: "Differences between each release's schema after today's upgrade path and a fresh file from today's tree. Made by freeze-storage-baseline.mjs. Frozen.",
    firstLaunchRows: Object.fromEntries(Object.entries(fresh).map(([k, v]) => [k, v.firstLaunchRows])),
    upgrades: report,
  });
  return report;
}

function dumpRows(db, table, skip = []) {
  const cols = db
    .prepare(`SELECT name FROM pragma_table_info(?)`)
    .all(table)
    .map((c) => c.name)
    .filter((c) => !skip.includes(c));
  const q = (c) => `"${c}"`;
  const sel = [...cols.map(q), ...cols.map((c) => `typeof(${q(c)})`)].join(", ");
  const order = cols.map((_, i) => i + 1).join(", ");
  const raw = db.prepare(`SELECT ${sel} FROM ${q(table)} ORDER BY ${order}`).raw().all();
  return {
    columns: cols,
    rows: raw.map((r) => r.slice(0, cols.length)),
    types: raw.map((r) => r.slice(cols.length)),
  };
}

let SqlDecimal;
let toStorable;

/** JSON form of a JS value as the fixtures record it. */
function enc(v) {
  if (v === undefined) return undefined;
  if (v instanceof Date) return { $date: Number.isNaN(v.getTime()) ? null : v.toISOString() };
  if (
    typeof v === "bigint" ||
    v instanceof Uint8Array ||
    v instanceof SqlDecimal ||
    (typeof v === "number" && !Number.isFinite(v))
  ) {
    return toStorable(v);
  }
  if (Array.isArray(v)) return v.map((x) => (x === undefined ? null : enc(x)));
  if (v !== null && typeof v === "object") {
    if ("$sq" in v) return { $sq: "json", v };
    const out = {};
    for (const [k, x] of Object.entries(v)) if (x !== undefined) out[k] = enc(x);
    return out;
  }
  return v;
}

/** Rows through the SqliteDatabase interface, so both backends dump the same way. */
async function dumpRowsVia(a, table) {
  const cols = (await a.query(`SELECT name FROM pragma_table_info('${table}')`)).map((c) => c.name);
  const q = (c) => `"${c}"`;
  const sel = [...cols.map((c, i) => `${q(c)} AS v${i}`), ...cols.map((c, i) => `typeof(${q(c)}) AS t${i}`)].join(", ");
  const order = cols.map((_, i) => i + 1).join(", ");
  const res = await a.query(`SELECT ${sel} FROM ${q(table)} ORDER BY ${order}`);
  return {
    columns: cols,
    rows: res.map((r) => cols.map((_, i) => r[`v${i}`])),
    types: res.map((r) => cols.map((_, i) => r[`t${i}`])),
  };
}

async function runCase(c, a, repos) {
  const steps = [];
  const h = {
    async call(name, ...args) {
      const [repo, method] = name.split(".");
      const target = repos[repo];
      if (!target || typeof target[method] !== "function") throw new Error(`no ${name}`);
      const step = { call: name, args: enc(args) };
      try {
        const result = await target[method](a, ...args);
        if (result !== undefined) step.result = enc(result);
      } catch (e) {
        step.error = { message: String(e?.message ?? e), code: e?.code };
      }
      steps.push(step);
      return step.result;
    },
    async sql(sql, params = []) {
      await a.execute(sql, params);
      steps.push({ sql, params });
    },
  };
  await c.run(h);
  const rows = {};
  for (const t of c.tables) rows[t] = await dumpRowsVia(a, t);
  return { steps, rows };
}

const MASKS = [[/workflow-[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/g, "workflow-<random-uuid>"]];
function mask(value) {
  let s = JSON.stringify(value);
  for (const [re, rep] of MASKS) s = s.replace(re, rep);
  return JSON.parse(s);
}

async function freezeRepos(mods) {
  const cur = mods.current;
  SqlDecimal = cur.SqlDecimal;
  toStorable = cur.toStorable;
  const allCases = buildCases(cur);

  // better-sqlite3 through the web server's real openUserDb.
  process.env.DATA_DIR = path.join(WORK, "web");
  // sql.js through the demo's real WebSqliteDatabase; localStorage is a Map.
  const store = new Map();
  globalThis.localStorage = {
    getItem: (k) => (store.has(k) ? store.get(k) : null),
    setItem: (k, v) => store.set(k, String(v)),
    removeItem: (k) => store.delete(k),
    clear: () => store.clear(),
  };

  const byRepo = {};
  const backendDiffs = [];
  let n = 0;
  for (const c of allCases) {
    n++;
    const web = await cur.userStorage(`case-${n}`);
    const primary = mask(await runCase(c, web, cur.repos));

    store.clear();
    const demo = await new cur.WebSqliteProvider().open("seaquel.db");
    // What db.ts does for the demo on a fresh file (minus the legacy-JSON import).
    await demo.execute("PRAGMA journal_mode=WAL");
    await demo.execute("PRAGMA foreign_keys=ON");
    if (!(await cur.schema.initializeSchema(demo))) throw new Error("demo db not fresh");
    await demo.execute("INSERT INTO schema_version (version) VALUES (?)", [cur.schema.CURRENT_STORAGE_VERSION]);
    const secondary = mask(await runCase(c, demo, cur.repos));
    await demo.close();

    const cmp = (x) =>
      JSON.stringify({
        steps: x.steps.map(({ error, ...s }) => ({ ...s, error: error ? true : undefined })),
        rows: x.rows,
      });
    const entry = { name: c.name, description: c.description, ...primary };
    if (c.notes) entry.notes = c.notes;
    if (cmp(primary) !== cmp(secondary)) {
      backendDiffs.push({ case: c.name, betterSqlite3: primary, sqlJs: secondary });
      const steps = primary.steps
        .map((s, i) => [s, secondary.steps[i], i])
        .filter(([a, b]) => JSON.stringify({ ...a, error: !!a.error }) !== JSON.stringify({ ...b, error: !!b.error }))
        .map(([, , i]) => i);
      const tables = Object.keys(primary.rows).filter((t) => JSON.stringify(primary.rows[t]) !== JSON.stringify(secondary.rows[t]));
      entry.demoDiffers = { steps, tables };
    }
    (byRepo[c.repo] ??= []).push(entry);
  }
  cur.closeAllUserDbs();

  const repoFiles = {
    projectsRepo: "projects",
    appStateRepo: "app-state",
    connectionsRepo: "connections",
    connectionOverridesRepo: "connection-overrides",
    projectStateRepo: "project-state",
    savedQueriesRepo: "saved-queries",
    queryVersionsRepo: "query-versions",
    queryHistoryRepo: "query-history",
    sharedReposRepo: "shared-repos",
    themeRepo: "themes",
    licenseRepo: "license",
    onboardingRepo: "onboarding",
    tutorialRepo: "tutorial",
    importStateRepo: "import-state",
    dashboardsRepo: "dashboards",
    dashboardVersionsRepo: "dashboard-versions",
    aiChatsRepo: "ai-chats",
    vaultStateRepo: "vault-state",
    userCredentialsRepo: "user-credentials",
  };
  const counts = {};
  for (const [repo, file] of Object.entries(repoFiles)) {
    const list = byRepo[repo] ?? [];
    if (!list.length) throw new Error(`no cases for ${repo}`);
    counts[repo] = list.length;
    writeJson(path.join(OUT, "repos", `${file}.json`), {

      source: `src/lib/storage/repos/${file === "themes" ? "themes" : file}-repo.ts`,
      note: "Frozen. Each case runs on a fresh database from today's schema (schema_version 4). See ../README.md for the format.",
      cases: list,
    });
  }
  log("cases per repo:", JSON.stringify(counts));
  return { byRepo, counts, backendDiffs };
}

// Row shapes: the declared TS types of what each repository method takes and
// returns, plus what the recorded cases actually returned.

const OPAQUE = new Set(["SavedWorkflow"]);

function freezeRowShapes(byRepo) {
  const lib = path.join(REPO, "src/lib");
  const probe = path.join(WORK, "shape-probe.ts");
  fs.writeFileSync(probe, `export * as repos from ${JSON.stringify(path.join(lib, "storage/repository.ts"))};\n`);
  const program = ts.createProgram([probe], {
    strict: true,
    noEmit: true,
    skipLibCheck: true,
    allowImportingTsExtensions: true,
    module: ts.ModuleKind.ESNext,
    moduleResolution: ts.ModuleResolutionKind.Bundler,
    target: ts.ScriptTarget.ES2022,
    baseUrl: REPO,
    paths: { $lib: [lib], "$lib/*": [lib + "/*"] },
  });
  const checker = program.getTypeChecker();
  const sf = program.getSourceFile(probe);
  const modSym = checker.getSymbolAtLocation(sf).exports.get("repos");
  const reposType = checker.getTypeOfSymbolAtLocation(modSym, sf);

  const named = {};
  const str = (t) => checker.typeToString(t, undefined, ts.TypeFormatFlags.NoTruncation | ts.TypeFormatFlags.UseFullyQualifiedType).replace(/import\("[^"]+"\)\./g, "");

  function jsonKinds(t, seen = new Set()) {
    if (t.isUnion()) return [...new Set(t.types.flatMap((x) => jsonKinds(x, seen)))];
    const f = t.flags;
    if (f & (ts.TypeFlags.Any | ts.TypeFlags.Unknown)) return ["any"];
    if (f & ts.TypeFlags.StringLike) return ["string"];
    if (f & ts.TypeFlags.NumberLike) return ["number"];
    if (f & ts.TypeFlags.BooleanLike) return ["boolean"];
    if (f & ts.TypeFlags.Null) return ["null"];
    if (f & (ts.TypeFlags.Undefined | ts.TypeFlags.Void)) return ["absent"];
    const sym = t.getSymbol();
    if (sym?.name === "Date") return ["Date"];
    if (checker.isArrayType(t)) return ["array"];
    return ["object"];
  }

  function describe(t, depth = 0) {
    const nonNull = checker.getNonNullableType(t);
    const base = checker.isArrayType(nonNull) ? checker.getTypeArguments(nonNull)[0] : nonNull;
    const out = { ts: str(t), json: jsonKinds(t) };
    const lits = (nonNull.isUnion() ? nonNull.types : [nonNull]).filter((x) => x.isStringLiteral());
    if (lits.length && lits.length === (nonNull.isUnion() ? nonNull.types.length : 1)) out.enum = lits.map((x) => x.value);
    for (const part of base.isUnion() ? base.types : [base]) {
      if (!(part.flags & ts.TypeFlags.Object) || checker.isArrayType(part)) continue;
      const sym = part.aliasSymbol ?? part.getSymbol();
      const name = sym?.name;
      if (name === "Date" || name === "Promise") continue;
      if (name && OPAQUE.has(name)) {
        named[name] ??= { opaque: true, note: "stored as JSON text (saved_canvases.data via toStorable); Rust keeps it as opaque JSON" };
        out.ref = name;
      } else if (name && name !== "__type" && name !== "__object") {
        if (!named[name]) {
          named[name] = { fields: {} };
          named[name].fields = fields(part, depth + 1);
        }
        out.ref = name;
      } else if (depth < 6) {
        out.fields = fields(part, depth + 1);
      }
    }
    return out;
  }

  function fields(t, depth) {
    const res = {};
    for (const p of checker.getPropertiesOfType(t)) {
      const decl = p.valueDeclaration ?? p.declarations?.[0];
      const pt = checker.getTypeOfSymbolAtLocation(p, decl ?? sf);
      if (pt.getCallSignatures().length) continue;
      const d = describe(pt, depth);
      d.optional = (p.flags & ts.SymbolFlags.Optional) !== 0;
      d.json = d.json.filter((k) => k !== "absent" || !d.optional);
      res[p.name] = d;
    }
    return res;
  }

  const repos = {};
  for (const repoSym of checker.getPropertiesOfType(reposType)) {
    const repoType = checker.getTypeOfSymbolAtLocation(repoSym, sf);
    const methods = {};
    for (const m of checker.getPropertiesOfType(repoType)) {
      const mt = checker.getTypeOfSymbolAtLocation(m, sf);
      const sig = mt.getCallSignatures()[0];
      if (!sig) continue;
      let ret = sig.getReturnType();
      if (ret.getSymbol()?.name === "Promise") ret = checker.getTypeArguments(ret)[0];
      methods[m.name] = {
        params: sig.parameters.slice(1).map((p) => {
          const pt = checker.getTypeOfSymbolAtLocation(p, sf);
          return { name: p.name, ...describe(pt) };
        }),
        returns: describe(ret),
      };
    }
    repos[repoSym.name] = methods;
  }

  // Observed: which keys the recorded loads returned, and their JSON kinds.
  const observed = {};
  const kind = (v) => (v === null ? "null" : Array.isArray(v) ? "array" : v && typeof v === "object" && "$date" in v ? "Date" : typeof v);
  const visit = (key, obj, prefix = "") => {
    if (!obj || typeof obj !== "object" || Array.isArray(obj) || "$date" in obj) return;
    const o = (observed[key] ??= { samples: 0, keys: {} });
    if (!prefix) o.samples++;
    for (const [k, v] of Object.entries(obj)) {
      const e = (o.keys[prefix + k] ??= { present: 0, kinds: [] });
      e.present++;
      if (!e.kinds.includes(kind(v))) e.kinds.push(kind(v));
    }
  };
  for (const cases of Object.values(byRepo)) {
    for (const c of cases) {
      for (const s of c.steps) {
        if (!s.call || s.result === undefined) continue;
        const items = Array.isArray(s.result) ? s.result : [s.result];
        for (const it of items) visit(s.call, it);
        if (s.call === "sharedReposRepo.loadAll") for (const r of s.result.repos) visit("sharedReposRepo.loadAll repos[]", r);
      }
    }
  }
  for (const o of Object.values(observed)) {
    for (const e of Object.values(o.keys)) e.always = e.present === o.samples;
  }

  writeJson(path.join(OUT, "row-shapes.json"), {
    note:
      "Frozen. `repos`: each repository method's declared parameters and result, from the TypeScript checker over src/lib/storage/repository.ts (the db argument dropped). `types`: the named types they use. `json` lists the JSON kinds a value can take; `absent` means the key may be missing, `Date` means a JS Date (an ISO string on the wire). `observed`: the keys each load returned across the recorded cases, with `always` when every sample had it. Task 4's ts-rs types must regenerate to the same shapes.",
    repos,
    types: named,
    observed,
  });
}

    fs.unlinkSync(path.join(WORK, "node_modules")); // the symlink, never the repo's packages

  }
}
