› top 10 customers by revenue this quarter, only @invoices with status paid, include @customers.country

1SELECT c.name, c.country, sum(li.quantity * li.unit_price) AS revenue
2FROM   invoices i
3JOIN   customers c ON c.id = i.customer_id
4JOIN   invoice_line_items li ON li.invoice_id = i.id
5WHERE  i.status = 'paid'
6  AND  i.issued_at >= date_trunc('quarter', now())
7GROUP BY c.id, c.name, c.country
8ORDER BY revenue DESC LIMIT 10;
