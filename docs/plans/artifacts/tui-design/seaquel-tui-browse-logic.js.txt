  fmtTime(sec) { const p = n => String(n).padStart(2, '0'); return `${p(Math.floor(sec / 3600) % 24)}:${p(Math.floor(sec / 60) % 60)}:${p(sec % 60)}`; }
  addLog(s, entries) {
    let clock = s.clock + 3 + Math.floor(Math.random() * 6);
    const log = [...s.log, ...entries.map(e => ({ tag: '', tagColor: '', ms: '', ...e, time: this.fmtTime(clock) }))].slice(-4);
    return { log, clock };
  }
  sqlVal(v, k) { if (v === null) return 'NULL'; return (k === 'id' || k === 'total') ? v : `'${v}'`; }
  fmt(v, k) { if (v === null || v === undefined) return 'NULL'; if (k === 'total' && !isNaN(+v) && v !== '') return (+v).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); return v; }
  visRows(s = this.state) { const f = s.filter.toLowerCase(); return s.base.filter(r => !f || [r.id, r.customer, r.status].some(v => String(v).toLowerCase().includes(f))); }
  val(s, r, k) { const key = r.id + '.' + k; return key in s.edits ? s.edits[key] : r[k]; }
  pend(s = this.state) {
    return [...Object.keys(s.edits).map(key => { const [id, col] = key.split('.'); return { t: 'M', key, id, col }; }), ...s.dels.map(id => ({ t: 'D', id }))];
  }
  tableName(s = this.state) { return Component.TABS2.Tables[s.ti][0]; }
  isGrid(s = this.state) { return s.ctx === 'tables' && s.tablesTab === 0 && this.tableName(s) === 'invoices' && s.mainTab === 0; }

  focus(p) { this.setState(s => ({ focus: p, ctx: p === 'main' ? s.ctx : (p === 'conn' ? s.ctx : p), editing: null, filtering: false })); }
  clamp(v, n) { return Math.max(0, Math.min(n - 1, v)); }

  startEdit() {
    const s = this.state, r = this.visRows()[s.row]; if (!r) return;
    const k = Component.COLS[s.col][0];
    if (k === 'id') { this.setState(s => this.addLog(s, [{ tag: 'read-only ', tagColor: '#ff7b72', text: 'id is the primary key' }])); return; }
    if (s.dels.includes(r.id)) return;
    const v = this.val(s, r, k);
    this.setState({ editing: { id: r.id, k, v: v === null ? 'NULL' : v } });
  }
  applyEdit() {
    this.setState(s => {
      const ed = s.editing, r = s.base.find(x => x.id === ed.id), key = ed.id + '.' + ed.k;
      const nv = ed.v === 'NULL' ? null : ed.v, edits = { ...s.edits }, prev = s.edits[key];
      if (nv === r[ed.k]) delete edits[key]; else edits[key] = nv;
      if (edits[key] === prev) return { editing: null };
      return { editing: null, edits, hist: [...s.hist, { t: 'edit', key, prev }],
        ...this.addLog(s, [{ tag: 'staged ', tagColor: '#e3b341', text: `UPDATE invoices SET ${ed.k} = ${this.sqlVal(nv, ed.k)} WHERE id = ${ed.id}` }]) };
    });
  }
  toggleDel(id) {
    this.setState(s => {
      const has = s.dels.includes(id);
      return { dels: has ? s.dels.filter(x => x !== id) : [...s.dels, id], hist: [...s.hist, { t: has ? 'undel' : 'del', id }],
        ...this.addLog(s, [has ? { tag: 'unstaged ', tagColor: '#8b949e', text: `DELETE FROM invoices WHERE id = ${id}` } : { tag: 'staged ', tagColor: '#ff7b72', text: `DELETE FROM invoices WHERE id = ${id}` }]) };
    });
  }
  unstage(item) {
    this.setState(s => {
      if (item.t === 'M') { const edits = { ...s.edits }; const prev = edits[item.key]; delete edits[item.key]; return { edits, hist: [...s.hist, { t: 'edit', key: item.key, prev }], ...this.addLog(s, [{ tag: 'unstaged ', tagColor: '#8b949e', text: `UPDATE invoices … WHERE id = ${item.id}` }]) }; }
      return { dels: s.dels.filter(x => x !== item.id), hist: [...s.hist, { t: 'undel', id: item.id }], ...this.addLog(s, [{ tag: 'unstaged ', tagColor: '#8b949e', text: `DELETE FROM invoices WHERE id = ${item.id}` }]) };
    });
  }
  undo() {
    this.setState(s => {
      const h = s.hist[s.hist.length - 1]; if (!h) return null;
      const out = { hist: s.hist.slice(0, -1) };
      if (h.t === 'edit') { const edits = { ...s.edits }; if (h.prev === undefined) delete edits[h.key]; else edits[h.key] = h.prev; out.edits = edits; }
      if (h.t === 'del') out.dels = s.dels.filter(x => x !== h.id);
      if (h.t === 'undel') out.dels = [...s.dels, h.id];
      return { ...out, ...this.addLog(s, [{ tag: 'undo ', tagColor: '#d2a8ff', text: h.t === 'edit' ? `edit ${h.key.replace('.', ' · ')}` : `${h.t === 'del' ? 'delete' : 'unstage'} id ${h.id}` }]) };
    });
  }
  commit() {
    this.setState(s => {
      const p = this.pend(s);
      const base = s.base.filter(r => !s.dels.includes(r.id)).map(r => { const n = { ...r }; Component.COLS.forEach(([k]) => { const key = r.id + '.' + k; if (key in s.edits) n[k] = s.edits[key]; }); return n; });
      const lg = this.addLog(s, [
        { tag: '', text: 'BEGIN;', ms: '1 ms' },
        ...p.map(i => i.t === 'M' ? { text: `UPDATE invoices SET ${i.col} = ${this.sqlVal(s.edits[i.key], i.col)} WHERE id = ${i.id};`, ms: '3 ms' } : { text: `DELETE FROM invoices WHERE id = ${i.id};`, ms: '2 ms' }),
        { tag: 'committed ', tagColor: '#7ee787', text: `COMMIT; ${p.length} statement${p.length === 1 ? '' : 's'}`, ms: '6 ms' }
      ]);
      const n = base.filter(r => !s.filter || this.visRows({ ...s, base }).includes(r)).length;
      return { base, edits: {}, dels: [], hist: [], modal: null, pi: 0, row: this.clamp(s.row, Math.max(1, n)), ...lg };
    });
  }
  cycleTab(d) {
    this.setState(s => {
      const f = s.focus === 'main' ? s.ctx : s.focus;
      if (s.focus === 'main' && s.ctx === 'tables' && s.tablesTab === 0) return { mainTab: (s.mainTab + d + 5) % 5 };
      if (f === 'tables') return { tablesTab: (s.tablesTab + d + 4) % 4, oi: 0 };
      if (f === 'saved') return { savedTab: (s.savedTab + 1) % 2, si: 0 };
      return null;
    });
  }

  onKey(e) {
    const tg = e.target; if (tg && (tg.tagName === 'INPUT' || tg.tagName === 'TEXTAREA' || tg.isContentEditable)) return;
    if (e.metaKey || e.ctrlKey || e.altKey) return;
    const k = e.key, s = this.state, done = () => e.preventDefault(), P = Component.P;
    if (s.modal === 'help') { if (['Escape', '?', 'q', 'Enter'].includes(k)) this.setState({ modal: null }); return done(); }
    if (s.modal === 'commit') { if (k === 'Enter') this.commit(); else if (k === 'Escape' || k === 'q') this.setState({ modal: null }); return done(); }
    if (s.editing) {
      const ed = s.editing;
      if (k === 'Escape') this.setState({ editing: null });
      else if (k === 'Enter') this.applyEdit();
      else if (k === 'Backspace') this.setState({ editing: { ...ed, v: ed.v.slice(0, -1) } });
      else if (k.length === 1) this.setState({ editing: { ...ed, v: ed.v + k } });
      else return;
      return done();
    }
    if (s.filtering) {
      if (k === 'Escape') this.setState({ filtering: false, filter: '', row: 0 });
      else if (k === 'Enter') this.setState({ filtering: false });
      else if (k === 'Backspace') this.setState({ filter: s.filter.slice(0, -1), row: 0 });
      else if (k.length === 1) this.setState({ filter: s.filter + k, row: 0 });
      else return;
      return done();
    }
    if (k >= '0' && k <= '4') { this.focus(k === '0' ? 'main' : P[+k - 1]); return done(); }
    if (k === 'Tab') { const i = P.indexOf(s.focus); this.focus(P[(i + (e.shiftKey ? P.length - 1 : 1)) % P.length]); return done(); }
    if (k === '?') { this.setState({ modal: 'help' }); return done(); }
    if (k === 'c') { if (this.pend().length) this.setState({ modal: 'commit' }); return done(); }
    if (k === 'u') { this.undo(); return done(); }
    if (k === '[' || k === ']') { this.cycleTab(k === ']' ? 1 : -1); return done(); }
    const down = k === 'j' || k === 'ArrowDown', up = k === 'k' || k === 'ArrowUp', left = k === 'h' || k === 'ArrowLeft', right = k === 'l' || k === 'ArrowRight';
    const dy = down ? 1 : up ? -1 : 0, f = s.focus;
    if (f === 'main') {
      if (k === 'Escape') { if (s.filter) this.setState({ filter: '', row: 0 }); else this.focus(s.ctx); return done(); }
      if (!this.isGrid()) return;
      const n = this.visRows().length;
      if (dy) this.setState({ row: this.clamp(s.row + dy, n) });
      else if (left || right) this.setState({ col: this.clamp(s.col + (right ? 1 : -1), 7) });
      else if (k === 'g') this.setState({ row: 0 });
      else if (k === 'G') this.setState({ row: n - 1 });
      else if (k === 'e' || k === 'Enter') this.startEdit();
      else if (k === 'd') { const r = this.visRows()[s.row]; if (r) this.toggleDel(r.id); }
      else if (k === '/') this.setState({ filtering: true });
      else return;
      return done();
    }
    if (left || right) { const i = P.indexOf(f); this.focus(P[(i + (right ? 1 : 3)) % 4]); return done(); }
    if (f === 'tables') {
      const tab = Object.keys(Component.TABS2)[s.tablesTab], n = Component.TABS2[tab].length;
      if (dy) this.setState(s.tablesTab === 0 ? { ti: this.clamp(s.ti + dy, n), mainTab: 0, row: 0, filter: '' } : { oi: this.clamp(s.oi + dy, n) });
      else if (k === 'Enter') this.focus('main');
      else return;
      return done();
    }
    if (f === 'saved') {
      const n = s.savedTab === 0 ? Component.SAVED.length : Component.HIST.length;
      if (dy) this.setState({ si: this.clamp(s.si + dy, n) });
      else if (k === 'Enter') this.focus('main');
      else return;
      return done();
    }
    if (f === 'pending') {
      const p = this.pend();
      if (dy) this.setState({ pi: this.clamp(s.pi + dy, p.length) });
      else if ((k === ' ' || k === 'd') && p[s.pi]) { this.unstage(p[s.pi]); this.setState({ pi: this.clamp(s.pi, p.length - 1) }); }
      else if (k === 'Enter') this.focus('main');
      else return;
      return done();
    }
  }

        '  issued_at   date NOT NULL DEFAULT current_date,', '  due_at      date,', '  total       numeric(12,2) NOT NULL DEFAULT 0 CHECK (total >= 0),', '  paid_at     timestamptz', ');'].map(t => L(t));
    } else if (s.ctx === 'tables') {
      const [n, kind] = list[s.oi];
      mainTabs = tabs(['Definition'], 0, 'main'); mainRight = 'public.' + n;
      lines = [L(`-- ${tabName.slice(0, -1).toLowerCase()} · ${kind}`, '#6e7681'), L(`public.${n}`, '#d2a8ff'), L(''), L('Definitions are stubbed in this mockup.', '#8b949e')];

    } else {
      mainTabs = tabs(['Diff', 'SQL'], 0, 'main');
      const it = p[s.pi];

      const cells = {};
      C.COLS.forEach(([k], ci) => {
        const key = r.id + '.' + k, v = this.val(s, r, k), isEd = s.editing && s.editing.id === r.id && s.editing.k === k;
        let color = v === null ? '#6e7681' : k === 'id' ? '#8b949e' : k === 'status' ? (C.STATUS[v] || '#c9d1d9') : '#c9d1d9';
        if (key in s.edits) color = '#e3b341';
        let bg = 'transparent', text = this.fmt(v, k);
        if (del) color = '#ff7b72';
        if (sel && ci === s.col && on('main')) { bg = key in s.edits ? '#e3b341' : '#58a6ff'; color = '#0d1117'; }
        if (isEd) { bg = '#30363d'; color = '#ffffff'; text = s.editing.v + '▌'; }
        cells['c' + ci] = { text, color, bg, align: k === 'total' && !isEd ? 'right' : 'left', deco: del ? 'line-through' : 'none',
          onClick: e => { e.stopPropagation(); this.setState({ focus: 'main', row: ri, col: ci }); } };
      });
      return { ...cells, bg: selBg(sel, 'main'), weight: selW(sel, 'main'), mark: del ? '-' : edited ? '~' : '', markColor: del ? '#ff7b72' : '#e3b341' };
    });
    const cr = vr[row], ck = C.COLS[s.col];
    let cellInfo = '', cellChange = '';
    if (cr) { cellInfo = `${ck[0]} · ${ck[1]}  `; const key = cr.id + '.' + ck[0]; if (key in s.edits) cellChange = `${this.fmt(cr[ck[0]], ck[0])} → ${this.fmt(s.edits[key], ck[0])}`; }
    if (showGrid) mainCounter = vr.length ? `${row + 1} of ${vr.length}` : '0 of 0';
    else if (s.ctx === 'pending') mainCounter = '';

    // key bar
    const K = arr => arr.map(([label, key], i) => ({ label, key, sep: i ? ' | ' : '' }));
    let keys, modeLabel = 'seaquel 2026.4.5';
    if (s.modal === 'commit') keys = K([['Execute', 'enter'], ['Cancel', 'esc']]);
    else if (s.modal === 'help') keys = K([['Close', 'esc']]);
    else if (s.editing) { keys = K([['Save', 'enter'], ['Cancel', 'esc'], ['Set NULL', 'type NULL']]); modeLabel = 'EDIT'; }
    else if (s.filtering) { keys = K([['Apply', 'enter'], ['Clear', 'esc']]); modeLabel = 'FILTER'; }
    else if (on('main') && showGrid) keys = K([['Move', 'hjkl'], ['Edit', 'e'], ['Stage delete', 'd'], ['Filter', '/'], ['Tabs', '[ ]'], ['Undo', 'u'], ['Commit', 'c'], ['Back', 'esc'], ['Keybindings', '?']]);
    else if (on('main')) keys = K([['Tabs', '[ ]'], ['Back', 'esc'], ['Commit', 'c'], ['Keybindings', '?']]);
    else if (on('tables')) keys = K([['Select', 'j/k'], ['Open', 'enter'], ['Tabs', '[ ]'], ['Next panel', 'tab'], ['Undo', 'u'], ['Commit', 'c'], ['Keybindings', '?']]);
    else if (on('saved')) keys = K([['Select', 'j/k'], ['Open', 'enter'], ['Saved/History', '[ ]'], ['Next panel', 'tab'], ['Keybindings', '?']]);
    else if (on('pending')) keys = K([['Select', 'j/k'], ['Unstage', 'space'], ['Undo', 'u'], ['Commit', 'c'], ['View diff', 'enter'], ['Keybindings', '?']]);
    else keys = K([['Next panel', 'tab'], ['Focus panel', '1-4'], ['Main', '0'], ['Keybindings', '?']]);

    const nIns = 0;
    const commitLines = [];
    if (nUpd) commitLines.push({ text: `~${nUpd}  UPDATE   public.invoices`, color: '#e3b341' });
    if (nDel) commitLines.push({ text: `-${nDel}  DELETE   public.invoices`, color: '#ff7b72' });

    const H = (k, d, head) => ({ k, d, kc: head ? G : '#58a6ff', kw: head ? 700 : 400 });
    const helpLines = [H('Global', '', 1), H('1 2 3 4 / 0', 'focus panel / main view'), H('tab  shift+tab', 'next / previous panel'), H('[  ]', 'previous / next tab'), H('u', 'undo last staged change'), H('c', 'commit pending changes'),
      H('Data grid', '', 1), H('h j k l  arrows', 'move cell cursor'), H('g  G', 'first / last row'), H('e  enter', 'edit cell (type NULL for null)'), H('d', 'stage / unstage row delete'), H('/', 'filter rows · esc clears'),
      H('Pending changes', '', 1), H('space', 'unstage selected change'), H('enter', 'view diff in main')];

    const f = fn => e => { fn(); };
    return {
      b0: bd('main'), b1: bd('conn'), b2: bd('tables'), b3: bd('saved'), b4: bd('pending'),
      n1: nc('conn'), n2: nc('tables'), n3: nc('saved'), n4: nc('pending'), w1: on('conn') ? 700 : 400, w4: on('pending') ? 700 : 400,
      n2c: nc('tables'), n3c: nc('saved'), n4c: nc('pending'), n0c: nc('main'),
      fConn: f(() => this.focus('conn')), fTables: f(() => this.focus('tables')), fSaved: f(() => this.focus('saved')), fPending: f(() => this.focus('pending')), fMain: f(() => s.focus !== 'main' && this.focus('main')),
      connPending: p.length ? `● ${p.length}` : '○ 0', connColor: p.length ? '#e3b341' : '#6e7681',
      tablesTabs: tabs(Object.keys(C.TABS2), s.tablesTab, 'tables'), tableItems, tablesCounter: `${idx2 + 1} of ${list.length}`,
      savedTabs: tabs(['Saved', 'History'], s.savedTab, 'saved'), savedItems, savedCounter: `${s.si + 1} of ${savedList.length}`,
      pendingItems, noPending: !p.length, pendingCounter: p.length ? `${this.clamp(s.pi, p.length) + 1} of ${p.length}` : '',
      pIns: `+${nIns}`, pUpd: `~${nUpd}`, pDel: `-${nDel}`,
      mainTabs, mainRight, showGrid, lines: showGrid ? [] : lines, rows, noRows: showGrid && !vr.length,
      filterText: s.filtering ? s.filter + '▌' : (s.filter || 'filter…'), filterColor: s.filtering || s.filter ? '#c9d1d9' : '#6e7681',
      cellInfo, cellChange, gridFooter: s.filter ? `${vr.length} match${vr.length === 1 ? '' : 'es'} · 18 ms` : `rows 1–${vr.length} of 48,112 · 18 ms`, mainCounter,
      log: s.log, keys, modeLabel,
      showCommit: s.modal === 'commit', commitTitle: `Commit ${p.length} change${p.length === 1 ? '' : 's'}`, commitLines, commitHasDelete: nDel > 0,
      showHelp: s.modal === 'help', helpLines
    };
  }
