const SCHEMA_DIRS = ["postgresql", "mysql", "mariadb", "sqlite", "sqlserver", "duckdb"];

export function buildScannerCorpus(): ScannerInput[] {
  const out: ScannerInput[] = [];
  const seen = new Map<string, string>();
  const add = (id: string, sql: string, values?: Record<string, ParamValue>[]) => {
    const dup = seen.get(sql);
    if (dup !== undefined && !values) return; // the same SQL from another source adds nothing
    if (out.some((e) => e.id === id)) throw new Error(`duplicate scanner corpus id ${id}`);
    seen.set(sql, id);
    out.push(values ? { id, sql, values } : { id, sql });
  };
  for (const [id, sql] of SPIKE) add(id, sql);
  SQL_SCAN_TEST.forEach((sql, i) => add(`sql-scan-test:${i}`, sql));
  QUERY_PARAMS_TEST.forEach(([sql, values], i) => add(`query-params-test:${i}`, sql, values));
  for (const [id, sql, values] of PHASE2_REVIEW) add(id, sql, values);
  for (const dir of SCHEMA_DIRS) {
    add(
      `schema:${dir}`,
      readFileSync(new URL(`e2e/test-databases/${dir}/schema.sql`, repo), "utf8"),
    );
  }
  for (const qs of Object.values(sampleQueries)) for (const q of qs) add(`sample:${q.id}`, q.query);
  for (const [id, sql, values] of EDGE) add(id, sql, values);
  return out;
}
