/** The fixed quote: `"` doubled (fix 1). */
const qd = (s: string) => `"${s.replaceAll('"', '""')}"`;

/**
 * The fixes the DDL and EXPLAIN models apply. Every fix can be switched off:
 * with none on, the models must reproduce the recorded TypeScript output (checked
 * below), and a case's fix numbers are the ones whose removal changes its output.
 * Fix 6 here is its DDL side: a listed schema is split quote-aware
 * (`quoteSchema`, the port's `Dialect::quote_schema`).
 */
const ON = new Set<number>([1, 4, 5, 6, 8, 9, 10, 11, 13, 14]);
const DDL_FIXES = [1, 4, 6, 8, 9, 10, 11];
const EXPLAIN_FIXES = [5, 13, 14];
const quote = () => (ON.has(1) ? qd : (s: string) => `"${s}"`);

/**
 * The parts of a dotted name in DuckDB's identifier syntax: `"…"` parts with
 * `""` for `"`, bare parts up to the next dot. Null when it doesn't parse.
 */
function parseDotted(name: string): string[] | null {
  const parts: string[] = [];
  let i = 0;
  while (i < name.length) {
    let part = "";
    if (name[i] === '"') {
      i++;
      for (;;) {
        if (i >= name.length) return null;
        if (name[i] === '"') {
          if (name[i + 1] === '"') {
            part += '"';
            i += 2;
            continue;
          }
          i++;
          break;
        }
        part += name[i++];
      }
    } else {
      while (i < name.length && name[i] !== ".") part += name[i++];
    }
    parts.push(part);
    if (i < name.length) {
      if (name[i] !== ".") return null;
      i++;
      if (i === name.length) return null;
    }
  }
  return parts.length > 0 ? parts : null;
}

/** Fix 14: the last part of a dotted name; an unparseable name stays as it is. */
function lastPart(name: string): string {
  const parts = parseDotted(name);
  return parts ? parts[parts.length - 1] : name;
}

/**
 * A schema as the tree lists it, quoted for SQL (fix 6's DDL side): `main` →
 * `"main"`, `"fx.a.b"` → `"fx.a.b"`, `fx_aux.main` → `"fx_aux"."main"`.
 */
function quoteSchema(schema: string): string {
  const q = quote();
  if (!ON.has(6)) return q(schema);
  const parts = parseDotted(schema);
  return parts && parts.length <= 2 ? parts.map((p) => q(p)).join(".") : q(schema);
}
/** The attached catalog a listed schema names, if any. */
function catalogOf(schema: string): string | undefined {
  const parts = ON.has(6) ? parseDotted(schema) : null;
  return parts && parts.length === 2 ? parts[0] : undefined;
}

/** Fix 9: a NOT NULL column added without a default can't be made NOT NULL on a table with rows. */
const notNullNote = (table: string, q: (s: string) => string, column: string) =>
  note(
    `DuckDB can't add column ${q(column)} as NOT NULL without a default; fill it, then run ALTER TABLE ${table} ALTER COLUMN ${q(column)} SET NOT NULL`,
  );

/** TS `buildColumnLine`. */
function columnLine(c: CreateTableColumn, q: (s: string) => string): string {
  let line = `  ${q(c.name)} ${buildColumnType(c)}`;
  if (!c.nullable) line += " NOT NULL";
  if (c.defaultValue) {
    const safe = sanitizeDefaultValue(c.defaultValue);
    if (safe) line += ` DEFAULT ${safe}`;
  }
  return line;
}

/**
 * Fixed CREATE TABLE (the TypeScript `generateCreateTableDdl` with every fix
 * off). Fix 1: the quote doubles ". Fix 6: the schema is split. Fix 11: a
 * foreign key references the table's own schema; in an attached catalog it
 * can't be created at all, so it is a note.
 */
function createSql(def: CreateTableDefinition): string {
  const q = quote();
  const table = `${quoteSchema(def.schemaName)}.${q(def.tableName)}`;
  const lines = def.columns.map((c) => columnLine(c, q));
  const pk = def.columns.filter((c) => c.isPrimaryKey);
  if (pk.length > 0) lines.push(`  PRIMARY KEY (${pk.map((c) => q(c.name)).join(", ")})`);
  for (const c of def.columns) if (c.isUnique && !c.isPrimaryKey) lines.push(`  UNIQUE (${q(c.name)})`);
  const notes: string[] = [];
  const catalog = catalogOf(def.schemaName);
  for (const f of def.foreignKeys) {
    const ref = ON.has(11)
      ? `${quoteSchema(def.schemaName)}.${q(f.referencedTable)}`
      : `${q(f.referencedSchema)}.${q(f.referencedTable)}`;
    const clause = `(${q(f.column)}) REFERENCES ${ref} (${q(f.referencedColumn)})`;
    if (ON.has(11) && catalog !== undefined) {
      notes.push(
        note(`DuckDB can't create a foreign key in attached catalog ${q(catalog)} from here: ${clause}; create the table after USE ${q(catalog)} to add it`),
      );
    } else {
      lines.push(`  FOREIGN KEY ${clause}`);
    }
  }
  let sql = `CREATE TABLE ${table} (\n${lines.join(",\n")}\n);`;
  for (const i of def.indexes) {
    sql += `\n\nCREATE ${i.unique ? "UNIQUE " : ""}INDEX ${q(i.name)} ON ${table} (${i.columns.map((c) => q(c)).join(", ")});`;
  }
  return [sql, ...notes].join("\n");
}

/** Fixed ADD COLUMN: fix 1's quote, fix 6's schema; fix 9 adds NOT NULL as its own statement (or a note). */
function addColumnSql(schema: string, tableName: string, col: CreateTableColumn): string {
  const q = quote();
  const table = `${quoteSchema(schema)}.${q(tableName)}`;
  const safe = col.defaultValue ? sanitizeDefaultValue(col.defaultValue) : "";
  if (!ON.has(9)) {
    return `ALTER TABLE ${table} ADD COLUMN ${q(col.name)} ${buildColumnType(col)}${col.nullable ? "" : " NOT NULL"}${safe ? ` DEFAULT ${safe}` : ""};`;
  }
  const out = [`ALTER TABLE ${table} ADD COLUMN ${q(col.name)} ${buildColumnType(col)}${safe ? ` DEFAULT ${safe}` : ""};`];
  if (!col.nullable) {
    out.push(safe ? `ALTER TABLE ${table} ALTER COLUMN ${q(col.name)} SET NOT NULL;` : notNullNote(table, q, col.name));
  }
  return out.join("\n");
}

/**
 * Fixed ALTER TABLE. With every fix off this is the TypeScript generator.
 * Fix 1: the quote doubles ". Fix 4: an added foreign key is a note. Fix 6:
 * the schema is split. Fix 8: DROP INDEX is qualified with the original
 * schema. Fix 9: NOT NULL on an added column is a separate SET NOT NULL (or a
 * note without a default). Fix 10: removed indexes are dropped first
 * (`drop_indexes_first`); while indexes the edit keeps remain, every statement
 * DuckDB rejects on an indexed table (RENAME COLUMN, DROP COLUMN, TYPE,
 * SET/DROP NOT NULL) is a note carrying that statement instead, and so is a
 * DROP COLUMN or TYPE change of a column in a PRIMARY KEY or UNIQUE
 * constraint, and a DROP COLUMN of a column that comes before one (the
 * constraint's ART index depends on a later column); DuckDB rejects those
 * whatever the CREATE INDEX indexes.
 */
function alterSql(from: CreateTableDefinition, to: CreateTableDefinition): string {
  const q = quote();
  const table = `${quoteSchema(to.schemaName)}.${q(to.tableName)}`;
  const stmts: string[] = [];
  const notes: string[] = [];
  const origCols = new Map(from.columns.map((c) => [c.id, c]));
  const newCols = new Map(to.columns.map((c) => [c.id, c]));
  const origIdx = new Set(from.indexes.map((i) => i.name));
  const newIdx = new Set(to.indexes.map((i) => i.name));
  const kept = ON.has(10) ? from.indexes.filter((i) => i.name && newIdx.has(i.name)) : [];
  const keptNames = kept.map((i) => `${quoteSchema(from.schemaName)}.${q(i.name)}`).join(", ");
  /** A statement DuckDB rejects while the table has indexes. */
  const blocked = (stmt: string) => {
    if (kept.length === 0) {
      stmts.push(stmt);
      return;
    }
    notes.push(
      note(
        `DuckDB can't run this while the table has indexes; drop ${keptNames} first and recreate ${kept.length > 1 ? "them" : "it"} afterwards: ${stmt.replace(/;$/, "")}`,
      ),
    );
  };
  /** DuckDB can't drop or retype a column in a PRIMARY KEY or UNIQUE constraint. */
  const constrained = (c: CreateTableColumn, stmt: string) => {
    if (ON.has(10) && (c.isPrimaryKey || c.isUnique)) {
      notes.push(
        note(
          `DuckDB can't drop or change the type of a column in a PRIMARY KEY or UNIQUE constraint; recreate the table to change it: ${stmt.replace(/;$/, "")}`,
        ),
      );
      return;
    }
    blocked(stmt);
  };
  /** New names of columns whose rename became a note: later statements naming them can't run either. */
  const unrenamed = new Set<string>();
  const afterRename = (names: string[], stmt: string) => {
    if (names.some((n) => unrenamed.has(n))) {
      notes.push(note(`DuckDB can't run this before the rename above: ${stmt.replace(/;$/, "")}`));
    } else {
      stmts.push(stmt);
    }
  };
  for (const c of to.columns) {
    const o = origCols.get(c.id);
    if (o && o.name !== c.name) {
      blocked(`ALTER TABLE ${table} RENAME COLUMN ${q(o.name)} TO ${q(c.name)};`);
      if (kept.length > 0) unrenamed.add(c.name);
    }
  }
  for (const c of to.columns) {
    if (origCols.has(c.id)) continue;
    const safe = c.defaultValue ? sanitizeDefaultValue(c.defaultValue) : "";
    if (!ON.has(9)) {
      let st = `ALTER TABLE ${table} ADD COLUMN ${q(c.name)} ${buildColumnType(c)}`;
      if (!c.nullable) st += " NOT NULL";
      if (safe) st += ` DEFAULT ${safe}`;
      stmts.push(st + ";");
      continue;
    }
    stmts.push(`ALTER TABLE ${table} ADD COLUMN ${q(c.name)} ${buildColumnType(c)}${safe ? ` DEFAULT ${safe}` : ""};`);
    if (!c.nullable) {
      if (safe) blocked(`ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} SET NOT NULL;`);
      else notes.push(notNullNote(table, q, c.name));
    }
  }
  for (const c of from.columns) {
    if (newCols.has(c.id)) continue;
    const stmt = `ALTER TABLE ${table} DROP COLUMN ${q(c.name)};`;
    const later = from.columns.slice(from.columns.indexOf(c) + 1);
    if (ON.has(10) && !(c.isPrimaryKey || c.isUnique) && later.some((x) => x.isPrimaryKey || x.isUnique)) {
      // The constraint's ART index depends on a column after this one.
      notes.push(
        note(
          `DuckDB can't drop a column that comes before a PRIMARY KEY or UNIQUE column; recreate the table to change it: ${stmt.replace(/;$/, "")}`,
        ),
      );
      continue;
    }
    constrained(c, stmt);
  }
  for (const c of to.columns) {
    const o = origCols.get(c.id);
    if (!o) continue;
    const ot = buildColumnType(o);
    const nt = buildColumnType(c);
    if (ot !== nt) constrained(o, `ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} TYPE ${nt};`);
    if (o.nullable && !c.nullable) blocked(`ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} SET NOT NULL;`);
    else if (!o.nullable && c.nullable) blocked(`ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} DROP NOT NULL;`);
    if (o.defaultValue !== c.defaultValue) {
      const safe = sanitizeDefaultValue(c.defaultValue);
      afterRename(
        [c.name],
        safe
          ? `ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} SET DEFAULT ${safe};`
          : `ALTER TABLE ${table} ALTER COLUMN ${q(c.name)} DROP DEFAULT;`,
      );
    }
  }
  const dropped = from.indexes
    .filter((i) => i.name && !newIdx.has(i.name))
    .map((i) => `DROP INDEX ${ON.has(8) ? `${quoteSchema(from.schemaName)}.` : ""}${q(i.name)};`);
  if (ON.has(10)) stmts.unshift(...dropped);
  else stmts.push(...dropped);
  for (const i of to.indexes) {
    if (i.name && !origIdx.has(i.name)) {
      afterRename(
        i.columns,
        `CREATE ${i.unique ? "UNIQUE " : ""}INDEX ${q(i.name)} ON ${table} (${i.columns.map((c) => q(c)).join(", ")});`,
      );
    }
  }
  const fkKey = (f: { column: string; referencedTable: string; referencedColumn: string }) =>
    `${f.column}->${f.referencedTable}.${f.referencedColumn}`;
  const origFks = new Set(from.foreignKeys.map(fkKey));
  for (const f of to.foreignKeys) {
    if (origFks.has(fkKey(f))) continue;
    const ref = f.referencedSchema ? `${q(f.referencedSchema)}.${q(f.referencedTable)}` : q(f.referencedTable);
    if (ON.has(4)) {
      notes.push(
        note(
          `DuckDB can't add a foreign key to an existing table: (${q(f.column)}) REFERENCES ${ref} (${q(f.referencedColumn)}); recreate the table to add it`,
        ),
      );
    } else {
      stmts.push(`ALTER TABLE ${table} ADD FOREIGN KEY (${q(f.column)}) REFERENCES ${ref} (${q(f.referencedColumn)});`);
    }
  }
  const all = [...stmts, ...notes];
  return all.length === 0 ? "-- No changes detected" : all.join("\n");
}

/** Fixed EXPLAIN parsing: the TypeScript output with fixes 5, 13 and 14 applied node by node. */
// eslint-disable-next-line @typescript-eslint/no-explicit-any
function explainOutput(input: any, tsOut: any): any {
  const out = structuredClone(tsOut);
  let parsed: unknown;
  try {
    parsed = JSON.parse(input.rows[0]?.explain_value ?? "");
  } catch {
    return out;
  }
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  let root: any;
  if (input.analyze) {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const first = (parsed as any)?.children?.[0];
    root = first?.operator_name?.trim() === "EXPLAIN_ANALYZE" ? first?.children?.[0] : first;
  } else {
    root = Array.isArray(parsed) ? parsed[0] : undefined;
  }
  if (!root) return out;
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const walk = (raw: any, node: any) => {
    const extra = raw.extra_info ?? {};
    if (ON.has(5)) {
      const rows = cardinality(extra["Estimated Cardinality"]);
      if (rows !== undefined) node.planRows = rows;
    }
    if (ON.has(13) && node.nodeType === "FILTER" && typeof extra.Expression === "string" && node.filter === undefined) {
      node.filter = extra.Expression;
    }
    if (ON.has(14) && typeof node.relationName === "string") node.relationName = lastPart(node.relationName);
    (raw.children ?? []).forEach((c: unknown, k: number) => walk(c, node.children[k]));
  };
  walk(root, out.plan);
  return out;
}

/** The fixes among `candidates` whose removal changes `compute()`'s result. */
function fixesOf(candidates: number[], compute: () => unknown): number[] {
  const full = JSON.stringify(compute());
  return candidates.filter((f) => {
    ON.delete(f);
    const without = JSON.stringify(compute());
    ON.add(f);
    return without !== full;
  });
}

/** With every fix off, `compute()` must equal the recorded TypeScript output. */
function assertTsParity(label: string, compute: () => unknown, recordedOutput: unknown): void {
  const saved = [...ON];
  ON.clear();
  const got = JSON.stringify(toJson(compute()));
  for (const f of saved) ON.add(f);
  if (got !== JSON.stringify(recordedOutput)) throw new Error(`model differs from the TypeScript: ${label}\n${got}`);
}

/** Fixed indexes parser: rows of the fixed query, grouped in order (fix 2). */
function parseIndexes(rows: Record<string, unknown>[]) {
  const out: { name: string; columns: string[]; unique: boolean; type: string }[] = [];
  for (const r of rows) {
    const name = r.index_name as string;
    let idx = out.find((i) => i.name === name);
    if (!idx) {
      idx = { name, columns: [], unique: r.is_unique === true, type: "art" };
      out.push(idx);
    }
    idx.columns.push(r.column_name as string);
  }
  return out;
}

/** `format_bytes` in the SQLite port (the TS `formatBytes`). */
function formatBytes(bytes: number): string {
  if (bytes === 0) return "0 bytes";
  const sizes = ["bytes", "KB", "MB", "GB", "TB"];
  const i = Math.floor(Math.log(bytes) / Math.log(1024));
  return parseFloat((bytes / Math.pow(1024, i)).toFixed(2)) + " " + sizes[i];
}

/** Fixed overview parser (fix 3). */
function parseOverview(rows: Record<string, unknown>[]) {
  const row = rows[0];
  const bytes = Number(row?.total_size_bytes) || 0;
  return {
    databaseName:
      typeof row?.database_name === "string" && row.database_name ? row.database_name : "DuckDB Database",
    totalSize: formatBytes(bytes),
    totalSizeBytes: bytes,
    tableCount: Number(row?.table_count) || 0,
    indexCount: Number(row?.index_count) || 0,
  };
}

/** Fixed estimated cardinality (fix 5). */
function cardinality(ec: unknown): number | undefined {
  if (typeof ec === "number") return ec;
  if (typeof ec !== "string") return undefined;
  const s = ec.trim().replace(/^~\s*/, "").replace(/\s*rows?$/i, "").replace(/[_,]/g, "");
  return /^\d+$/.test(s) ? Number(s) : undefined;
}

let session: Session;
const select = async (sql: string, params?: unknown[]) => session.select(sql, params);
const rowsOf = async (sql: string, params?: unknown[]) => {
  const { columns, rows } = await select(sql, params);
  return { columns, rows: rows as unknown as Json };
};

/**
 * Runs the script's statements one by one in autocommit, as the table editor
 * does (splitDdlScript, then one execute per statement), on a fresh in-memory
 * database after `setup`; prints "ok" or the first error. DuckDB keeps a
 * dropped index's dependencies until the transaction ends, so a transaction
 * would hide fix 10.
 */
async function tryDdl(label: string, setup: string[], script: string): Promise<void> {
  const statements = script
    .split(/;\n/)
    .map((s) => s.replace(/;$/, "").trim())
    .filter((s) => s && !s.startsWith("--"));
  const scratchDb = await openSession(process.env.SEAQUEL_SERVER_URL!, { driver: "duckdb", path: ":memory:" });
  let result = "ok";
  try {
    for (const s of setup) await scratchDb.execute(s);
    for (const s of statements) {
      try {
        await scratchDb.execute(s);
      } catch (e) {
        result = `FAILED: ${s}\n    ${String(e).split("\n")[0]}`;
        break;
      }
    }
  } finally {
    await scratchDb.close();
  }
  console.log(`[ddl] ${label}: ${result}`);
}

    });
    const to7 = plain();
    to7.columns = to7.columns.filter((c) => c.name !== "name");
    const out7 = alterSql(plain(), to7);
    add({
      name: "fix 10: dropping a column that comes before a UNIQUE column is a note (customers.name, before email)",
      fix: 10,
      kind: "ddl-alter",
      input: { from: toJson(plain()), to: toJson(to7) },
      output: out7,
    });
    await tryDdl("alter: fix 10 drop before a UNIQUE column", tableSetup(plain()), out7);
    await tryDdl("alter: the same DROP COLUMN run as it is (expected to fail)", tableSetup(plain()), 'ALTER TABLE "main"."customers" DROP COLUMN "name";');
    const to8 = plain();
    to8.columns = to8.columns.filter((c) => c.name !== "status");
    const out8 = alterSql(plain(), to8);
    add({
      name: "fix 10: dropping a column after the last PRIMARY KEY or UNIQUE column runs (customers.status)",
      fix: 10,
      kind: "ddl-alter",
      input: { from: toJson(plain()), to: toJson(to8) },
      output: out8,
    });
    await tryDdl("alter: fix 10 drop after the last constrained column", tableSetup(plain()), out8);
  }

  // ── Fix 12: column types ──
  {
    const c = recorded("column-types.json", "getColumnTypes");
    const bare = ["ARRAY", "LIST", "MAP", "STRUCT", "UNION"];
    const out = (c.output as { name: string }[]).filter((t) => !bare.includes(t.name));
    add({
      name: "fix 12: no bare ARRAY, LIST, MAP, STRUCT or UNION (DuckDB needs their type arguments)",
      fix: 12,
      kind: "column-types",
      replaces: "column-types.json: getColumnTypes",
      input: null,
      output: out,
    });
    for (const t of out as { name: string; hasLength?: boolean; hasPrecision?: boolean }[]) {
      const type = t.hasPrecision ? `${t.name}(10,2)` : t.hasLength ? `${t.name}(20)` : t.name;
      await tryDdl(`column type ${type}`, [], `CREATE TABLE main.fx_type_probe (c ${type});`);
    }
    for (const t of bare) await tryDdl(`bare column type ${t} (expected to fail)`, [], `CREATE TABLE main.fx_type_probe (c ${t});`);
  }

  // ── Fixes 5, 13, 14: EXPLAIN ──
  for (const c of read("parse-explain.json").cases) {
    const compute = () => explainOutput(c.input, c.output);
    const out = toJson(compute());
    if (JSON.stringify(out) !== JSON.stringify(c.output)) {
      const fixes = fixesOf(EXPLAIN_FIXES, compute);
      add({ name: `${label(fixes)}: ${c.name}`, fix: fixes[0], kind: "parse-explain", replaces: `parse-explain.json: ${c.name}`, input: null, output: out });
    }
  }
  const fresh = (input: { rows: unknown; analyze: boolean }) =>
    explainOutput(input, adapter.parseExplainResult(decodeStoredRows(input.rows), input.analyze));
  for (const ec of ["~ 7", "~1_000 rows", "1 row", " 12 "]) {
    const c = recorded("parse-explain.json", "synthetic: estimated cardinality, plain digits");
    const input = structuredClone(c.input);
    const plan = JSON.parse(input.rows[0].explain_value);
    plan[0].extra_info["Estimated Cardinality"] = ec;
    input.rows[0].explain_value = JSON.stringify(plan, null, 4);
    add({ name: `fix 5: Estimated Cardinality ${JSON.stringify(ec)}`, fix: 5, kind: "parse-explain", input, output: fresh(input) });
  }
  {
    const dotted = `fx_sales."fx.dotted ""t"""`;
    await session.execute(`CREATE TABLE ${dotted} (id INTEGER)`);
    await session.execute(`INSERT INTO ${dotted} VALUES (1), (2)`);
    const sql = `SELECT * FROM ${dotted} WHERE id > 0`;
    const r = await rowsOf(adapter.getExplainQuery(sql, false));
    await session.execute(`DROP TABLE ${dotted}`);
    const input = { sql, analyze: false, columns: r.columns, rows: r.rows };
    add({
      name: 'fix 14: a table named with a dot and quotes keeps its whole last part (fx.dotted "t")',
      fix: 14,
      kind: "parse-explain",
      input,
      output: fresh(input),
    });
  }

  // A replacing case must change the recorded output, and replace it once.
  const names = new Set<string>();
  for (const c of cases) {
    if (names.has(c.name)) throw new Error(`duplicate case name ${c.name}`);
    names.add(c.name);
  }
  const replaced = new Set<string>();
  for (const c of cases.filter((x) => x.replaces)) {
    if (replaced.has(c.replaces!)) throw new Error(`${c.replaces} replaced twice`);
    replaced.add(c.replaces!);
  }
  for (const c of cases.filter((x) => x.replaces)) {
    const [file, ...rest] = c.replaces!.split(": ");
    const r = recorded(file, rest.join(": "));
    if (JSON.stringify(toJson(r.output)) === JSON.stringify(c.output)) {
      throw new Error(`${c.name}: same output as ${c.replaces}`);
    }
  }

  await session.close();
  rmSync(DIR, { recursive: true, force: true });

  const about = readFileSync(new URL("./about.txt", import.meta.url), "utf-8").trim().replace(/\s*\n\s*/g, " ");
  writeFileSync(
    `${FIXTURES}bugfixes.json`,
    JSON.stringify({ about, scratch, cases }, null, 2) + "\n",
  );
  console.log(`${cases.length} cases, ${cases.filter((c) => c.replaces).length} replaces`);
}, 120_000);
