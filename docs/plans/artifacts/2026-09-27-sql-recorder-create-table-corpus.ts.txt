// The CREATE TABLE corpus for create-table.json (`parseCreateTableSql`, the
// table editor's SQL pane): every ddl-create output in the engine crates'
// frozen fixtures (the DDL the pane shows first), including the bug-fix
// outputs that superseded some of them, the e2e schema files' CREATE TABLE
// statements, and hand-written cases for what users type.

export function CREATE_TABLE_CASES(): CreateTableInput[] {
  const out: CreateTableInput[] = [];
  for (const [engine, dir] of ENGINE_FIXTURES) {
    const ddl = JSON.parse(readFileSync(new URL(`${dir}ddl-create.json`, repo), "utf8")) as {
      cases: { name: string; output: string }[];
    };
    for (const c of ddl.cases)
      out.push({ name: `${engine}: ddl-create: ${c.name}`, sql: c.output });
    const bugfixesPath = new URL(`${dir}bugfixes.json`, repo);
    if (!existsSync(bugfixesPath)) continue;
    const bugfixes = JSON.parse(readFileSync(bugfixesPath, "utf8")) as {
      cases: { name: string; kind: string; output: unknown }[];
    };
    for (const c of bugfixes.cases) {
      if (c.kind === "ddl-create" && typeof c.output === "string") {
        out.push({ name: `${engine}: bugfixes: ${c.name}`, sql: c.output });
      }
    }
  }
  for (const [dir, engine] of SCHEMA_FILES) {
    const sql = readFileSync(new URL(`e2e/test-databases/${dir}/schema.sql`, repo), "utf8");
    splitModel(sql, engine).forEach((s) => {
      if (/\bCREATE\s+TABLE\b/i.test(s.sql))
        out.push({ name: `e2e ${dir}: statement ${s.index}`, sql: s.sql });
    });
  }
  for (const [name, sql] of HAND) out.push({ name: `hand: ${name}`, sql });
  return out;
}
