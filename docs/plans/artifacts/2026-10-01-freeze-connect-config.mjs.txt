/**
 * Runs what connection-tab-view.svelte does for a tab opened in "reconnect"
 * mode once its credentials have loaded. Returns whether it auto-connected
 * and the error it would show.
 */
export async function runReconnectTab(tab, db) {
  let formData;
  let isConnecting = false;
  let connectionError = null;
  let autoConnected = false;

  // lines 67-74: initialize local state from the tab prop
  formData = { ...tab.formData };
  connectionError = tab.error;

  // lines 79-92: when credentials are loaded, sync non-empty values
  if (tab.formData.password) {
    formData.password = tab.formData.password;
  }
  if (tab.formData.sshPassword) {
    formData.sshPassword = tab.formData.sshPassword;
  }
  if (tab.formData.sshKeyPassphrase) {
    formData.sshKeyPassphrase = tab.formData.sshKeyPassphrase;
  }

  // lines 124-150: handleAutoConnect (toasts, onboarding and tab removal left out)
  const handleAutoConnect = async () => {
    isConnecting = true;
    autoConnected = true;
    const tabConnectionId = tab.connectionId;
    try {
      const connectionData = getConnectionData(formData);
      if (tabConnectionId) {
        await db.connections.reconnect(tabConnectionId, connectionData);
      }
    } catch (error) {
      isConnecting = false;
      connectionError = error instanceof Error ? error.message : String(error);
    }
  };

  // lines 109-122: auto-connect when credentials are loaded in reconnect mode
  const hasAll = hasAllCredentials(formData);
  if (tab.mode === "reconnect" && tab.credentialsLoaded && !isConnecting && !connectionError) {
    if (hasAll) {
      await handleAutoConnect();
    }
  }
  return { hasAllCredentials: hasAll, autoConnected, error: connectionError };
}

async function bundle() {
  const stubDir = path.join(WORK, "stubs");
  fs.mkdirSync(stubDir, { recursive: true });
  const stubPaths = {};
  for (const [name, code] of Object.entries(STUBS)) {
    const file = path.join(stubDir, name.replace(/[^a-z0-9]+/gi, "_") + ".ts");
    fs.writeFileSync(file, code);
    stubPaths[name] = file;
  }
  const viewFile = path.join(WORK, "connection-tab-view.extract.ts");
  fs.writeFileSync(viewFile, VIEW_EXTRACT);
  const entry = path.join(WORK, "entry.ts");
  fs.writeFileSync(
    entry,
    [
      `export { ConnectionManager } from ${JSON.stringify(path.join(LIB, "hooks/database/connection-manager.svelte.ts"))};`,
      `export { ConnectionTabManager } from ${JSON.stringify(path.join(LIB, "hooks/database/connection-tabs.svelte.ts"))};`,
      `export { UnifiedTauriProvider } from ${JSON.stringify(path.join(LIB, "providers/unified-tauri-provider.ts"))};`,
      `export { runReconnectTab } from ${JSON.stringify(viewFile)};`,
      `export { defaultFormData } from ${JSON.stringify(path.join(LIB, "hooks/database/connection-tabs.svelte.ts"))};`,
      `export { getConnectionData, parseConnectionString } from ${JSON.stringify(path.join(LIB, "utils/connection-string.ts"))};`,
      "",
    ].join("\n"),
  );
  const outfile = path.join(WORK, "bundle.mjs");
  await esbuild.build({
    entryPoints: [entry],
    outfile,
    bundle: true,
    format: "esm",
    platform: "node",
    target: "node20",
    logLevel: "error",
    plugins: [
      {
        name: "freeze-connect-config",
        setup(build) {
          build.onResolve({ filter: /.*/ }, async (args) => {
            if (stubPaths[args.path]) return { path: stubPaths[args.path] };
            if (args.path === "$lib" || args.path.startsWith("$lib/")) {
              const rest = args.path.slice("$lib".length) || "/index";
              return build.resolve("." + rest, { resolveDir: LIB, kind: args.kind });
            }
            return undefined;
          });
        },
      },
    ],
  });
  return import(pathToFileURL(outfile).href);
}

/** Per-run recorder, reset for each path of each case. */
let run = null;

function newRun(c) {
  return {
    secrets: c.secrets,
    secretErrors: new Set(c.secretErrors ?? []),
    tunnelPort: c.tunnelPort,
    secretsRead: [],
    tunnel: undefined,
    tunnelsClosed: 0,
    configs: [],
    logs: [],
  };
}

globalThis.__FREEZE__ = {
  log(level, args) {
    if (run) run.logs.push({ level, message: args.map((a) => (typeof a === "string" ? a : String(a))).join(" ") });
  },
  async invoke(cmd, args) {
    if (cmd === "core_call") {
      const req = JSON.parse(new TextDecoder().decode(args));
      if (req.method === "secret" && req.params.method === "get") {
        const key = req.params.params.key;
        run.secretsRead.push(key);
        if (run.secretErrors.has(key)) {
          throw { code: "SECRET_ERROR", message: "keychain read denied" };
        }
        const value = Object.prototype.hasOwnProperty.call(run.secrets, key) ? run.secrets[key] : null;
        return { method: "secret", result: { method: "get", result: value } };
      }
      if (req.method === "ssh" && req.params.method === "open") {
        if (run.tunnel !== undefined) throw new Error("second tunnel in one run");
        run.tunnel = JSON.parse(JSON.stringify(req.params.params.config));
        return {
          method: "ssh",
          result: { method: "open", result: { tunnelId: "tunnel-1", localPort: run.tunnelPort } },
        };
      }
      if (req.method === "ssh" && req.params.method === "close") {
        run.tunnelsClosed += 1;
        return { method: "ssh", result: { method: "close", result: null } };
      }
      throw new Error(`unexpected core_call ${req.method}/${req.params?.method}`);
    }
    if (cmd === "db_connect" || cmd === "db_test") {
      // What crosses the IPC boundary: JSON, so `undefined` fields are gone.
      // `db_test` (the form's Test button) takes the same config.
      const config = JSON.parse(JSON.stringify(args.config));
      run.configs.push(config);
      // Core's one check that needs no server: the sqlx drivers refuse a config
      // without a connection string (seaquel-engine-{postgres,mysql,sqlite}
      // driver.rs, `DbError::connection_error`). Every other config "connects".
      const label = { postgres: "PostgreSQL", mysql: "MySQL", sqlite: "SQLite" }[config.driver];
      if (label && config.connection_string === undefined) {
        throw {
          code: "CONNECTION_ERROR",
          message: `Failed to connect: connection_string is required for ${label}`,
        };
      }
      return { connection_id: `provider-${run.configs.length}` };
    }
    if (cmd === "db_disconnect") return null;
    throw new Error(`unexpected invoke ${cmd}`);
  },
};

function newManager(mod, row, state) {
  const persistence = {
    loadPersistedConnections: async () => [structuredClone(row)],
    persistConnection: async () => {},
    scheduleProject: () => {},
  };
  const stateRestoration = {
    initializeConnectionMaps() {},
    ensureConnectionMapsExist() {},
    cleanupConnectionMaps() {},
    loadConnectionData: async () => {},
  };
  const providers = { getForType: async () => new mod.UnifiedTauriProvider() };
  const manager = new mod.ConnectionManager(
    state,
    persistence,
    stateRestoration,
    {},
    providers,
    async () => {},
    () => {},
  );
  // Record what reconnect throws; autoReconnect swallows it.
  const reconnect = manager.reconnect.bind(manager);
  manager.reconnect = async (...args) => {
    try {
      return await reconnect(...args);
    } catch (error) {
      run.reconnectError = error instanceof Error ? error.message : String(error);
      throw error;
    }
  };
  return manager;
}

/** Keep the run's result: which secrets were read, the tunnel, the config. */
function outcome(r, extra) {
  const o = { secretsRead: r.secretsRead, ...extra };
  if (r.tunnel !== undefined) o.tunnel = r.tunnel;
  if (r.configs.length > 1) throw new Error("more than one db_connect in one run");
  if (r.configs.length === 1) o.config = r.configs[0];
  if (r.reconnectError !== undefined) o.error = r.reconnectError;
  if (r.tunnelsClosed > 0) o.tunnelClosed = true;
  return o;
}

async function runCase(mod, c) {
  const result = {};

  // 1. App start: initializePersistedConnections builds the in-memory
  //    connection (username fallback, db password prefetch).
  // 2. The user clicks the connection: autoReconnect.
  {
    run = newRun(c);
    const state = freshState();
    const manager = newManager(mod, c.row, state);
    await manager.initializePersistedConnections();
    const loaded = state.connections.find((x) => x.id === c.row.id);
    result.load = {
      secretsRead: run.secretsRead,
      username: loaded.username,
      passwordFromKeychain: loaded.password !== "",
    };
    run.secretsRead = [];
    run.logs = [];
    const ok = await manager.autoReconnect(c.row.id);
    const skipped = run.logs
      .map((l) => /Auto-reconnect skipped \((.*)\)/.exec(l.message)?.[1])
      .find((x) => x !== undefined);
    result.autoReconnect = outcome(run, { connected: ok, ...(skipped ? { skipped } : {}) });
    run = null;
  }

  // 3. When autoReconnect returns false the app opens the reconnect tab with
  //    the connection as prefill (the same object in every caller:
  //    sidebar/manage/connections.svelte, empty-states/connection-card.svelte,
  //    command-palette.svelte, connection-selector.svelte,
  //    starter-tabs/getting-started-content.svelte). Recorded for every case,
  //    from a fresh start, whatever autoReconnect did.
  {
    run = newRun(c);
    const state = freshState();
    const manager = newManager(mod, c.row, state);
    await manager.initializePersistedConnections();
    const connection = state.connections.find((x) => x.id === c.row.id);
    run.secretsRead = [];
    const tabs = new mod.ConnectionTabManager(
      state,
      { add() {}, removeTabGeneric() {} },
      () => {},
      () => {},
    );
    const tabId = await tabs.open({
      id: connection.id,
      name: connection.name,
      type: connection.type,
      host: connection.host,
      port: connection.port,
      databaseName: connection.databaseName,
      username: connection.username,
      sslMode: connection.sslMode,
      connectionString: connection.connectionString,
      sshTunnel: connection.sshTunnel,
      savePassword: connection.savePassword,
      saveSshPassword: connection.saveSshPassword,
      saveSshKeyPassphrase: connection.saveSshKeyPassphrase,
    });
    const tab = state.connectionTabsByProject["default-seaquel"].find((t) => t.id === tabId);
    const secretsRead = run.secretsRead;
    const view = await mod.runReconnectTab(tab, { connections: manager });
    const o = outcome(run, {
      hasAllCredentials: view.hasAllCredentials,
      autoConnected: view.autoConnected,
    });
    o.secretsRead = secretsRead;
    result.reconnectTab = o;
    run = null;
  }

  // Which path ends up connecting in the app: autoReconnect first, the tab's
  // auto-connect when that returns false, else the form waits for the user.
  result.gui = result.autoReconnect.connected
    ? "autoReconnect"
    : result.reconnectTab.autoConnected && result.reconnectTab.error === undefined
      ? "reconnectTab"
      : "form";
  return result;
}

/**
 * One form case: the form the wizard tab holds when the user clicks Connect
 * (`add`) or Test (`test`), sent the way `connection-tab-view.svelte` sends
 * it. Nothing is read from the keychain: the form carries the secrets.
 */
async function runFormCase(mod, c) {
  run = newRun({ secrets: {}, tunnelPort: c.tunnelPort });
  const state = freshState();
  state.connectionsLoading = false;
  const manager = newManager(mod, null, state);
  let formData = { ...mod.defaultFormData, ...(c.fields ?? {}) };
  if (c.paste !== undefined) {
    // wizard-step-method binds the paste box to formData.connectionString,
    // then handleParse merges the parsed fields.
    formData.connectionString = c.paste;
    const parsed = mod.parseConnectionString(c.paste.trim());
    if (!parsed.success) throw new Error(`${c.name}: paste doesn't parse: ${parsed.error}`);
    Object.assign(formData, parsed.formData);
  }
  // What the user typed on the details step, after the paste.
  formData = { ...formData, ...(c.typed ?? {}) };
  const sent = mod.getConnectionData(formData);
  let error;
  try {
    if (c.op === "add") {
      await manager.add({ ...sent, createIfMissing: c.createIfMissing });
    } else {
      await manager.test(sent);
    }
  } catch (e) {
    error = e instanceof Error ? e.message : typeof e === "object" && e?.message ? `${e.code}: ${e.message}` : String(e);
  }
  const o = {};
  if (run.secretsRead.length) throw new Error(`${c.name}: the form path read the keychain`);
  o.connectionString = sent.connectionString;
  if (run.tunnel !== undefined) o.tunnel = run.tunnel;
  if (run.configs.length > 1) throw new Error("more than one connect in one run");
  if (run.configs.length === 1) o.config = run.configs[0];
  if (error !== undefined) o.error = error;
  if (run.tunnelsClosed > 0) o.tunnelClosed = true;
  run = null;
  return { formData, ...o };
}

/**
 * One case. `row` is a `PersistedConnection` as `connectionsRepo.loadAll`
 * returns it (flags always booleans; optional fields absent when unset).
 * `secrets` uses the short names db/ssh/ssh-key and is stored under the
 * full keychain keys. `tunnelPort` is the local port the tunnel reports.
 */
function kase(name, row, { secrets = {}, secretErrors = [], tunnelPort, notes } = {}) {
  const id = "conn-" + name.replace(/[^a-z0-9]+/gi, "-").replace(/-+$/, "");
  const fullRow = { id, ...BASE, ...row };
  const fullSecrets = {};
  for (const [k, v] of Object.entries(secrets)) fullSecrets[`${k}:${id}`] = v;
  const c = {
    name,
    ...(notes ? { notes } : {}),
    row: fullRow,
    secrets: fullSecrets,
    ...(secretErrors.length ? { secretErrors: secretErrors.map((k) => `${k}:${id}`) } : {}),
    tunnelPort: tunnelPort ?? (row.sshTunnel?.enabled ? 50022 : null),
  };
  return c;
}

/**
 * One form case. `fields` is what the user entered on the details step
 * (merged over `defaultFormData`; picking a type sets its default port, as
 * `selectDatabaseType` does, so the helpers carry it). `paste` is a string
 * pasted on the method step and parsed; `typed` is what was entered after
 * the paste. `createIfMissing` is `add` only, as in the component.
 */
function fcase(name, op, { fields, paste, typed, createIfMissing = false, tunnelPort, notes } = {}) {
  if (!name.startsWith(op + "/")) throw new Error(`${name}: id must start with ${op}/`);
  const sshOn = (typed?.sshEnabled ?? fields?.sshEnabled) || (paste ?? "").split(":")[0].endsWith("+ssh");
  return {
    name,
    ...(notes ? { notes } : {}),
    op,
    ...(fields ? { fields } : {}),
    ...(paste !== undefined ? { paste } : {}),
    ...(typed ? { typed } : {}),
    ...(op === "add" ? { createIfMissing } : {}),
    tunnelPort: tunnelPort ?? (sshOn ? 50122 : null),
  };
}
