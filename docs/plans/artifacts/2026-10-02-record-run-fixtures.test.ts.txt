 * points the output at another directory (to check that two runs agree).
 *
 * What runs unmodified:
 * - `QueryExecutionManager` (`query-execution.svelte.ts`), with Task 1's fix;
 * - `createExecution` (`components/query-editor/execution.svelte.ts`), the
 *   editor's Run and Run current buttons: the destructive prompt (confirmed by

 *   the calls into the manager, exactly as the editor makes them;
 * - `resolve-query.ts`, every `$lib/sql` function (the real seaquel-wasm);
 * - `CoreProvider` (`providers/core-provider.ts`), the desktop and web
 *   provider, over a scripted `CoreClient`. So the driver log is the `db`
 *   calls today's GUI sends Core: `query`, `execute` and `queryStream`.
 *
 * What is stubbed:
 * - the `CoreClient`: it answers each call from the case's `driver` list, in
 *   order, and fails the case on a call it doesn't expect or an answer left over;
 * - `getEngineClient(…).paginate`, which returns its arguments in a marker the
 *   client reads back. The fixtures record `paginate` as `{limit, offset}`, so
 *   the Rust replay expands them with the connection's `Dialect`;

 * - `$lib/sql`'s `countQuery`, `extractTableFromSelect`,

 *   `splitSqlStatementsOrThrow` are the real functions, wrapped only to see
 *   what they returned (the count SQL, the table and column references the

 * - `QueryExecutionManager#filterAndIndexResults`, wrapped to see the results
 *   before utility results are hidden.
 *
 * The fixtures README says what each field means.

type Wire = unknown;
type Answer =
  | { op: "page" | "utility" | "count"; columns?: string[]; rows?: Wire[][]; error?: RpcErr }
  | { op: "write"; rowsAffected?: number; lastInsertId?: number; error?: RpcErr }
  | {
      op: "stream";
      columns?: string[];
      rows?: Wire[][];
      batches?: { columns?: string[]; rows: Wire[][] }[];
      error?: RpcErr;
    };
type RpcErr = { code: string; message: string };
type Target = { type: "all" } | { type: "current"; cursor: number };
type Action =
  | { type: "goToPage"; page: number; resultIndex?: number }
  | { type: "setPageSize"; pageSize: number; resultIndex?: number };

interface Case {
  file: string;
  name: string;
  engine: DatabaseType;
  notes?: string;
  text: string;
  target: Target;
  /** The values typed into the parameter dialog, in the wire format. */
  dialog?: { name: string; value: Wire }[];
  pageSize?: number;
  deferWrites?: boolean;
  /** `rerun`: `db.queries.execute(tabId)` as the file drop and the reruns call it. */
  via?: "editor" | "rerun";
  driver: Answer[];
  pages?: { action: Action; driver: Answer[] }[];
}

const all: Target = { type: "all" };
/** The UTF-16 offset of `needle` in `text`, plus `delta`. */
function at(text: string, needle: string, delta = 0, from = 0): Target {
  const i = text.indexOf(needle, from);
  if (i < 0) throw new Error(`no ${needle} in ${text}`);
  return { type: "current", cursor: i + delta };
}
const cur = (cursor: number): Target => ({ type: "current", cursor });
const rows = (n: number, from = 1): Wire[][] =>
  Array.from({ length: n }, (_, i) => [from + i]);
const page = (columns: string[], r: Wire[][]): Answer => ({ op: "page", columns, rows: r });
const count = (total: Wire): Answer => ({ op: "count", columns: ["total"], rows: [[total]] });
const util = (): Answer => ({ op: "utility", columns: [], rows: [] });
const write = (rowsAffected: number, lastInsertId?: number): Answer =>
  lastInsertId === undefined
    ? { op: "write", rowsAffected }
    : { op: "write", rowsAffected, lastInsertId };
const stream = (columns: string[], r: Wire[][]): Answer => ({ op: "stream", columns, rows: r });
const fail = (op: Answer["op"], message: string, code = "QUERY_ERROR"): Answer =>
  ({ op, error: { code, message } }) as Answer;
const bytes = { $sq: "bytes", v: "AQ==" };

const cases: Case[] = [];
function add(c: Case) {
  cases.push(c);
}

// ---------------------------------------------------------------- the client

interface DriverOp {
  op: string;
  sql: string;
  paginate?: { limit: number; offset: number };
  params: Wire[];
  answer: Omit<Answer, "op">;
}

const PAGE_RE = new RegExp(`^${PAGE}(\\d+):(\\d+)${PAGE}([\\s\\S]*)$`);

class ScriptedClient {
  log: DriverOp[] = [];
  failures: string[] = [];
  constructor(private answers: Answer[]) {}

  private next(op: string, sqlText: string, params: Wire[], paginate?: DriverOp["paginate"]) {
    const a = this.answers.shift();
    if (!a || a.op !== op) {
      const msg = `unexpected ${op} (${sqlText}); expected ${a ? a.op : "no call"}`;
      this.failures.push(msg);
      throw new CoreCallError({ code: "RECORDER", message: msg });
    }
    const { op: _op, ...answer } = a;
    this.log.push({
      op,
      sql: sqlText,
      ...(paginate ? { paginate } : {}),
      params: structuredClone(params),
      answer: structuredClone(answer),
    });
    return a;
  }

  left(): Answer[] {
    return this.answers;
  }

  async call(request: { method: string; params: { method: string; params: Record<string, unknown> } }) {
    const { method, params } = request.params;
    const text = params.sql as string;
    const binds = params.params as Wire[];
    if (method === "query") {
      const m = PAGE_RE.exec(text);
      const a = m
        ? this.next("page", m[3], binds, { limit: Number(m[1]), offset: Number(m[2]) })
        : this.next(rec.counts.has(text) ? "count" : "utility", text, binds);
      if (a.error) throw new CoreCallError(a.error);
      const ans = a as { columns?: string[]; rows?: Wire[][] };
      return {
        method: "db",
        result: {
          method: "query",
          result: { columns: ans.columns ?? [], rows: structuredClone(ans.rows ?? []) },
        },
      };
    }
    if (method === "execute") {
      const a = this.next("write", text, binds) as {
        rowsAffected?: number;
        lastInsertId?: number;
        error?: RpcErr;
      };
      if (a.error) throw new CoreCallError(a.error);
      return {
        method: "db",
        result: {
          method: "execute",
          result: { rows_affected: a.rowsAffected ?? 0, last_insert_id: a.lastInsertId ?? null },
        },
      };
    }
    this.failures.push(`unexpected db.${method}`);
    throw new CoreCallError({ code: "RECORDER", message: `unexpected db.${method}` });
  }

  async *stream(request: { params: { params: { sql: string; params: Wire[] } } }) {
    const p = request.params.params;
    let a: Answer;
    try {
      a = this.next("stream", p.sql, p.params);
    } catch (e) {
      yield { type: "error", code: "RECORDER", message: String(e) };
      return;
    }
    const s = a as Extract<Answer, { op: "stream" }>;
    const batches = s.batches ?? (s.rows || s.columns ? [{ columns: s.columns, rows: s.rows ?? [] }] : []);
    for (let i = 0; i < batches.length; i++) {
      const b = batches[i];
      yield {
        type: "batch",
        columns: b.columns ?? null,
        rows: structuredClone(b.rows),
        is_final: !s.error && i === batches.length - 1,
      };
    }
    if (s.error) yield { type: "error", ...s.error };
    else yield { type: "done" };
  }

  events() {
    return () => {};
  }
}

const wire = (v: unknown): Wire => {
  if (v instanceof Date) return v.toISOString();
  return encodeParam(v);
};
const strip = (s: string) => s.replace(/;$/, "").trim();
const PRIOR = "\u0001prior\u0001";

function recordCase(c: Case) {
  rec.toasts.length = 0;
  rec.counts.clear();
  rec.tables.clear();
  rec.refs.clear();

  const client = new ScriptedClient(structuredClone(c.driver));
  const provider = new CoreProvider(() => client as unknown as CoreClient);
  const connection = { id: "conn-1", type: c.engine, name: "Local", providerConnectionId: "pc-1" };
  const pageSize = c.pageSize ?? 100;
  // The editor never passes a page size: the runner takes the tab's last
  // SELECT result's, else 100. A case with another page size starts from a
  // tab whose last result had it.
  const prior: StatementResult[] | undefined =
    pageSize === 100
      ? undefined
      : [
          {
            columns: [],
            rows: [],
            rowCount: 0,
            totalRows: 0,
            executionTime: 0,
            queryType: "select",
            page: 1,
            pageSize,
            totalPages: 1,
            statementIndex: 0,
            statementSql: PRIOR,
            isError: false,
          },
        ];
  const tab = { id: "tab-1", query: c.text, results: prior } as {
    id: string;
    query: string;
    results?: StatementResult[];
    activeResultIndex?: number;
  };
  const state = {
    activeProjectId: "p",
    queryTabsByProject: { p: [tab] },
    activeConnection: connection,
    activeConnectionId: "conn-1",
    connections: [connection],
    schemas: {},
  } as unknown as DatabaseState;
  const providers = {
    getForType: async () => provider,
  } as unknown as ProviderRegistry;
  const historyCalls: { query: string; rowCount: number | undefined }[] = [];
  const history = {
    addToHistory: (query: string, r: StatementResult) => {
      // `QueryHistoryManager.addToHistory`'s row count.
      historyCalls.push({ query, rowCount: r.affectedRows ?? r.totalRows });
    },
  } as unknown as QueryHistoryManager;
  const pendingCalls: { sql: string; queryType: string; params: Wire[]; origin: string; tabId: string }[] = [];
  const pending = {
    isEnabled: () => !!c.deferWrites,
    add: (
      _conn: string,
      s: string,
      queryType: string,
      origin: string,
      tabId: string,
      binds?: unknown[],
    ) => {
      pendingCalls.push({ sql: s, queryType, params: (binds ?? []).map(wire), origin, tabId });
    },
    openSheet: () => {},
  } as unknown as PendingChangesManager;
  const manager = new QueryExecutionManager(state, history, providers, pending);

  let unfiltered: unknown[] | null = null;
  const m = manager as unknown as { filterAndIndexResults: (r: StatementResult[]) => StatementResult[] };
  const filter = m.filterAndIndexResults.bind(manager);
  m.filterAndIndexResults = (list) => {
    unfiltered = list.map((r) => snapshot(r));
    return filter(list);
  };

  return {
    client,
    manager,
    state,
    tab,
    pageSize,
    historyCalls,
    pendingCalls,
    getUnfiltered: () => unfiltered,
  };
}

function snapshot(r: StatementResult) {
  return {
    ...r,
    columns: [...r.columns],
    rows: r.rows.map((row) => [...row]),
    pageSource: r.pageSource
      ? { sql: r.pageSource.sql, params: [...r.pageSource.params] }
      : undefined,
    hasStreamSeed: "isStreaming" in r,
  };
}

type Snap = ReturnType<typeof snapshot>;

function errorText(e: RpcErr) {
  return `${e.code}: ${e.message}`;
}

/** Which driver calls each result came from, in order. */
function attribute(results: Snap[], ops: DriverOp[], failures: string[]) {
  let k = 0;
  const out: { kind: string | null; ops: DriverOp[] }[] = [];
  for (const r of results) {
    const mine: DriverOp[] = [];
    let kind: string | null = null;
    const op = ops[k];
    const err = op?.answer && (op.answer as { error?: RpcErr }).error;
    if (!r.isError || r.pageSource) {
      if (!op) {
        failures.push(`no driver call for result ${r.statementIndex}`);
      } else {
        kind = op.op;
        mine.push(op);
        k++;
        if (kind === "page" && ops[k]?.op === "count") {
          mine.push(ops[k]);
          k++;
        }
      }
    } else if (op && err && r.error === errorText(err)) {
      kind = op.op;
      mine.push(op);
      k++;
    }
    out.push({ kind, ops: mine });
  }
  return { attributed: out, used: k };
}

function shapeResult(r: Snap, kind: string | null, ops: DriverOp[], shown: boolean) {
  const base = r.pageSource ? strip(r.pageSource.sql) : null;
  const out: Record<string, unknown> = {
    index: r.statementIndex,
    shown,
    sql: r.statementSql,
    source: r.pageSource ? { sql: r.pageSource.sql, params: r.pageSource.params.map(wire) } : null,
    kind,
    queryType: r.queryType ?? null,
    columns: r.columns,
    rows: r.rows.map((row) => row.map(wire)),
    rowCount: r.rowCount,
    totalRows: r.totalRows,
    page: r.page,
    pageSize: r.pageSize,
    totalPages: r.totalPages,
    error: r.error ?? null,
    affectedRows: r.affectedRows ?? null,
    lastInsertId: r.lastInsertId ?? null,
  };
  if (kind === "page" && !r.error && ops[0]?.paginate) {
    const fetched = ((ops[0].answer as { rows?: unknown[] }).rows ?? []).length;
    const full = fetched > ops[0].paginate.limit - 1;
    const countOp = ops.find((o) => o.op === "count");
    out.countEstimated = full && (!countOp || !!(countOp.answer as { error?: unknown }).error);
  }
  if (base !== null && rec.tables.has(base)) out.table = rec.tables.get(base) ?? null;
  if (base !== null && rec.refs.has(base)) out.columnRefs = rec.refs.get(base) ?? null;
  return out;
}

const recorded = new Map<string, Record<string, unknown>[]>();

async function run(c: Case): Promise<Record<string, unknown>> {
  const { client, manager, state, tab, pageSize, historyCalls, pendingCalls, getUnfiltered } =
    recordCase(c);

  // The editor, as query-editor.svelte wires it.
  const inFlight: Promise<unknown>[] = [];
  const track =
    <A extends unknown[]>(fn: (...a: A) => Promise<unknown>) =>
    (...a: A) => {
      const p = fn(...a);
      inFlight.push(p);
      return p;
    };
  const queries = {
    execute: track((...a: Parameters<typeof manager.execute>) => manager.execute(...a)),
    executeCurrent: track((...a: Parameters<typeof manager.executeCurrent>) =>
      manager.executeCurrent(...a),
    ),
    executeWithParams: track((...a: Parameters<typeof manager.executeWithParams>) =>
      manager.executeWithParams(...a),
    ),
    executeCurrentWithParams: track(
      (...a: Parameters<typeof manager.executeCurrentWithParams>) =>
        manager.executeCurrentWithParams(...a),
    ),
  };
  const paramDialog = {
    params: [] as unknown[],
    action: null as unknown,
    show: false,
    getParameterDefinitions: () => [],
  };
  const cursor = c.target.type === "current" ? c.target.cursor : 0;
  const ctx = {
    db: { state, queries },
    getMonacoRef: () => ({ getCursorOffset: () => cursor }),
    getActiveTab: () => tab,
    getActiveTabId: () => tab.id,
    getActiveResult: () => null,
    getActiveResultIndex: () => 0,
    getResultKey: () => null,
  };

  let destructive: { index: number; sql: string; reason: unknown }[] = [];
  let sentParams: { name: string; value: Wire }[] | null = null;
  const via = c.via ?? "editor";
  if (via === "editor") {
    const exec = createExecution(
      ctx as never,
      paramDialog as never,
      () => {},
    );
    if (c.target.type === "all") exec.handleExecute();
    else exec.handleExecuteCurrent();
    if (exec.showDestructiveConfirm) {
      destructive = exec.destructiveStatements.map((d) => ({
        index: d.index,
        sql: d.sql,
        reason: JSON.parse(JSON.stringify(d.reason)),
      }));
      exec.handleDestructiveConfirm();
    }
    if (paramDialog.show) {
      if (!c.dialog) throw new Error(`${c.name}: the parameter dialog opened with no values`);
      paramDialog.show = false;
      sentParams = c.dialog;
      const values: ParameterValue[] = c.dialog.map((p) => ({
        name: p.name,
        value: decodeCell(structuredClone(p.value)),
      }));
      exec.handleParamExecute(values);
    } else if (c.dialog) {
      throw new Error(`${c.name}: dialog values given but the dialog didn't open`);
    }
  } else {
    // The file drop and the reruns: `db.queries.execute(tabId)`, no prompt.
    if (c.target.type !== "all") throw new Error("rerun runs the whole tab");
    const split = sql.splitSqlStatementsOrThrow(c.text, c.engine);
    destructive = sql.findDestructiveStatements(split, c.engine).map((d) => ({ ...d }));
    inFlight.push(manager.execute(tab.id));
  }
  await Promise.all(inFlight);

  // What ran.
  const statements =
    c.target.type === "all"
      ? sql.splitSqlStatementsOrThrow(c.text, c.engine).map((s) => ({ index: s.index, sql: s.sql }))
      : (() => {
          const s = sql.getStatementAtOffsetOrThrow(c.text, c.target.cursor, c.engine);
          return s ? [{ index: s.index, sql: s.sql }] : [];
        })();

  const current = (tab.results ?? []).filter((r) => r.statementSql !== PRIOR);
  const unfiltered = (getUnfiltered() as Snap[] | null) ?? current.map(snapshot);
  const shownCount = current.length;
  const shownSet = new Set<Snap>();
  if (c.target.type === "all" && getUnfiltered()) {
    const anyShown = unfiltered.some((r) => !r.isUtility);
    for (const r of unfiltered) if (!anyShown || !r.isUtility) shownSet.add(r);
  } else {
    for (const r of unfiltered) shownSet.add(r);
  }
  expect(shownSet.size).toBe(shownCount);

  const failures = [...client.failures];
  const ops = client.log.splice(0);
  const { attributed, used } = attribute(unfiltered, ops, failures);
  if (used !== ops.length) failures.push(`${ops.length - used} driver calls not attributed`);
  const results = unfiltered.map((r, i) =>
    shapeResult(r, attributed[i].kind, attributed[i].ops, shownSet.has(r)),
  );

  // Pending changes carry no index: they are the statements with no result.
  // In run all a result's `statementIndex` is its statement's position, so
  // the deferred ones are the positions with no result, in order.
  const withResult = new Set(unfiltered.map((r) => r.statementIndex));
  const deferredAt =
    c.target.type === "all" ? statements.map((_, i) => i).filter((i) => !withResult.has(i)) : [0];
  if (deferredAt.length !== pendingCalls.length && pendingCalls.length > 0) {
    throw new Error(`${c.name}: ${pendingCalls.length} deferred, ${deferredAt.length} without a result`);
  }
  const deferred = pendingCalls.map((p, i) => ({
    index: deferredAt[i],
    sql: statements[deferredAt[i]]?.sql ?? null,
    source: { sql: p.sql, params: p.params },
    queryType: p.queryType,
  }));

  const historyOut = historyCalls.length === 0 ? null : historyCalls.map((h) => ({ ...h }));
  const pagesOut: Record<string, unknown>[] = [];
  const displayed = () => (tab.results ?? []).filter((r) => r.statementSql !== PRIOR);

  for (const p of c.pages ?? []) {
    (client as unknown as { answers: Answer[] }).answers.push(...structuredClone(p.driver));
    const before = historyCalls.length;
    const idx = p.action.resultIndex ?? tab.activeResultIndex ?? 0;
    const shownBefore = results.filter((r) => r.shown);
    if (p.action.type === "goToPage") await manager.goToPage(tab.id, p.action.page, p.action.resultIndex);
    else await manager.setPageSize(tab.id, p.action.pageSize, p.action.resultIndex);
    const pops = client.log.splice(0);
    failures.push(...client.failures.splice(0));
    const after = snapshot(displayed()[idx]);
    const kind = pops[0]?.op ?? null;
    const shaped = shapeResult(after, kind, pops, true);
    shaped.index = shownBefore[idx]?.index ?? idx;
    pagesOut.push({
      action: p.action,
      driver: pops,
      result: shaped,
      history: historyCalls.length > before ? historyCalls.slice(before) : null,
    });
  }

  if (client.left().length) failures.push(`${client.left().length} answers left over`);
  expect(failures, c.name).toEqual([]);

  const out: Record<string, unknown> = {
    name: c.name,
    engine: c.engine,
    ...(c.notes ? { notes: c.notes } : {}),
    input: {
      text: c.text,
      target: c.target,
      params: sentParams,
      pageSize,
      deferWrites: !!c.deferWrites,
      confirmed: destructive.length > 0,
      via,
    },
    statements,
    destructive,
    driver: ops,
    results,
    deferred,
    history: historyOut ? historyOut[0] : null,
    toasts: [...rec.toasts],
  };
  if (historyOut && historyOut.length > 1) out.historyCalls = historyOut;
  if (pagesOut.length) out.pages = pagesOut;
  return out;
}

describe("record run fixtures", () => {
  for (const c of cases) {
    it(c.name, async () => {
      const out = await run(c);
      const list = recorded.get(c.file) ?? [];
      list.push(out);
      recorded.set(c.file, list);
    });
  }

  it("has unique names", () => {
    expect(new Set(cases.map((c) => c.name)).size).toBe(cases.length);
  });

  afterAll(() => {
    if (!process.env.FREEZE_RUN) return;
    if ([...recorded.values()].flat().length !== cases.length) {
      throw new Error("not every case recorded; nothing written");
    }
    const dir =
      process.env.FREEZE_RUN_OUT ?? join(process.cwd(), "crates/seaquel-workspace/tests/fixtures/run");
    mkdirSync(dir, { recursive: true });
    const files = [
      "plan-postgres",
      "plan-mysql",
      "plan-mariadb",
      "plan-sqlite",
      "plan-mssql",
      "plan-duckdb",
      "cursor",
      "execute",
      "history",
      "pending",
    ];
    for (const f of files) {
      writeFileSync(join(dir, `${f}.json`), JSON.stringify(recorded.get(f) ?? [], null, 2) + "\n");
    }
  });
});
