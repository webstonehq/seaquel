/** JS → the cell wire format (`undefined` is a missing value: NULL). */
function wire(v: unknown): Wire {
  if (v === undefined) return null;
  return JSON.parse(JSON.stringify(encodeParam(v)));
}
function wireRecord(r: Record<string, unknown> | undefined): Record<string, Wire> | undefined {
  if (!r) return undefined;
  return Object.fromEntries(Object.entries(r).map(([k, v]) => [k, wire(v)]));
}
/** The wire format → JS, as the providers decode rows. */
const js = (v: Wire): unknown => decodeCell(structuredClone(v));

function col(
  name: string,
  type: string,
  o: { pk?: boolean; castType?: string; default?: string; nullable?: boolean } = {},
): SchemaColumn {
  return {
    name,
    type,
    ...(o.castType === undefined ? {} : { castType: o.castType }),
    nullable: o.nullable ?? !o.pk,
    ...(o.default === undefined ? {} : { defaultValue: o.default }),
    isPrimaryKey: !!o.pk,
    isForeignKey: false,
  };
}
function tbl(schema: string, name: string, columns: SchemaColumn[], indexes: SchemaIndex[] = []): SchemaTable {
  return { name, schema, type: "table", columns, indexes } as unknown as SchemaTable;
}
function metaOf(t: SchemaTable): Meta {
  return { schema: t.schema, table: t.name, columns: t.columns, indexes: t.indexes ?? [] };
}
const pkIndex = (table: string, ...columns: string[]): SchemaIndex => ({
  name: `${table}_pkey`,
  columns,
  unique: true,
  type: "btree",
});

const HELPER = process.env.SEAQUEL_DIALECT_HELPER;
/** The real Rust dialect's answer to one engine RPC builder call. */
function dialectBuild(engine: string, request: unknown): { sql: string; bindValues?: Wire[] } {
  if (!HELPER) throw new Error("set SEAQUEL_DIALECT_HELPER to the dialect helper binary");
  const r = spawnSync(HELPER, [], { input: JSON.stringify({ engine, request }) + "\n", encoding: "utf8" });
  if (r.status !== 0) throw new Error(`dialect helper failed: ${r.stderr}`);
  const out = JSON.parse(r.stdout) as { sql: string; bindValues?: Wire[]; error?: string };
  if (out.error) throw new Error(`dialect helper: ${out.error}`);
  return { sql: out.sql, ...(out.bindValues ? { bindValues: out.bindValues } : {}) };
}

// ---------------------------------------------------------------- the client

type Answer =
  | { op: "execute"; rowsAffected?: number; lastInsertId?: number; error?: RpcErr }
  | { op: "tableMetadata"; error: RpcErr };

interface DriverOp {
  op: string;
  [k: string]: unknown;
}

const BUILDERS = new Set(["buildUpdate", "buildSetDefault", "buildInsert", "buildDelete"]);

/** How `db.query` calls are answered: a data tab's page and count. */
interface PageSource {
  columns: string[];
  /** Every row the filters match, in order; a page is a slice. */
  matching: Wire[][];
  page: number;
  pageSize: number;
  countCell?: Wire;
  countError?: RpcErr;
  pageError?: RpcErr;
}

class ScriptedClient {
  log: DriverOp[] = [];
  failures: string[] = [];
  /** Data tab refreshes answered without logging (plan and apply cases). */
  quietQueries = 0;
  quiet = false;
  constructor(
    private engine: string,
    private answers: Answer[],
    private metadata: Meta[],
    public pageSource: PageSource | null,
  ) {}

  left(): Answer[] {
    return this.answers;
  }

  private fail(msg: string): never {
    this.failures.push(msg);
    throw new CoreCallError({ code: "RECORDER", message: msg });
  }

  private next<O extends Answer["op"]>(op: O, what: string): Extract<Answer, { op: O }> {
    const a = this.answers[0];
    if (!a || a.op !== op) this.fail(`unexpected ${op} (${what}); expected ${a ? a.op : "no call"}`);
    this.answers.shift();
    return a as Extract<Answer, { op: O }>;
  }

  async call(request: { method: string; params: { method: string; params: Record<string, unknown> } }) {
    const { method, params } = request.params;
    const res = (result: unknown) => ({ method: "db", result: { method, result } });
    if (method === "engine") {
      const req = params.request as { method: string; params: Record<string, unknown> };
      if (req.method === "tableMetadata") {
        const { schema, table } = req.params as { schema: string; table: string };
        if (this.answers[0]?.op === "tableMetadata") {
          const a = this.next("tableMetadata", `${schema}.${table}`);
          this.log.push({ op: "tableMetadata", schema, table, answer: { error: a.error } });
          throw new CoreCallError(a.error);
        }
        const m = this.metadata.find((t) => t.schema === schema && t.table === table);
        if (!m) this.fail(`no metadata for ${schema}.${table}`);
        const answer = { columns: m.columns, indexes: m.indexes };
        this.log.push({ op: "tableMetadata", schema, table, answer: structuredClone(answer) });
        return res({ kind: "tableMetadata", data: structuredClone(answer) });
      }
      if (BUILDERS.has(req.method)) {
        const out = dialectBuild(this.engine, req);
        this.log.push({
          op: "build",
          method: req.method,
          params: structuredClone(req.params),
          answer: structuredClone(out),
        });
        return res({ kind: "sqlWithBindings", data: structuredClone(out) });
      }
      this.fail(`unexpected engine call ${req.method}`);
    }
    if (method === "execute") {
      const text = params.sql as string;
      const a = this.next("execute", text);
      const { op: _op, ...answer } = a;
      this.log.push({ op: "execute", sql: text, params: structuredClone(params.params), answer });
      if (a.error) throw new CoreCallError(a.error);
      return res({ rows_affected: a.rowsAffected ?? 0, last_insert_id: a.lastInsertId ?? null });
    }
    if (method === "query") {
      const text = params.sql as string;
      const binds = structuredClone(params.params) as Wire[];
      const src = this.pageSource;
      if (!src) this.fail(`unexpected query (${text})`);
      const isCount = text.startsWith("SELECT COUNT(*) FROM");
      let answer: Record<string, unknown>;
      if (isCount) {
        answer = src.countError
          ? { error: src.countError }
          : { columns: ["count"], rows: [[src.countCell ?? src.matching.length]] };
      } else {
        const offset = (src.page - 1) * src.pageSize;
        answer = src.pageError
          ? { error: src.pageError }
          : {
              columns: src.columns,
              rows: src.matching.slice(offset, offset + src.pageSize),
            };
      }
      if (this.quiet) this.quietQueries++;
      else this.log.push({ op: isCount ? "count" : "page", sql: text, params: binds, answer });
      if (answer.error) throw new CoreCallError(answer.error as RpcErr);
      return res(structuredClone({ columns: answer.columns, rows: answer.rows }));
    }
    this.fail(`unexpected db.${method}`);
  }

  // eslint-disable-next-line require-yield
  async *stream() {
    this.failures.push("unexpected stream");
  }

  events() {
    return () => {};
  }
}

/** Stable names for change ids, in order of first appearance. */
function idNames() {
  const names = new Map<string, string>();
  return (id: string | undefined) => {
    if (id === undefined) return undefined;
    let n = names.get(id);
    if (!n) {
      n = `c${names.size + 1}`;
      names.set(id, n);
    }
    return n;
  };
}

function targetOut(t: PendingChangeTarget | undefined) {
  if (!t) return null;
  return {
    schema: t.schema,
    table: t.table,
    ...(t.column !== undefined ? { column: t.column } : {}),
    ...(t.primaryKeyValues ? { primaryKeyValues: wireRecord(t.primaryKeyValues) } : {}),
    ...("newValue" in t ? { newValue: wire(t.newValue) } : {}),
    ...(t.insertValues ? { insertValues: wireRecord(t.insertValues) } : {}),
  };
}

/** The edit intent (Decision 1) the GUI will send for a builder call it made today. */
function intentFrom(op: DriverOp) {
  const p = op.params as {
    schema: string;
    table: string;
    column?: string;
    value?: Wire;
    primary_keys?: string[];
    row?: [string, Wire][];
    values?: [string, Wire][];
  };
  const target = { schema: p.schema, table: p.table };
  const key = () =>
    (p.primary_keys ?? []).map((k) => [k, (p.row ?? []).find(([c]) => c === k)?.[1] ?? null]);
  switch (op.method) {
    case "buildUpdate":
      return { type: "updateCell", target, key: key(), column: p.column, value: p.value ?? null };
    case "buildSetDefault":
      return { type: "setDefault", target, key: key(), column: p.column };
    case "buildInsert":
      return { type: "insertRow", target, values: p.values };
    case "buildDelete":
      return { type: "deleteRow", target, key: key() };
  }
  throw new Error(`no intent for ${String(op.method)}`);
}

/** The code of a failure: the `CODE: ` prefix, or NO_ROWS_AFFECTED for the stale-key message. */
function codeOf(error: string | undefined, stale: boolean): string | undefined {
  if (error === undefined) return undefined;
  if (stale) return "NO_ROWS_AFFECTED";
  const m = /^([A-Z][A-Z0-9_]*): /.exec(error);
  return m ? m[1] : undefined;
}
const STALE_PREFIX = "No row in ";

// ---------------------------------------------------------------- summaries

/** `describePendingChange`'s fallback text per origin: no summary. */
const FALLBACKS = new Set([
  "Update cell",
  "Insert row",
  "Delete row",
  "Set column default",
  "Create table",
  "Alter table",
  "Drop table",
  "Truncate table",
]);
const SUMMARY_PATTERNS: [RegExp, string, boolean][] = [
  [/^Insert row into (.+)$/s, "insert", false],
  [/^Update (.+)$/s, "update", true],
  [/^Delete row from (.+)$/s, "delete", false],
  [/^Create table (.+)$/s, "createTable", false],
  [/^Create index (.+)$/s, "createIndex", false],
  [/^Drop table (.+)$/s, "dropTable", false],
  [/^Drop index (.+)$/s, "dropIndex", false],
  [/^Drop view (.+)$/s, "dropView", false],
  [/^Truncate table (.+)$/s, "truncate", false],
  [/^Alter table (.+)$/s, "alterTable", false],
];
/**
 * The `ChangeSummary` a description stands for (Decision 12), for the Rust
 * side, which compares these fields and never the English text. Derived from
 * the description; `null` where it is the SQL itself or an origin fallback.
 * For an index, `table` holds the index's name. The table and column are
 * split at the first `.` (no case has one inside a name).
 */
function summaryOf(description: string, origin: PendingChange["origin"]) {
  if (FALLBACKS.has(description)) return null;
  void origin;
  for (const [re, verb, hasColumn] of SUMMARY_PATTERNS) {
    const m = re.exec(description);
    if (!m) continue;
    if (!hasColumn) return { verb, table: m[1], column: null };
    const dot = m[1].indexOf(".");
    return dot < 0
      ? { verb, table: m[1], column: null }
      : { verb, table: m[1].slice(0, dot), column: m[1].slice(dot + 1) };
  }
  return null;
}

/** Decision 5's rule: every intent but truncate/drop is DML (a SQLite truncate is too), and typed insert/update/delete. */
function dmlOf(change: { type: string; edit?: { type: string } | null; sql?: string }, engine: DatabaseType) {
  if (change.type === "edit") {
    const t = change.edit?.type;
    if (t === "dropObject") return false;
    if (t === "truncateTable") return engine === "sqlite";
    return true;
  }
  const q = sql.detectQueryType(change.sql ?? "", engine);
  return q === "insert" || q === "update" || q === "delete";
}

  via: "dataTab" | "queryTab";
  /** Data tab: its table. */
  table?: { schema: string; name: string };
  /** Query tab: the SELECT the result came from. */
  query?: string;
  columns: string[];
  rows: Wire[][];
  steps: Step[];
  answers?: Answer[];
  change?: ChangeSpec;
}

const planCases: PlanCase[] = [];
const plan = (c: PlanCase) => planCases.push(c);
const ok = (rowsAffected = 1, lastInsertId?: number): Answer =>
  lastInsertId === undefined
    ? { op: "execute", rowsAffected }
    : { op: "execute", rowsAffected, lastInsertId };
const execErr = (message: string, code = "EXECUTE_ERROR"): Answer => ({
  op: "execute",
  error: { code, message },
});
const bigint = (v: string) => ({ $sq: "bigint", v });
const decimal = (v: string) => ({ $sq: "decimal", v });
const bytes = (v: string) => ({ $sq: "bytes", v });
const json = (v: unknown) => ({ $sq: "json", v });

/** `describePendingChange` for a queued entry, fixed where the regexes misread a quoted name. */
function describedAs(descriptions: string[], origins?: string[]): ChangeSpec["expected"] {
  return (r) => ({
    queue: (r.queue as { description: string; origin: string }[]).map((q, i) => {
      const origin = origins?.[i] ?? q.origin;
      return {
        ...q,
        description: descriptions[i],
        origin,
        summary: summaryOf(descriptions[i], origin as PendingChange["origin"]),
      };
    }),
  });
}

/**
 * The fixed expectation for a JSON column's values (the new Decision 19):
 * Core binds a value for a json/jsonb/JSON column as `Json`, from the
 * column's type in the metadata. `binds[i]` lists the bind positions of step
 * i (and of queue entry i) that are such values.
 */
function jsonBound(binds: number[][]): ChangeSpec["expected"] {
  const tag = (v: unknown) =>
    v !== null && typeof v === "object" && (v as { $sq?: string }).$sq === "json" ? v : { $sq: "json", v };
  const fix = (list: unknown[] | undefined, at: number[]) =>
    list?.map((v, i) => (at.includes(i) ? tag(v) : v));
  return (r) => ({
    steps: (r.steps as Record<string, unknown>[]).map((step, i) => ({
      ...step,
      driver: (step.driver as DriverOp[]).map((o) =>
        o.op === "build"
          ? { ...o, answer: { ...(o.answer as object), bindValues: fix((o.answer as { bindValues?: unknown[] }).bindValues, binds[i]) } }
          : o,
      ),
    })),
    queue: (r.queue as Record<string, unknown>[]).map((q, i) => ({ ...q, params: fix(q.params as unknown[], binds[i]) })),
  });
}
const JSON_WHY =
  "An array, number or bool bound to a json/jsonb/JSON column binds as JSON (Decision 19: Core tags it from the column's type in the metadata). Today a JSON cell that decoded to an array is sent as a SQL array and a number as a number: queued here, so nothing ran; run, each fails. Text stays text, so the database parses it as today: the grid sends typed JSON as a string, and a decoded JSON string (`\"abc\"`) can't be told apart from it (a quirk, kept).";

async function runPlan(c: PlanCase): Promise<Record<string, unknown>> {
  const metadata = c.metadata ?? c.cache.map(metaOf);
  const w = world(c.engine, c.cache, metadata, c.answers ?? []);
  rec.settings.enabled = c.pending;
  const names = idNames();
  const grid = c.rows.map((r) => r.map(js));

  let tabId = "tab-1";
  let result: StatementResult | undefined;
  if (c.via === "dataTab") {
    const t = c.table!;
    const cached = c.cache.find((x) => x.schema === t.schema && x.name === t.name);
    w.client.pageSource = { columns: c.columns, matching: c.rows, page: 1, pageSize: 100 };
    w.client.quiet = true;
    tabId = w.dataTabs.addWithoutRefresh(cached ?? tbl(t.schema, t.name, []))!;
    await w.dataTabs.refresh(tabId);
    result = w.state.dataTabsByProject.p!.find((x) => x.id === tabId)!.results;
  } else {
    const query = c.query!;
    const cache = w.state.schemas["conn-1"];
    result = {
      columns: c.columns,
      rows: grid,
      rowCount: grid.length,
      totalRows: grid.length,
      executionTime: 0,
      queryType: "select",
      page: 1,
      pageSize: 100,
      totalPages: 1,
      statementIndex: 0,
      statementSql: query,
      isError: false,
      connectionId: "conn-1",
      sourceTable: sql.sourceTableFromRef(sql.extractTableFromSelect(query, c.engine), cache),
      columnSources: sql.columnSourcesFromRefs(sql.columnRefs(query, c.engine), cache),
    };
    (w.state.queryTabsByProject as Record<string, unknown[]>).p = [
      { id: tabId, query, results: [result], activeResultIndex: 0 },
    ];
    w.client.pageSource = null;
  }
  const sourceTable = result?.sourceTable;
  if (!sourceTable && c.steps.some((s) => s.action !== "insertRow")) {
    throw new Error(`${c.name}: the result has no source table, so the grid can't edit it`);
  }

  const changeOf = new Map<string, unknown>();
  const steps: Record<string, unknown>[] = [];
  for (const step of c.steps) {
    const before = new Map(w.queue().map((q) => [q.id, q.sql + JSON.stringify(targetOut(q.target))]));
    const from = w.client.log.length;
    const quietBefore = w.client.quietQueries;
    const rerunsBefore = w.reruns();
    let outcome: Record<string, unknown>;
    if (c.via === "dataTab") {
      const tab = () => w.state.dataTabsByProject.p!.find((x) => x.id === tabId)!;
      if (step.action === "updateCell") {
        outcome = await w.dataTabs.updateCell(tabId, step.row, step.column, js(step.value));
      } else if (step.action === "setDefault") {
        outcome = await w.dataTabs.setCellDefault(tabId, step.row, step.column);
      } else if (step.action === "deleteRow") {
        outcome = await w.dataTabs.deleteRow(tabId, tab().results!.rows[step.row]);
      } else {
        const values = Object.fromEntries(Object.entries(step.values).map(([k, v]) => [k, js(v)]));
        w.dataTabs.addNewRow(tabId);
        const saved = await w.dataTabs.saveNewRow(tabId, 0, values);
        outcome = { saved };
      }
    } else {
      const st = sourceTable!;
      if (step.action === "updateCell") {
        outcome = await w.queries.updateCell(tabId, 0, step.row, step.column, js(step.value), st);
      } else if (step.action === "setDefault") {
        outcome = await w.queries.setCellDefault(tabId, 0, step.row, step.column, st);
      } else if (step.action === "deleteRow") {
        outcome = await w.queries.deleteRowAt(tabId, 0, step.row, st);
      } else {
        throw new Error("a query tab has no insert");
      }
    }
    await settle();
    const ops = w.client.log.slice(from);
    const build = ops.find((o) => o.op === "build");
    // A step that failed on the metadata read before the builder ran (SQLite
    // Set default) still sends its intent to Core, which fails the same way.
    const metadataFailed = ops.some((o) => o.op === "tableMetadata" && "error" in (o.answer as object));
    const edit = build
      ? intentFrom(build)
      : metadataFailed && c.via === "dataTab" && step.action !== "insertRow" && sourceTable
        ? {
            type: step.action === "setDefault" ? "setDefault" : step.action,
            target: { schema: sourceTable.schema, table: sourceTable.name },
            key: sourceTable.primaryKeys.map((k) => [k, c.rows[step.row][c.columns.indexOf(k)] ?? null]),
            ...("column" in step ? { column: step.column } : {}),
            ...("value" in step ? { value: step.value } : {}),
          }
        : null;
    for (const q of w.queue()) {
      if (before.get(q.id) !== q.sql + JSON.stringify(targetOut(q.target))) {
        changeOf.set(q.id, edit);
      }
    }
    const stale =
      typeof outcome.error === "string" && (outcome.error as string).startsWith(STALE_PREFIX);
    const out: Record<string, unknown> = { ...outcome };
    const code = codeOf(outcome.error as string | undefined, stale);
    if (code) out.code = code;
    steps.push({
      ...step,
      edit,
      driver: ops,
      outcome: out,
      ...(w.client.quietQueries !== quietBefore ? { refreshed: true } : {}),
      ...(w.reruns() !== rerunsBefore ? { reran: true } : {}),
    });
  }

  if (w.client.left().length) w.client.failures.push(`${w.client.left().length} answers left over`);
  expect(w.client.failures, c.name).toEqual([]);

  const queue = w.queue().map((q) => {
    const change = { type: "edit", id: names(q.id), edit: changeOf.get(q.id) ?? null };
    return {
      id: names(q.id),
      change,
      sql: q.sql,
      params: (q.bindValues ?? []).map(wire),
      queryType: q.queryType,
      dml: dmlOf(change as never, c.engine),
      summary: summaryOf(q.description, q.origin),
      description: q.description,
      origin: q.origin,
      target: targetOut(q.target),
    };
  });

  return {
    name: c.name,
    engine: c.engine,
    ...(c.notes ? { notes: c.notes } : {}),
    input: {
      via: c.via,
      pending: c.pending,
      ...(c.table ? { table: { schema: c.table.schema, table: c.table.name } } : {}),
      ...(c.query ? { query: c.query } : {}),
      result: {
        columns: c.columns,
        rows: c.rows,
        sourceTable: result?.sourceTable ?? null,
        columnSources: result?.columnSources ?? null,
      },
      schemaCache: c.cache,
    },
    metadata,
    steps,
    queue,
    toasts: [...rec.toasts],
  };
}

type QItem =
  | { edit: "updateCell"; table: SchemaTable; row: Record<string, Wire>; column: string; value: Wire }
  | { edit: "setDefault"; table: SchemaTable; row: Record<string, Wire>; column: string }
  | { edit: "deleteRow"; table: SchemaTable; row: Record<string, Wire> }
  | { edit: "insertRow"; table: SchemaTable; values: Record<string, Wire> }
  | { typed: string; params?: Wire[] }
  | { sidebar: "truncate" | "drop"; table: SchemaTable; kind?: "table" | "view" | "materialized-view" };

interface ApplyCase {
  name: string;
  engine: DatabaseType;
  notes?: string;
  /** Decision 5's mode for this queue (not recorded: the TS has none). */
  mode: "single" | "atomic" | "inOrder";
  cache: SchemaTable[];
  metadata?: Meta[];
  queue: QItem[];
  answers: Answer[];
  /** What the sheet sends as `confirmed` (it always asks first, so true unless a case says). */
  confirmed?: boolean;
  change?: ChangeSpec;
}

type Rec = Record<string, unknown>;
const outcomeOf = (r: Rec) => r.outcome as Rec;
const queueIds = (r: Rec) => (r.queue as Rec[]).map((q) => q.id);
const histFor = (r: Rec, rows: (number | null)[]) =>
  (r.queue as Rec[])
    .map((q, i) => ({ query: q.sql, rowCount: rows[i] }))
    .filter((h) => h.rowCount !== null);
/** An atomic batch that failed at `index`: nothing applied, the queue whole, no history. */
/** Refused unconfirmed: the one destructive statement (change 0) listed, nothing run. */
const confirmRequired = (text: string, engine: DatabaseType) => (r: Rec) => {
  const found = sql.findDestructiveStatements(sql.splitSqlStatementsOrThrow(text, engine), engine);
  expect(found.length, `${text} is destructive`).toBe(1);
  return {
    driver: [],
    outcome: {
      confirmRequired: true,
      destructive: [{ index: 0, sql: text, reason: JSON.parse(JSON.stringify(found[0].reason)) }],
      destructiveTotal: 1,
    },
    queueAfter: queueIds(r),
    history: [],
  };
};
const atomicFailure = (index: number, code: string) => (r: Rec) => ({
  outcome: {
    ...outcomeOf(r),
    executed: 0,
    failed: 1,
    failedAt: index,
    failedChangeId: queueIds(r)[index],
    code,
    hasDdl: false,
  },
  queueAfter: queueIds(r),
  history: [],
});

function isDml(item: QItem, engine: DatabaseType): boolean {
  if ("edit" in item) return true;
  if ("sidebar" in item) return item.sidebar === "truncate" && engine === "sqlite";
  const t = sql.detectQueryType(item.typed, engine);
  return t === "insert" || t === "update" || t === "delete";
}

async function runApply(c: ApplyCase): Promise<Record<string, unknown>> {
  const metadata = c.metadata ?? c.cache.map(metaOf);
  const w = world(c.engine, c.cache, metadata, c.answers);
  const names = idNames();
  rec.settings.enabled = true;

  const expectedMode =
    c.queue.length === 1 ? "single" : c.queue.every((i) => isDml(i, c.engine)) ? "atomic" : "inOrder";
  expect(c.mode, `${c.name}: Decision 5's mode`).toBe(expectedMode);

  const changeOf = new Map<string, unknown>();
  const connection = w.state.connections[0];
  for (const item of c.queue) {
    const before = new Set(w.queue().map((q) => q.id));
    const from = w.client.log.length;
    let change: (id: string) => unknown;
    if ("edit" in item) {
      const t = item.table;
      const st = {
        schema: t.schema,
        name: t.name,
        primaryKeys: t.columns.filter((x) => x.isPrimaryKey).map((x) => x.name),
      };
      const decode = (r: Record<string, Wire>) =>
        Object.fromEntries(Object.entries(r).map(([k, v]) => [k, js(v)]));
      let res: { success: boolean; error?: string };
      if (item.edit === "updateCell") {
        res = await w.queries.crud.updateCellDirect("conn-1", st, decode(item.row), item.column, js(item.value));
      } else if (item.edit === "setDefault") {
        res = await w.queries.crud.setCellDefaultDirect("conn-1", st, decode(item.row), item.column);
      } else if (item.edit === "deleteRow") {
        res = await w.queries.crud.deleteRow("conn-1", st, decode(item.row));
      } else {
        res = await w.queries.crud.insertRow("conn-1", st, decode(item.values));
      }
      if (!res.success) throw new Error(`${c.name}: queueing failed: ${res.error}`);
      const build = w.client.log.slice(from).find((o) => o.op === "build")!;
      const edit = intentFrom(build);
      change = (id) => ({ type: "edit", id, edit });
    } else if ("typed" in item) {
      const binds = (item.params ?? []).map(js);
      // `statementDeferred`: the substituted SQL, its query type, the binds.
      w.pending.add(
        "conn-1",
        item.typed,
        sql.detectQueryType(item.typed, c.engine),
        "query-editor",
        "tab-1",
        binds.length > 0 ? binds : undefined,
      );
      const params = item.params ?? [];
      change = (id) => ({ type: "sql", id, sql: item.typed, params });
    } else {
      // components/sidebar/manage/schema-tab.svelte, handleDropTable/handleTruncateTable.
      const from = getEngineClient(connection, w.state).qualifiedTable(item.table.schema, item.table.name);
      const target = { schema: item.table.schema, table: item.table.name };
      let text: string;
      if (item.sidebar === "truncate") {
        text = c.engine === "sqlite" ? `DELETE FROM ${from}` : `TRUNCATE TABLE ${from}`;
        change = (id) => ({ type: "edit", id, edit: { type: "truncateTable", target } });
      } else {
        const kind = item.kind ?? "table";
        const keyword =
          kind === "materialized-view" ? "MATERIALIZED VIEW" : kind === "view" ? "VIEW" : "TABLE";
        text = `DROP ${keyword} ${from}`;
        const objectKind = kind === "materialized-view" ? "materializedView" : kind;
        change = (id) => ({ type: "edit", id, edit: { type: "dropObject", target, kind: objectKind } });
      }
      const res = await w.queries.executeRawDdl("conn-1", text);
      if (!res.queued) throw new Error(`${c.name}: not queued`);
    }
    const added = w.queue().filter((q) => !before.has(q.id));
    if (added.length !== 1) throw new Error(`${c.name}: ${added.length} changes queued for one item`);
    changeOf.set(added[0].id, change(names(added[0].id)!));
  }

  const queue = w.queue().map((q) => ({
    id: names(q.id),
    change: changeOf.get(q.id),
    sql: q.sql,
    params: (q.bindValues ?? []).map(wire),
    queryType: q.queryType,
    dml: dmlOf(changeOf.get(q.id) as never, c.engine),
    summary: summaryOf(q.description, q.origin),
    description: q.description,
    origin: q.origin,
    target: targetOut(q.target),
  }));

  w.client.log.length = 0;
  const staleAnswer = (i: number | undefined) => {
    if (i === undefined) return false;
    const a = c.answers[i];
    return a?.op === "execute" && !a.error && (a.rowsAffected ?? 0) === 0;
  };
  const r = await w.pending.executeAll("conn-1");
  // The sheet's rule (`pending-changes-sheet.svelte`): a full success clears the queue.
  if (r.failed === 0) w.pending.clear("conn-1");
  const ops = w.client.log.splice(0);
  if (w.client.left().length) w.client.failures.push(`${w.client.left().length} answers left over`);
  expect(w.client.failures, c.name).toEqual([]);

  const stale = r.error !== undefined && staleAnswer(r.failedAt) && r.error.startsWith(STALE_PREFIX);
  const outcome: Record<string, unknown> = {
    executed: r.executed,
    failed: r.failed,
    ...(r.failedAt !== undefined ? { failedAt: r.failedAt } : {}),
    ...(r.failedChangeId !== undefined ? { failedChangeId: names(r.failedChangeId) } : {}),
    ...(r.error !== undefined ? { error: r.error, code: codeOf(r.error, stale) } : {}),
    hasDdl: r.hasDdl ?? false,
  };

  return {
    name: c.name,
    engine: c.engine,
    ...(c.notes ? { notes: c.notes } : {}),
    mode: c.mode,
    input: { confirmed: c.confirmed ?? true },
    metadata,
    queue,
    driver: ops,
    outcome,
    queueAfter: w.queue().map((q) => names(q.id)),
    history: w.historyCalls.map((h) => ({ ...h })),
    toasts: [...rec.toasts],
  };
}

/** The TS page SQL split into the base SELECT and the paging it appended. */
function splitPaging(engine: string, text: string) {
  if (engine === "mssql") {
    const m = /^([\s\S]*?)( ORDER BY \(SELECT NULL\))? OFFSET (\d+) ROWS FETCH NEXT (\d+) ROWS ONLY$/.exec(text);
    if (!m) throw new Error(`no paging in ${text}`);
    return { base: m[1], limit: Number(m[4]), offset: Number(m[3]) };
  }
  const m = /^([\s\S]*) LIMIT (\d+) OFFSET (\d+)$/.exec(text);
  if (!m) throw new Error(`no paging in ${text}`);
  return { base: m[1], limit: Number(m[2]), offset: Number(m[3]) };
}

/**
 * Placeholders as the engine's `crud.rs` builders write them: `@P1` on SQL
 * Server and `?` on DuckDB, where the data tab writes `@p1` and `$1`. (The
 * cases hold no `@p`/`$` inside names or strings.)
 */
const crudPlaceholders = (engine: string, text: string) =>
  engine === "mssql"
    ? text.replace(/@p(\d+)/g, "@P$1")
    : engine === "duckdb"
      ? text.replace(/\$\d+/g, "?")
      : text;

async function runPage(c: PageCase): Promise<Record<string, unknown>> {
  const w = world(c.engine, c.cache, c.cache.map(metaOf), []);
  const page = c.page ?? 1;
  const pageSize = c.pageSize ?? 100;
  w.client.pageSource = {
    columns: c.columns,
    matching: c.matching,
    page,
    pageSize,
    countCell: c.countCell,
    countError: c.countError,
    pageError: c.pageError,
  };
  const cached = c.cache.find((x) => x.schema === c.table.schema && x.name === c.table.name)!;
  const tabId = w.dataTabs.addWithoutRefresh(cached)!;
  const filters: DataFilter[] = (c.filters ?? []).map((x, i) => ({
    id: `f${i + 1}`,
    column: x.column ?? "",
    operator: x.operator ?? "=",
    value: x.value ?? "",
    enabled: x.enabled ?? true,
  }));
  const tabs = w.state.dataTabsByProject.p!;
  tabs[0] = { ...tabs[0], filters, filterLogic: c.logic ?? "AND", sortColumns: c.sort ?? [], page, pageSize };
  await w.dataTabs.refresh(tabId);
  const ops = w.client.log.splice(0);
  expect(w.client.failures, c.name).toEqual([]);

  const tab = w.state.dataTabsByProject.p!.find((x) => x.id === tabId)!;
  const res = tab.results!;
  const countOp = ops.find((o) => o.op === "count");
  const pageOp = ops.find((o) => o.op === "page");
  const split = pageOp ? splitPaging(c.engine, pageOp.sql as string) : null;
  if (split) {
    expect(split.offset, `${c.name}: offset`).toBe((page - 1) * pageSize);
    expect(split.limit, `${c.name}: limit`).toBe(pageSize);
  }

  const tableQuery = {
    target: { schema: c.table.schema, table: c.table.name },
    filters: filters.filter((x) => x.enabled && x.column).map((x) => ({ column: x.column, op: x.operator, value: x.value })),
    logic: c.logic ?? "AND",
    sort: (c.sort ?? []).map((s) => ({ column: s.column, direction: s.direction })),
  };

  const recorded: Record<string, unknown> = {
    name: c.name,
    engine: c.engine,
    ...(c.notes ? { notes: c.notes } : {}),
    input: {
      tableQuery,
      page,
      pageSize,
      filters,
      schemaCache: c.cache,
      columns: c.columns,
      matching: c.matching,
      ...(c.countCell !== undefined ? { countCell: c.countCell } : {}),
      ...(c.countError ? { countError: c.countError } : {}),
      ...(c.pageError ? { pageError: c.pageError } : {}),
    },
    driver: ops,
    select: split ? { sql: split.base, params: pageOp!.params } : null,
    paging: split ? { limit: split.limit, offset: split.offset } : null,
    count: countOp ? { sql: countOp.sql, params: countOp.params } : null,
    result: {
      columns: res.columns,
      rows: res.rows.map((row) => row.map(wire)),
      rowCount: res.rowCount,
      totalRows: res.totalRows,
      page: res.page,
      pageSize: res.pageSize,
      totalPages: res.totalPages,
      error: res.error ?? null,
      sourceTable: res.sourceTable ?? null,
    },
  };

  // Decision 9's rules, where Core differs.
  if (split) {
    const offset = (page - 1) * pageSize;
    const fetched = c.pageError ? [] : c.matching.slice(offset, offset + pageSize + 1);
    const full = !c.pageError && fetched.length > pageSize;
    const selectSql = crudPlaceholders(c.engine, split.base);
    const rowsShown = Math.min(fetched.length, pageSize);
    const totalRows = c.pageError
      ? 0
      : !full
        ? offset + rowsShown
        : c.countError
          ? offset + pageSize + 1
          : Number(js(c.countCell ?? c.matching.length));
    const expected: Record<string, unknown> = {};
    const why: string[] = [];
    if (!full && countOp) {
      expected.count = null;
      why.push(
        c.pageError
          ? "The page runs first and fails, so no count runs."
          : "A partial page runs no count: its total is `offset + rows`.",
      );
    }
    if (full && c.countError) {
      why.push("The page was full and the count failed: the total is estimated (`offset + pageSize + 1`) and flagged, not 0.");
    } else if (!full && c.countError) {
      why.push("Today the failed count shows a total of 0; Core runs no count for a partial page, so the total is exact.");
    }
    if (selectSql !== split.base) {
      expected.select = { sql: selectSql, params: pageOp!.params };
      why.push(
        c.engine === "mssql"
          ? "The builder writes SQL Server placeholders as crud.rs does, `@P1…` (the data tab writes `@p1`)."
          : "The builder writes DuckDB placeholders as crud.rs does, `?` (the data tab writes `$1`).",
      );
    }
    const result = recorded.result as Record<string, unknown>;
    if (!c.pageError && (result.totalRows !== totalRows || (full && c.countError))) {
      expected.result = {
        ...result,
        totalRows,
        totalPages: Math.max(1, Math.ceil(totalRows / pageSize)),
        countEstimated: !!(full && c.countError),
      };
    }
    if (why.length > 0) {
      changes[c.name] = { decision: "9", why: why.join(" "), expected };
    }
  }
  return recorded;
}

function runSummary(c: SummaryCase): Record<string, unknown> {
  const description = describePendingChange(c.sql, c.origin);
  if (c.fixed !== undefined) {
    expect(c.fixed, `${c.name}: the fix changes nothing`).not.toBe(description);
    changes[c.name] = {
      decision: "12",
      why: "The regexes misread a quoted, qualified or commented name; `change_summary` reads the statement with the scanner.",
      expected: { description: c.fixed },
    };
  }
  if (c.fixed !== undefined) {
    changes[c.name].expected.summary = summaryOf(c.fixed, c.origin);
  }
  return {
    name: c.name,
    engine: c.engine,
    ...(c.notes ? { notes: c.notes } : {}),
    origin: c.origin,
    sql: c.sql,
    description,
    summary: summaryOf(description, c.origin),
  };
}

// ---------------------------------------------------------------- recording

const recorded = new Map<string, Record<string, unknown>[]>();
const push = (file: string, out: Record<string, unknown>) => {
  const list = recorded.get(file) ?? [];
  list.push(out);
  recorded.set(file, list);
};

describe("record edit fixtures", () => {
  for (const c of planCases) {
    it(c.name, async () => {
      const out = await runPlan(c);
      push(c.file, out);
      if (c.change) {
        changes[c.name] = { decision: c.change.decision, why: c.change.why, expected: c.change.expected(out) };
      }
    });
  }
  for (const c of castCases) {
    it(c.name, () => {
      push("cast-map", {
        name: c.name,
        ...(c.notes ? { notes: c.notes } : {}),
        columns: c.columns,
        casts: castMapForColumns(c.columns),
      });
    });
  }
  for (const c of applyCases) {
    it(c.name, async () => {
      const out = await runApply(c);
      push("apply", out);
      if (c.change) {
        changes[c.name] = { decision: c.change.decision, why: c.change.why, expected: c.change.expected(out) };
      }
    });
  }
  for (const c of pageCases) {
    it(c.name, async () => {
      push(c.file, await runPage(c));
    });
  }
  for (const c of summaryCases) {
    it(c.name, () => {
      push("summary", runSummary(c));
    });
  }

  it("has unique names", () => {
    const all = [...planCases, ...castCases, ...applyCases, ...pageCases, ...summaryCases].map((c) => c.name);
    expect(new Set(all).size).toBe(all.length);
  });

  afterAll(() => {
    if (!process.env.FREEZE_EDITS) return;
    const total = planCases.length + castCases.length + applyCases.length + pageCases.length + summaryCases.length;
    if ([...recorded.values()].flat().length !== total) {
      throw new Error("not every case recorded; nothing written");
    }
    const dir =
      process.env.FREEZE_EDITS_OUT ?? join(process.cwd(), "crates/seaquel-workspace/tests/fixtures/edits");
    mkdirSync(dir, { recursive: true });
    const engines = ["postgres", "mysql", "mariadb", "sqlite", "mssql", "duckdb"];
    const files = [
      ...engines.map((e) => `plan-${e}`),
      "cast-map",
      "apply",
      ...engines.map((e) => `table-page-${e}`),
      "summary",
    ];
    // Keep each file's cases in definition order.
    const order = new Map(
      [...planCases, ...castCases, ...applyCases, ...pageCases, ...summaryCases].map((c, i) => [c.name, i]),
    );
    for (const file of files) {
      const list = (recorded.get(file) ?? []).sort(
        (a, b) => order.get(a.name as string)! - order.get(b.name as string)!,
      );
      writeFileSync(join(dir, `${file}.json`), JSON.stringify(list, null, 2) + "\n");
    }
    const sorted = Object.fromEntries(Object.keys(changes).sort().map((k) => [k, changes[k]]));
    writeFileSync(join(dir, "changes.json"), JSON.stringify(sorted, null, 2) + "\n");
  });
});
