// ---------------------------------------------------------------- the in-memory disk

const disk = vi.hoisted(() => {
  type Node =
    | { kind: "file"; text: string }
    | { kind: "dir" }
    | { kind: "symlink"; target: string };
  const nodes = new Map<string, Node>();
  /** Real directory paths whose listing fails (EACCES). */
  const unreadable = new Set<string>();
  /** Real file paths whose writes fail. */
  const failing = new Set<string>();
  const opts = { caseInsensitive: false };

  const norm = (p: string) => {
    const out: string[] = [];
    for (const part of p.split("/")) {
      if (!part || part === ".") continue;
      if (part === "..") out.pop();
      else out.push(part);
    }
    return "/" + out.join("/");
  };
  const parent = (p: string) => {
    const n = norm(p);
    return n.slice(0, n.lastIndexOf("/")) || "/";
  };
  const base = (p: string) => norm(p).slice(norm(p).lastIndexOf("/") + 1);
  /** The stored key for `p`: itself or, case-insensitively, one that differs only in case. */
  const keyOf = (p: string) => {
    if (nodes.has(p) || !opts.caseInsensitive) return p;
    const lower = p.toLowerCase();
    return [...nodes.keys()].find((k) => k.toLowerCase() === lower) ?? p;
  };
  const err = (code: string, p: string) => new Error(`${code}: ${p}`);

  /** The real path of `p`, following symlinks (the last one only with `followLast`). */
  const resolve = (p: string, followLast = true, depth = 0): string => {
    const parts = norm(p).split("/").filter(Boolean);
    let cur = "";
    for (let i = 0; i < parts.length; i++) {
      const next = keyOf(`${cur}/${parts[i]}`);
      const node = nodes.get(next);
      const last = i === parts.length - 1;
      if (node?.kind === "symlink" && (!last || followLast)) {
        if (depth > 40) throw err("ELOOP", p);
        const target = node.target.startsWith("/") ? node.target : `${cur}/${node.target}`;
        cur = resolve(target, true, depth + 1);
        if (cur === "/") cur = "";
      } else {
        cur = next;
      }
    }
    return cur || "/";
  };
  const isDir = (real: string) => real === "/" || nodes.get(real)?.kind === "dir";
  const mkdirp = (p: string) => {
    const parts = norm(p).split("/").filter(Boolean);
    let cur = "";
    for (const part of parts) {
      const next = resolve(`${cur}/${part}`);
      if (!nodes.has(next)) nodes.set(next, { kind: "dir" });
      else if (!isDir(next)) throw err("ENOTDIR", p);
      cur = next;
    }
  };
  const reset = () => {
    nodes.clear();
    unreadable.clear();
    failing.clear();
    opts.caseInsensitive = false;
  };
  /** Files and symlinks by path, sorted (directories aren't listed: git doesn't keep them). */
  const tree = (): Record<string, unknown> => {
    const out: Record<string, unknown> = {};
    for (const k of [...nodes.keys()].sort()) {
      const n = nodes.get(k)!;
      if (n.kind === "file") out[k] = n.text;
      else if (n.kind === "symlink") out[k] = { symlink: n.target };
    }
    return out;
  };
  /** Seeds `tree` (`{path: text | {symlink} | {dir: true}}`), making parent directories. */
  const seed = (t: Record<string, unknown>) => {
    for (const [p, v] of Object.entries(t)) {
      mkdirp(parent(p));
      if (typeof v === "string") nodes.set(norm(p), { kind: "file", text: v });
      else if (v && typeof v === "object" && "symlink" in v)
        nodes.set(norm(p), { kind: "symlink", target: (v as { symlink: string }).symlink });
      else mkdirp(p);
    }
  };
  /** The files and symlinks under real directory `dir`, relative to it. */
  const under = (dir: string): Map<string, string> => {
    const out = new Map<string, string>();
    for (const [k, n] of nodes) {
      if (!k.startsWith(`${dir}/`) || k.startsWith(`${dir}/.git/`)) continue;
      if (n.kind === "file") out.set(k.slice(dir.length + 1), n.text);
      else if (n.kind === "symlink") out.set(k.slice(dir.length + 1), `->${n.target}`);
    }
    return out;
  };
  return {
    nodes,
    unreadable,
    failing,
    opts,
    norm,
    parent,
    base,
    keyOf,
    resolve,
    isDir,
    mkdirp,
    reset,
    tree,
    seed,
    under,
    err,
  };
});

function repoStatus(path: string) {
  const g = rec.git.get(path) ?? {};
  if (g.statusError) throw new Error(g.statusError);
  const now = disk.under(disk.resolve(path));
  const was = rec.committed.get(path) ?? new Map<string, string>();
  const modified: string[] = [];
  const untracked: string[] = [];
  for (const [p, text] of now) {
    if (!was.has(p)) untracked.push(p);
    else if (was.get(p) !== text) modified.push(p);
  }
  for (const p of was.keys()) if (!now.has(p)) modified.push(p);
  modified.sort();
  untracked.sort();
  const conflictFiles = g.conflictFiles ?? [];
  return {
    isClean: modified.length + untracked.length === 0 && conflictFiles.length === 0,
    pendingChanges: modified.length + untracked.length,
    aheadBy: g.aheadBy ?? 0,
    behindBy: g.behindBy ?? 0,
    hasConflicts: conflictFiles.length > 0,
    currentBranch: "main",
    modifiedFiles: modified,
    untrackedFiles: untracked,
    conflictFiles,
  };
}

/** A value as JSON holds it: `undefined` dropped (and `null` in a list), `NaN` as the string "NaN". */
function plain(v: unknown): Json {
  if (v === undefined) return null;
  const text = JSON.stringify(v, (_k, x) => {
    if (typeof x === "number" && Number.isNaN(x)) return "NaN";
    if (x instanceof Date) return x.toISOString();
    return x;
  });
  return text === undefined ? null : JSON.parse(text);
}

function errorText(e: unknown): string {
  return e instanceof Error ? e.message : String(e);
}

/**
 * Ids made during a case become `<id:n>` (numbered in order of first
 * appearance in the case's JSON, then its `changes.json` entry, keeping
 * their prefix), and times from the pinned clock `<now>`. Seeded ids and
 * times (2024) stay.
 */
function normalise(caseValue: Json, change: Json): [Json, Json] {
  let text = JSON.stringify([caseValue, change]);
  const seen = new Map<string, string>();
  text = text.replace(/00000000-0000-4000-8000-\d{12}/g, (m) => {
    if (!seen.has(m)) seen.set(m, `<id:${seen.size + 1}>`);
    return seen.get(m)!;
  });
  text = text.replace(/2030-01-01T\d\d:\d\d:\d\d\.\d{3}Z/g, "<now>");
  const [a, b] = JSON.parse(text);
  return [a, b];
}

/**
 * Q21's file stem (Decision 39), as `changes.json` expects it: NFC, lower
 * case, NFC again; every run of characters that aren't letters, marks or
 * digits (any script) becomes `-`; `-` trimmed from both ends; `untitled`
 * when nothing is left; a Windows reserved device name (`con`, `prn`, `aux`,
 * `nul`, `com1`–`com9`, `lpt1`–`lpt9`) gets `-file` appended. A trailing
 * dot or space can't survive the rule. The collision suffix (`-2`, `-3`) is
 * `free_path`'s, which needs the directory, so the projection cases pin it.
 */
function q21Stem(name: string): string {
  const s = name
    .normalize("NFC")
    .toLowerCase()
    .normalize("NFC")
    .replace(/[^\p{L}\p{M}\p{N}]+/gu, "-")
    .replace(/^-+|-+$/g, "");
  if (!s) return "untitled";
  if (/^(con|prn|aux|nul|com[0-9¹²³]|lpt[0-9¹²³])$/.test(s)) return `${s}-file`;
  return cutBytes(s, STEM_MAX_BYTES);
}

/**
 * Q21's byte cap: a file name is at most 255 bytes, so the stem alone is cut
 * to 250 (255 less the longest extension, `.json` or `.yaml`), and
 * `free_path` cuts it again when a `-n` suffix would take the name past 255.
 * Cuts fall on a character boundary, and a `-` left at the end goes.
 */
const STEM_MAX_BYTES = 250;
function cutBytes(s: string, max: number): string {
  const enc = new TextEncoder();
  let out = "";
  let bytes = 0;
  for (const ch of s) {
    const n = enc.encode(ch).length;
    if (bytes + n > max) break;
    out += ch;
    bytes += n;
  }
  return out.replace(/-+$/, "");
}
/** `free_path`'s name for a stem taken `n - 1` times already (`n` ≥ 2), with extension `ext`. */
function q21Suffixed(stem: string, n: number, ext: string): string {
  const suffix = `-${n}`;
  return `${cutBytes(stem, 255 - new TextEncoder().encode(suffix + ext).length)}${suffix}${ext}`;
}

/** Decision 45's scalar writer: today's rule, with `"` and newlines fixed. */
function q45Scalar(v: string, inList = false): string {
  const needs =
    v.includes(":") ||
    v.includes("#") ||
    v.includes("\n") ||
    v.includes('"') ||
    v.startsWith(" ") ||
    v.endsWith(" ") ||
    v.includes("[") ||
    v.includes("]") ||
    (inList && v.includes(","));
  if (!needs) return v;
  // Single quotes keep `"` and `\` as they are, and older readers strip
  // them exactly (Decision 52); double quotes only when they can't be used.
  if (!v.includes("'") && !v.includes("\n")) {
    if (v.includes('"') || v.includes("\\")) return `'${v}'`;
    return `"${v}"`;
  }
  return `"${v.replace(/\\/g, "\\\\").replace(/"/g, '\\"').replace(/\n/g, "\\n")}"`;
}

/** Decision 45's scalar reader, the inverse of `q45Scalar`. */
function q45Read(raw: string): string {
  if (raw.length >= 2 && raw.startsWith("'") && raw.endsWith("'"))
    return raw.slice(1, -1).replace(/''/g, "'");
  if (raw.length >= 2 && raw.startsWith('"') && raw.endsWith('"')) {
    return raw.slice(1, -1).replace(/\\(["\\n])/g, (_m, c: string) => (c === "n" ? "\n" : c));
  }
  return raw;
}

interface FormatCase {
  name: string;
  note?: string;
  fn: string;
  args: Obj;
  /** Core's result where it's meant to differ (Decision 45, Q21, Q22). */
  change?: { decision: string; why: string; expected: Obj };
}

/** Query parse: `text` at `path` (relative to the repo). */
const pq = (
  name: string,
  text: string,
  path = `${QUERIES}/orders.sql`,
  extra: Partial<FormatCase> = {},
): FormatCase => ({
  name: `parse-query/${name}`,
  fn: "parseQuery",
  args: { text, path, queriesDir: QUERIES },
  ...extra,
});
const wq = (name: string, query: Obj, extra: Partial<FormatCase> = {}): FormatCase => ({
  name: `write-query/${name}`,
  fn: "writeQuery",
  args: { query },
  ...extra,
});

/** Runs one format case: today's output (and for a writer, the file read back). */
function recordFormat(c: FormatCase): Obj {
  const a = c.args as Record<string, never>;
  let output: Json;
  let reparsed: Json;
  switch (c.fn) {
    case "parseQuery":
      output = qfp.parseQueryFile(a.text, REPO, a.path, a.queriesDir);
      break;
    case "writeQuery": {
      const text = qfp.serializeQueryFile(a.query);
      output = text;
      reparsed = qfp.parseQueryFile(text, REPO, `${QUERIES}/x.sql`, QUERIES);
      const r = reparsed as Obj;
      delete r.id;
      delete r.repoId;
      delete r.filePath;
      break;
    }
    case "parseDashboard":
      output = dfp.parseDashboardFile(a.text, REPO, a.path);
      break;
    case "writeDashboard": {
      const text = dfp.serializeDashboardFile(a.dashboard);
      output = text;
      reparsed = dfp.parseDashboardFile(text, REPO, `${DASH}/x.json`);
      const r = reparsed as Obj;
      delete r.id;
      delete r.repoId;
      delete r.filePath;
      break;
    }
    case "parseTemplate":
      output = cfp.parseConnectionFile(a.text, REPO, SHARED_PROJECT, a.path);
      break;
    case "writeTemplate": {
      const text = cfp.serializeConnectionFile({
        ...(a.template as Obj),
        id: "",
        repoId: REPO,
        projectId: SHARED_PROJECT,
        filePath: TEMPLATE_PATH,
      } as never);
      output = text;
      reparsed = cfp.parseConnectionFile(text, REPO, SHARED_PROJECT, TEMPLATE_PATH);
      break;
    }
    case "parseProject":
      output = cfp.parseProjectFile(a.text, REPO, a.dir);
      break;
    case "writeProject": {
      const text = cfp.serializeProjectFile({
        ...(a.project as Obj),
        id: "",
        repoId: REPO,
        dirName: "team",
        connections: [],
      } as never);
      output = text;
      reparsed = cfp.parseProjectFile(text, REPO, "team");
      break;
    }
    case "parseLabels":
      output = cfp.parseLabelsFile(a.text);
      break;
    case "writeLabels": {
      const text = cfp.serializeLabelsFile({ labels: a.labels });
      output = text;
      reparsed = cfp.parseLabelsFile(text);
      break;
    }
    case "fileName":
      output = {
        stem: cfp.nameToFilename(a.name),
        query: qfp.queryNameToFilename(a.name),
        dashboard: dfp.dashboardNameToFilename(a.name),
      };
      break;
    default:
      throw new Error(`unknown fn ${c.fn}`);
  }
  return {
    name: c.name,
    ...(c.note ? { note: c.note } : {}),
    fn: c.fn,
    args: plain(c.args),
    output: plain(output),
    ...(reparsed !== undefined ? { reparsed: plain(reparsed) } : {}),
  };
}

/** Self-checks of the expected values written by hand above. */
function checkFormatExpectations(): void {
  for (const c of formatCases) {
    const exp = c.change?.expected as { output?: unknown } | undefined;
    if (c.fn.startsWith("write") && typeof exp?.output === "string") {
      // The expected text follows Decision 45's scalar rule for its quoted values.
      expect(exp.output.endsWith("\n")).toBe(true);
    }
  }
  expect(q45Scalar('Sales "Q3"')).toBe("'Sales \"Q3\"'");
  expect(q45Scalar(`Dana's "Q3"`)).toBe('"Dana\'s \\"Q3\\""');
  expect(q45Scalar("C:\\temp: scratch")).toBe("'C:\\temp: scratch'");
  expect(q45Scalar("it's: \\x")).toBe('"it\'s: \\\\x"');
  // Every value Core writes reads back exactly (Decision 34's hash needs it).
  for (const v of [
    'Sales "Q3"',
    `Dana's "Q3"`,
    "C:\\temp: scratch",
    "it's: \\x",
    "line one\nline two",
    "a,b",
    " padded ",
    "plain",
    "x: 'y' \\ \"z\"",
    "C:\\new: x",
  ]) {
    expect(q45Read(q45Scalar(v))).toBe(v);
    expect(q45Read(q45Scalar(v, true))).toBe(v);
  }
  expect(q45Scalar("a,b", true)).toBe('"a,b"');
  expect(q45Scalar("line one\nline two")).toBe('"line one\\nline two"');
}

/** An XML plist holding `value` (dicts, arrays, strings, integers, reals, booleans). */
function toPlist(value: unknown): string {
  const esc = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  const out: string[] = [];
  const walk = (v: unknown, depth: number) => {
    const pad = "\t".repeat(depth);
    if (Array.isArray(v)) {
      if (v.length === 0) return void out.push(`${pad}<array/>`);
      out.push(`${pad}<array>`);
      for (const x of v) walk(x, depth + 1);
      out.push(`${pad}</array>`);
    } else if (v && typeof v === "object") {
      const entries = Object.entries(v).filter(([, x]) => x !== undefined);
      if (entries.length === 0) return void out.push(`${pad}<dict/>`);
      out.push(`${pad}<dict>`);
      for (const [k, x] of entries) {
        out.push(`${pad}\t<key>${esc(k)}</key>`);
        walk(x, depth + 1);
      }
      out.push(`${pad}</dict>`);
    } else if (typeof v === "string") {
      out.push(`${pad}<string>${esc(v)}</string>`);
    } else if (typeof v === "number") {
      out.push(Number.isInteger(v) ? `${pad}<integer>${v}</integer>` : `${pad}<real>${v}</real>`);
    } else if (typeof v === "boolean") {
      out.push(`${pad}<${v}/>`);
    } else {
      throw new Error(`a plist can't hold ${String(v)}`);
    }
  };
  walk(value, 0);
  return [
    '<?xml version="1.0" encoding="UTF-8"?>',
    '<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">',
    '<plist version="1.0">',
    ...out,
    "</plist>",
    "",
  ].join("\n");
}

/**
 * Decision 47's candidate for one of today's importable connections: the
 * entry's id as `key`, today's fields, `duplicateOf` (the saved connection
 * with the same type, host, port, database and user) instead of
 * `isDuplicate`/`selected`, and `problem: "invalidPort"` (with port 0) for
 * a port that isn't a number.
 */
function candidateOf(
  c: Obj,
  existing: Existing[],
  id?: { key: string; problem?: "noId" | "duplicateId" },
): Obj {
  const original = c.original as Obj;
  const port = c.port as number;
  // Not a number, or outside 0–65535 (which the create would refuse).
  const invalid = Number.isNaN(port) || port < 0 || port > 65535;
  const problem = id?.problem ?? (invalid ? "invalidPort" : undefined);
  const dup = existing.find(
    (e) =>
      e.type === c.type &&
      e.host === c.host &&
      e.port === port &&
      e.databaseName === c.databaseName &&
      e.username === c.username,
  );
  return {
    key: id?.key ?? String(original.id),
    name: c.name,
    type: c.type,
    host: c.host,
    port: invalid ? 0 : port,
    databaseName: c.databaseName,
    username: c.username,
    ...(c.sslMode !== undefined ? { sslMode: c.sslMode } : {}),
    ...(c.sshTunnel !== undefined ? { sshTunnel: c.sshTunnel } : {}),
    ...(dup ? { duplicateOf: dup.id } : {}),
    ...(problem ? { problem } : {}),
  };
}

const IMPORT_WHY =
  "Decision 47 (bug 22): `importsCandidates` answers `{found: false}` when there is no file, `{found: true, unreadable: <message>}` when it can't be read or doesn't parse (`<message>` is any non-empty text naming no path), and otherwise `{found: true, candidates}`, today's list in today's order (unsupported drivers still left out): each candidate has `key` (TablePlus: `id:<ID>`, the entry's `ID` as text with an `id:` prefix, or `pos:<n>` for an entry keyed by its 0-based position in the list; DBeaver: the connection's key in `connections`, as it is), today's `name`, `type`, `host`, `port`, `databaseName`, `username`, `sslMode` and `sshTunnel`, `duplicateOf` (the id of the project's connection with the same type, host, port, database and user) in place of `isDuplicate` and `selected`, and `problem: \"invalidPort\"` with `port: 0` for a port that isn't a number (today `NaN`, recorded as \"NaN\") or is outside 0–65535 (`-1`, `65536`, `70000`). Other ports follow `parseInt` (leading digits). TablePlus entries are every dict in the list: one with no `ID` gets `key: \"pos:<n>\"` and `problem: \"noId\"` (today it's dropped), and entries sharing an `ID` each get `pos:<n>` and `problem: \"duplicateId\"`. The prefixes keep the two kinds of key apart: no `ID` can make an `id:` key equal a `pos:` one; those problems come before `invalidPort`. `importsCreate` refuses a key whose candidate has a `problem` (`INVALID_ARGUMENT`, naming the key), whatever the GUI sends.";

async function recordImport(source: "tableplus" | "dbeaver", c: ImportCase): Promise<[Obj, Obj]> {
  const existing = c.existing ?? [];
  const isError =
    c.input !== null &&
    typeof c.input === "object" &&
    !Array.isArray(c.input) &&
    "error" in (c.input as Obj) &&
    Object.keys(c.input as Obj).length === 1;
  if (source === "tableplus") {
    rec.importText.tableplus = isError
      ? new Error((c.input as { error: string }).error)
      : c.input === null
        ? null
        : JSON.stringify(c.input);
  } else {
    rec.importText.dbeaver = isError
      ? new Error((c.input as { error: string }).error)
      : (c.input as string | null);
  }
  const discovered =
    source === "tableplus"
      ? await discoverTablePlusConnections(existing as never)
      : await discoverDbeaverConnections(existing as never);
  const plistFile =
    source === "tableplus" && c.input !== null && (!isError || c.plistText)
      ? `plist/${c.name}.plist`
      : undefined;
  const recorded = {
    name: c.name,
    ...(c.note ? { note: c.note } : {}),
    input: plain(c.input),
    ...(plistFile ? { plist: plistFile } : {}),
    existing: plain(existing),
    output: plain(discovered),
  };
  let expected: Obj;
  if (c.input === null) expected = { found: false };
  else if (isError) expected = { found: true, unreadable: "<message>" };
  else if (source === "dbeaver") {
    let parsed: unknown;
    try {
      parsed = JSON.parse(c.input as string);
    } catch {
      parsed = undefined;
    }
    expected =
      parsed === undefined
        ? { found: true, unreadable: "<message>" }
        : {
            found: true,
            candidates: (discovered as unknown as Obj[]).map((d) => candidateOf(d, existing)),
          };
  } else if (!Array.isArray(c.input)) {
    expected = { found: true, unreadable: "<message>" };
  } else {
    // Every dict in the list, in order, mapped as today. One with no `ID`, or
    // whose `ID` another entry has too, is listed with `pos:<n>` as its
    // key and a problem (M8), instead of being dropped or confused.
    const entries = c.input as unknown[];
    const ids = entries.map((e) =>
      e && typeof e === "object" && !Array.isArray(e) && "ID" in e ? String((e as Obj).ID) : null,
    );
    const candidates: Obj[] = [];
    entries.forEach((e, i) => {
      if (!e || typeof e !== "object" || Array.isArray(e)) return;
      const mapped = mapTablePlus(toTablePlusConnection(e as Obj), existing as never);
      if (!mapped) return;
      const id = ids[i];
      const shared = id !== null && ids.filter((x) => x === id).length > 1;
      candidates.push(
        candidateOf(
          mapped as unknown as Obj,
          existing,
          id === null
            ? { key: `pos:${i}`, problem: "noId" }
            : shared
              ? { key: `pos:${i}`, problem: "duplicateId" }
              : { key: `id:${id}` },
        ),
      );
    });
    expected = { found: true, candidates };
  }
  return [recorded, plain(expected) as Obj];
}

/** The plist file for a TablePlus case, if it has one. */
function plistOf(c: ImportCase): string | null {
  if (c.plistText) return c.plistText;
  if (
    c.input === null ||
    (typeof c.input === "object" && !Array.isArray(c.input) && "error" in (c.input as Obj))
  )
    return null;
  return toPlist(c.input);
}

async function insertRows(db: Db, rows: PSeed["rows"]): Promise<void> {
  for (const table of SEED_TABLES) {
    for (const row of rows[table] ?? []) {
      const cols = Object.keys(row);
      await db.execute(
        `INSERT INTO ${table} (${cols.join(", ")}) VALUES (${cols.map(() => "?").join(", ")})`,
        cols.map((c) => row[c]),
      );
    }
  }
}

  await db.execute(`CREATE TABLE dashboards_beta (id TEXT PRIMARY KEY, name TEXT NOT NULL,
    viewport TEXT NOT NULL DEFAULT '{"x":0,"y":0,"zoom":1}', widgets TEXT NOT NULL DEFAULT '[]',
    date_filter TEXT, created_at TEXT NOT NULL, updated_at TEXT NOT NULL, starred INTEGER DEFAULT 0,
    shared INTEGER NOT NULL DEFAULT 0, description TEXT, project_id TEXT)`);
  await db.execute("DROP TABLE dashboards");
  await db.execute("ALTER TABLE dashboards_beta RENAME TO dashboards");
  await db.execute("DROP INDEX IF EXISTS idx_dashboards_project");
  await db.execute("CREATE INDEX idx_dashboards_project ON dashboards(project_id)");
  await db.execute("PRAGMA foreign_keys=ON");
}

async function dump(db: Db): Promise<Rows> {
  const out: Rows = {};
  for (const [table, cols, order] of DUMP) {
    const where = table === "app_state" ? " WHERE key = 'activeRepoId'" : "";
    const rows = await db.query<Row>(`SELECT ${cols} FROM ${table}${where} ORDER BY ${order}`);
    if (rows.length) out[table] = rows;
  }
  return out;
}

/** Lets fire-and-forget writes finish and the 500 ms repo-list debounce fire. */
async function settle(): Promise<void> {
  for (let i = 0; i < 8; i++) {
    await new Promise((r) => setImmediate(r));
    await vi.advanceTimersByTimeAsync(600);
  }
}

function recordingLibrary(lib: InstanceType<typeof TsLibrary>): InstanceType<typeof TsLibrary> {
  return new Proxy(lib, {
    get(target, key, receiver) {
      const value = Reflect.get(target, key, receiver);
      if (typeof value !== "function" || typeof key !== "string") return value;
      return async (...args: unknown[]) => {
        const entry: { method: string; args: unknown[]; made?: string; error?: string } = {
          method: key,
          args: plain(args) as unknown[],
        };
        rec.library.push(entry);
        try {
          const result = await value.apply(target, args);
          const made = (result as { value?: { id?: unknown } })?.value?.id;
          if (key.startsWith("create") && typeof made === "string") entry.made = made;
          return result;
        } catch (e) {
          entry.error = errorText(e);
          throw e;
        }
      };
    },
  });
}

/** One page: the managers as `UseDatabase` wires them. */
class Page {
  readonly state = new DatabaseState();
  readonly windowState: InstanceType<typeof WindowStateManager>;
  readonly restoration: InstanceType<typeof StateRestorationManager>;
  readonly dashboards: InstanceType<typeof DashboardManager>;
  readonly projects: InstanceType<typeof ProjectManager>;
  readonly connections: InstanceType<typeof ConnectionManager>;
  readonly savedQueries: InstanceType<typeof SavedQueryManager>;
  readonly sharedRepos: InstanceType<typeof SharedRepoManager>;
  readonly sharedQueries: InstanceType<typeof SharedQueryManager>;
  readonly sharedDashboards: InstanceType<typeof SharedDashboardManager>;

  constructor(db: Db) {
    const windowId = "main";
    const ui = new TsUi(db, { origin: () => windowId });
    this.windowState = new WindowStateManager(this.state, {
      ui: () => ui,
      windowId: () => windowId,
    });
    this.restoration = new StateRestorationManager(this.state);
    this.dashboards = new DashboardManager(
      this.state,
      async () => [],
      () => {},
    );
    this.projects = new ProjectManager(
      this.state,
      this.windowState,
      this.restoration,
      this.dashboards,
    );
    const provider = { connect: vi.fn(async () => "core-1"), disconnect: vi.fn(async () => {}) };
    this.connections = new ConnectionManager(
      this.state,
      this.windowState,
      this.restoration,
      {} as never,
      { getForType: async () => provider } as never,
      async () => {},
      () => {},
    );
    this.savedQueries = new SavedQueryManager(this.state, () => {});
    this.sharedRepos = new SharedRepoManager(this.state);
    this.sharedQueries = new SharedQueryManager(this.state, this.sharedRepos);
    this.sharedDashboards = new SharedDashboardManager(this.state, this.sharedRepos);
    this.savedQueries.setFileProjection({
      writeQueryFile: (q) => this.sharedQueries.writeQueryFile(q),
      deleteQueryFile: (q) => this.sharedQueries.deleteQueryFile(q),
    });
    this.dashboards.setFileProjection({
      writeDashboardFile: (d) => this.sharedDashboards.writeDashboardFile(d),
      deleteDashboardFile: (d) => this.sharedDashboards.deleteDashboardFile(d),
    });
    this.projects.setRemoveConnectionCallback((id, options) =>
      this.connections.remove(id, options),
    );
    this.projects.setSharedRepoManager(this.sharedRepos);
    this.projects.setSharedQueryManager(this.sharedQueries);
    this.projects.setSharedDashboardManager(this.sharedDashboards);
    this.projects.setConnectionManager(this.connections);
    this.connections.setSharedRepoManager(this.sharedRepos);
  }

  /** `UseDatabase.initializeApp`: projects, connections, then `initializeSharedRepos` (copied). */
  async open(): Promise<void> {
    await this.projects.initialize();
    await this.connections.initializePersistedConnections();
    const { repos } = await this.sharedRepos.loadPersistedRepos();
    this.state.sharedRepos = repos.map(deserializeRepo);
    await Promise.all(
      this.state.sharedRepos.map(async (repo) => {
        await this.sharedRepos.loadQueriesFromRepo(repo.id);
        await this.sharedRepos.refreshRepoStatus(repo.id);
      }),
    );
    await this.projects.activateLinkedRepo(this.state.activeProjectId);
    // `startBackgroundRefresh` isn't started: its ticks are steps of their own.
  }

  query(ref: string) {
    for (const list of Object.values(this.state.queriesByProject)) {
      const q = list.find((x) => x.id === ref || x.name === ref);
      if (q) return q;
    }
    throw new Error(`no saved query ${ref}`);
  }

  dashboard(ref: string) {
    for (const list of Object.values(this.state.dashboardsByProject)) {
      const d = list.find((x) => x.id === ref || x.name === ref);
      if (d) return d;
    }
    throw new Error(`no dashboard ${ref}`);
  }

  view(): Json {
    const s = this.state;
    return plain({
      activeProjectId: s.activeProjectId,
      activeRepoId: s.activeRepoId,
      projects: s.projects.map((p) => ({
        id: p.id,
        name: p.name,
        gitRepoPath: p.gitRepoPath ?? null,
      })),
      connections: s.connections.map((c) => ({
        id: c.id,
        projectId: c.projectId,
        name: c.name,
        host: c.host,
        port: c.port,
        databaseName: c.databaseName,
        isLocalOnly: !!c.isLocalOnly,
        sharedConnectionId: c.sharedConnectionId ?? null,
      })),
      queries: Object.fromEntries(
        Object.entries(s.queriesByProject).map(([p, qs]) => [
          p,
          qs.map((q) => ({
            id: q.id,
            name: q.name,
            folder: q.folder ?? null,
            shared: !!q.shared,
            query: q.query,
          })),
        ]),
      ),
      dashboards: Object.fromEntries(
        Object.entries(s.dashboardsByProject).map(([p, ds]) => [
          p,
          ds.map((d) => ({
            id: d.id,
            name: d.name,
            shared: !!d.shared,
            widgets: d.widgets.map((w) => w.id),
          })),
        ]),
      ),
      repos: s.sharedRepos.map((r) => ({ id: r.id, name: r.name, path: r.path })),
      sharedProjects: Object.fromEntries(
        Object.entries(s.sharedProjectsByRepo).map(([r, ps]) => [
          r,
          ps.map((p) => ({
            dirName: p.dirName,
            name: p.name,
            connections: p.connections.map((c) => c.filePath),
          })),
        ]),
      ),
      scan: Object.fromEntries(
        s.sharedRepos.map((r) => [
          r.id,
          {
            queries: (s.sharedQueriesByRepo[r.id] ?? []).map((q) => q.filePath),
            dashboards: (s.sharedDashboardsByRepo[r.id] ?? []).map((d) => d.filePath),
          },
        ]),
      ),
    });
  }
}

class PCtx {
  page: Page | null = null;
  /** Ids the case's steps made, by label. */
  made = new Map<string, string>();
  constructor(readonly db: Db) {}
  get p(): Page {
    if (!this.page) throw new Error("no page open");
    return this.page;
  }
  /** Whether a GUI on Core would sync `projectId`: it has a git path (Core resolves the repo itself). */
  linked(projectId: string | null): boolean {
    return !!this.page?.state.projects.find((p) => p.id === projectId)?.gitRepoPath;
  }
  repoIdOf(path: string): string | null {
    return this.page?.state.sharedRepos.find((r) => r.path === path)?.id ?? null;
  }
}

/** The library writes recorded during a step, as the `library` calls a GUI on Core sends. */
function libraryCalls(entries: typeof rec.library): CoreCall[] {
  const out: CoreCall[] = [];
  for (const e of entries) {
    const map = LIBRARY_WRITES[e.method];
    if (!map || e.error) continue;
    const { method, params } = map(e.args);
    out.push({
      group: "library",
      method,
      params,
      ...((e as { made?: string }).made ? { binds: (e as { made?: string }).made } : {}),
    });
  }
  return out;
}

async function recordProjection(c: PCase): Promise<RawPCase> {
  uuidSeq = 0;
  rec.toasts.length = 0;
  rec.library.length = 0;
  rec.committed.clear();
  rec.git.clear();
  disk.reset();
  vi.useFakeTimers({
    toFake: ["setTimeout", "clearTimeout", "setInterval", "clearInterval", "Date"],
    now: FIXED,
  });
  try {
    disk.opts.caseInsensitive = !!c.seed.caseInsensitive;
    disk.seed(c.seed.tree);
    for (const d of c.seed.unreadable ?? []) disk.unreadable.add(d);
    for (const f of c.seed.failing ?? []) disk.failing.add(f);
    // Each repo (a directory holding `.git`) has its seed files committed.
    for (const k of disk.nodes.keys()) {
      if (k.endsWith("/.git")) {
        const repo = k.slice(0, -"/.git".length);
        rec.committed.set(repo, disk.under(repo));
      }
    }
    for (const [path, g] of Object.entries(c.seed.git ?? {})) {
      rec.git.set(path, JSON.parse(JSON.stringify(g)));
    }
    const db = new WebSqliteDatabase(new SQL.Database());
    await bootstrapSqljsDatabase(db);
    if (c.file === "v2026.4.5-beta.1") await makeBetaEra(db);
    await insertRows(db, c.seed.rows);
    const library = recordingLibrary(new TsLibrary(db));
    setLibrary(library);
    storage.client = createSqljsStorageClient(db);
    const t = new PCtx(db);
    const before = await dump(db);
    const steps: RawPStep[] = [];
    for (const step of c.steps) {
      rec.toasts.length = 0;
      rec.library.length = 0;
      let outcome: Json;
      try {
        const value = await step.run(t);
        outcome = { ok: true, value: plain(value) };
      } catch (e) {
        outcome = { ok: false, error: errorText(e) };
      }
      await settle();
      const authored = step.core ? step.core(t) : [];
      const core = step.authoredOnly ? authored : [...authored, ...libraryCalls(rec.library)];
      steps.push({
        op: step.op,
        ...(step.args !== undefined ? { args: plain(step.args) } : {}),
        core: core.length ? (plain(core) as CoreCall[]) : null,
        outcome,
        library: rec.library.map((e) => plain(e)),
        rows: await dump(db),
        tree: disk.tree(),
        toasts: [...rec.toasts],
        view: t.page ? t.page.view() : null,
      });
    }
    return {
      name: c.name,
      ...(c.note ? { note: c.note } : {}),
      file: c.file ?? "current",
      seed: plain(c.seed) as PSeed,
      ...(c.secrets ? { secrets: c.secrets } : {}),
      steps,
      before,
    };
  } finally {
    setLibrary(null);
    vi.clearAllTimers();
    vi.useRealTimers();
  }
}

const clone = <T>(v: T): T => JSON.parse(JSON.stringify(v)) as T;
const treeAt = (raw: RawPCase, i: number): Tree =>
  clone(i < 0 ? disk_seed_tree(raw) : raw.steps[i].tree);
const rowsAt = (raw: RawPCase, i: number): Rows => clone(i < 0 ? raw.before : raw.steps[i].rows);

/** The seed's tree as the recorder lays it out (files and symlinks only). */
function disk_seed_tree(raw: RawPCase): Tree {
  const out: Tree = {};
  for (const [k, v] of Object.entries(raw.seed.tree)) {
    if (typeof v === "string" || (v && typeof v === "object" && "symlink" in (v as Obj)))
      out[disk.norm(k)] = v;
  }
  return Object.fromEntries(Object.entries(out).sort(([a], [b]) => (a < b ? -1 : 1)));
}

function sortTree(t: Tree): Tree {
  return Object.fromEntries(Object.entries(t).sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)));
}

/** `tree` with each path starting `from` moved to start `to`. */
function movePaths(tree: Tree, from: string, to: string): Tree {
  const out: Tree = {};
  for (const [k, v] of Object.entries(tree))
    out[k.startsWith(from) ? to + k.slice(from.length) : k] = v;
  return sortTree(out);
}

function withFiles(tree: Tree, files: Record<string, unknown>): Tree {
  const out = { ...tree };
  for (const [k, v] of Object.entries(files)) {
    if (v === undefined || v === null) delete out[k];
    else out[k] = v;
  }
  return sortTree(out);
}

function patchRow(rows: Rows, table: string, match: (r: Row) => boolean, patch: Row): Rows {
  const out = clone(rows);
  out[table] = (out[table] ?? []).map((r) => (match(r) ? { ...r, ...patch } : r));
  return out;
}

function dropRows(rows: Rows, table: string, match: (r: Row) => boolean): Rows {
  const out = clone(rows);
  out[table] = (out[table] ?? []).filter((r) => !match(r));
  if (out[table].length === 0) delete out[table];
  return out;
}

/** Drops connection `id` and takes it out of every connection order. */
function dropConnection(rows: Rows, id: string): Rows {
  let out = dropRows(rows, "connections", (r) => r.id === id);
  out = clone(out);
  out.project_state = (out.project_state ?? []).map((r) => ({
    ...r,
    connection_order: JSON.stringify(
      (JSON.parse(r.connection_order as string) as string[]).filter((x) => x !== id),
    ),
  }));
  return out;
}

/** The same expectation for each step from `from` to the last. */
function fromStep(
  raw: RawPCase,
  from: number,
  f: (i: number) => StepExpect,
): Record<number, StepExpect> {
  const out: Record<number, StepExpect> = {};
  for (let i = from; i < raw.steps.length; i++) out[i] = f(i);
  return out;
}

/**
 * `rows` without `project_state`: the connection order a page fills in on a
 * project's first activation is its own `projectSidebarSet`, which a step
 * whose `core` is only the projection's calls never sends (re-review I1).
 */
function withoutPageOrder(rows: Rows): Rows {
  const out = clone(rows);
  delete out.project_state;
  return out;
}

function rowBy(rows: Rows, table: string, match: (r: Row) => boolean): Row {
  const r = (rows[table] ?? []).find(match);
  if (!r) throw new Error(`no ${table} row`);
  return r;
}

/**
 * The standard seed: `p1` "Team" (active), linked to `/repos/a` (`repo-a`)
 * with its directory `team` (as linking made it), and `p2` "Solo" without a
 * git path. `linked: false` leaves `p1` unlinked and the repo unregistered.
 */
function teamSeed(
  o: {
    linked?: boolean;
    rows?: PSeed["rows"];
    tree?: Tree;
    git?: PSeed["git"];
    unreadable?: string[];
    failing?: string[];
  } = {},
): PSeed {
  const linked = o.linked ?? true;
  const extra = o.rows ?? {};
  return {
    rows: {
      projects: [projectRow("p1", "Team", linked ? A : null), projectRow("p2", "Solo")],
      ...(linked ? { shared_repos: [repoRow("repo-a", A, "team-repo")] } : {}),
      app_state: [
        { key: "lastActiveProjectId", value: "p1" },
        ...(linked ? [{ key: "activeRepoId", value: "repo-a" }] : []),
      ],
      ...Object.fromEntries(Object.entries(extra).filter(([k]) => k !== "projects")),
      ...(extra.projects ? { projects: extra.projects } : {}),
    },
    tree: {
      [`${A}/.git`]: { dir: true },
      ...(linked
        ? {
            [`${TEAM}/project.yaml`]: "name: Team\n",
            [`${TEAM}/connections`]: { dir: true },
            [`${TEAM}/queries`]: { dir: true },
            [`${TEAM}/dashboards`]: { dir: true },
          }
        : {}),
      ...o.tree,
    },
    ...(o.git ? { git: o.git } : {}),
    ...(o.unreadable ? { unreadable: o.unreadable } : {}),
    ...(o.failing ? { failing: o.failing } : {}),
  };
}

describe("shared fixtures: projection", () => {
  for (const c of projectionCases) {
    it(c.name, async () => {
      const raw = await recordProjection(c);
      const steps = c.change ? c.change.expected(raw) : null;
      // `app_state.activeRepoId` is checked against the seed (`*`), never
      // against expected rows (Decision 42): no expected step lists it.
      for (const step of Object.values(steps ?? {})) delete step.rows?.app_state;
      const [recorded, change] = normalise(
        pcaseJson(raw),
        c.change && steps
          ? { decision: c.change.decision, why: c.change.why, expected: { steps } }
          : null,
      );
      out.projection.push(recorded);
      if (change) out.projectionChanges[c.name] = change;
      // Every step ran: an outcome that isn't ok is recorded, but a case
      // whose steps all fail is a broken case, not a recording.
      expect(raw.steps.some((s) => (s.outcome as { ok: boolean }).ok)).toBe(true);
    });
  }
});
