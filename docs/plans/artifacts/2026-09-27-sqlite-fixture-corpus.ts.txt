/**
 * SQLite corpus (Rust Core Phase 2, Task 8), recorded into
 * `crates/seaquel-engine-sqlite/tests/fixtures/`:
 *
 *   npm run fixtures:record -- sqlite
 *
 * No server: the Rust driver creates a fresh temp-file database for each
 * recording (not `:memory:`, which would give every pooled connection its
 * own database) and `cleanup` deletes it. `SEAQUEL_TEST_SQLITE` (ConnectConfig
 * JSON) overrides the location.
 *
 * The scratch objects are the `scratch` block of `bugfixes.json`, so the Rust
 * tests build the same ones. Tables whose names fail the adapter's
 * `validateIdentifier` (`my-table`, `order items`, `it's`) can't be recorded
 * through the adapter's PRAGMA queries; their expectations are hand-written
 * bug-fix cases (fix 1) and checked live by `tests/smoke.rs`. Their per-table
 * row counts are skipped here, as `TsEngineClient` skips a count that throws.
 *
 * EXPLAIN: `EXPLAIN QUERY PLAN` is the same statement with and without
 * ANALYZE (TsEngineClient executes the query separately and times it; that
 * is checked live in Task 9, not replayed here), so most targets record the
 * plain form only. Synthetic plans (`parse.extra`) cover detail strings the
 * bundled SQLite no longer prints (pre-3.36 `SCALAR SUBQUERY` without a
 * number, ` AS alias`, bare `MATERIALIZE`/`CO-ROUTINE`, `AUTOMATIC INDEX`
 * without COVERING, the LEFT/RIGHT PART OF ORDER BY forms), orphan parent
 * ids, and an empty plan.
 */

/** A plan built by hand: `[id, parent, detail]` rows as EXPLAIN QUERY PLAN returns them. */
function synthetic(name: string, rows: [number, number, string][], analyze = false): NamedInput {
  return {
    name: `synthetic: ${name}`,
    input: {
      sql: "(synthetic)",
      analyze,
      columns: ["id", "parent", "notused", "detail"],
      rows: rows.map(([id, parent, detail]) => ({ id, parent, notused: 0, detail })),
    },
  };
}
