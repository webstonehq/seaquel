/**
 * DuckDB corpus (Rust Core Phase 2, Task 17), recorded into
 * `crates/seaquel-engine-duckdb/tests/fixtures/`:
 *
 *   npm run fixtures:record -- duckdb
 *
 * No server: the recorder copies `e2e/test-databases/duckdb/seaquel_test.duckdb`
 * (seeded by `npm run e2e:db:seed`) and its WAL into a fresh temp directory,
 * records against the copy through the native Rust driver, and `cleanup`
 * deletes it. The original file is never opened. The copy keeps the file
 * name, so its catalog (the file name up to its first dot) is `seaquel_test`,
 * which the plans' `Table` names and the scratch setup's `USE seaquel_test`
 * rely on.
 *
 * The scratch objects are the `scratch` block of `bugfixes.json`, so the Rust
 * tests build the same ones: a `fx_sales` schema, a table of every DuckDB type
 * (`main.fx_types`), tables whose names the adapter's `validateIdentifier`
 * rejects, an attached in-memory catalog `fx_aux` whose tables share names
 * with `seaquel_test`'s, and schema and catalog names with dots and quotes
 * (`"fx.a.b"`, `"fx_aux.main"`, `"fx.we""ird"`). The adapter's catalog
 * queries ignore the catalog, so:
 * - `main.users` and `fx_sales.regions` are recorded with the other catalog's
 *   columns mixed in (fix 6 replaces them);
 * - `main.categories` isn't recorded: its primary-key subquery matches both
 *   catalogs' constraints and DuckDB fails the whole query (fix 6, live);
 * - the table sizes list `fx_aux`'s tables under `main`/`fx_sales` again, so
 *   the per-table row counts (which would repeat case names and count the
 *   wrong table) are recorded once per name through `parse.extra`.
 *
 * Every parse input is text, booleans or BIGINT counts, so none of it depends
 * on the value decoding Task 16 changes (see the README).
 *
 * EXPLAIN ANALYZE plans carry timings, memory and CPU figures that change on
 * every run. They are recorded through `parse.extra` and normalized (see
 * `normalizeAnalyzed`) so re-recording is byte-identical.
 *
 * No CRUD group: the Rust dialect binds parameters (`?`, decision 4 of the
 * phase 2 plan) instead of the adapter's inline literals, and is checked with
 * live tests.
 */

let copied = false;

/** Copies the seeded database (and its WAL) once per recording. */
function copyDatabase(): void {
  if (copied) return;
  if (!existsSync(SOURCE)) {
    throw new Error(`${SOURCE} is missing; seed it with npm run e2e:db:seed`);
  }
  rmSync(DIR, { recursive: true, force: true });
  mkdirSync(DIR, { recursive: true });
  copyFileSync(SOURCE, DB);
  if (existsSync(`${SOURCE}.wal`)) copyFileSync(`${SOURCE}.wal`, `${DB}.wal`);
  copied = true;
}

/** Keys of an analyzed plan whose numbers change between runs. */
const VOLATILE = /timing|latency|cpu_time|thread_time|memory|bytes|temp_dir/;
/**
 * `operator_timing`, in document order. Includes a 0 so the parser's
 * zero-timing path is covered.
 */
const TIMINGS = [0.0005, 0.00025, 0, 0.001, 0.0002, 0.00015, 0.00075];
/** The root's `latency` (seconds): an execution time of 12.5 ms. */
const LATENCY = 0.0125;

type JsonObject = { [key: string]: Json };

/**
 * An analyzed plan with its run-dependent numbers replaced, so re-recording
 * is byte-identical: `operator_timing` takes `TIMINGS` in document order,
 * `latency` is `LATENCY`, and the other timing, CPU, memory and byte counters
 * are 0. Row counts and cardinalities are kept. DuckDB writes the keys of a
 * node in no fixed order, so they are sorted; the parser doesn't care.
 */
function normalizeAnalyzed(text: string): string {
  let next = 0;
  const walk = (value: Json, key?: string): Json => {
    if (Array.isArray(value)) return value.map((v) => walk(v));
    if (value !== null && typeof value === "object") {
      const out: JsonObject = {};
      for (const k of Object.keys(value).sort()) out[k] = walk(value[k], k);
      return out;
    }
    if (typeof value !== "number" || key === undefined) return value;
    if (key === "operator_timing") return TIMINGS[next++ % TIMINGS.length];
    if (key === "latency") return LATENCY;
    return VOLATILE.test(key) ? 0 : value;
  };
  return JSON.stringify(walk(JSON.parse(text) as Json), null, 4);
}

async function analyzedPlans(ctx: RecordContext): Promise<NamedInput[]> {
  const out: NamedInput[] = [];
  for (const target of EXPLAIN_TARGETS.filter((t) => t.analyzed)) {
    const { columns, rows } = await ctx.session.select(
      ctx.adapter.getExplainQuery(target.sql, true),
    );
    const normalized = rows.map((row) => ({
      ...row,
      explain_value:
        typeof row.explain_value === "string"
          ? normalizeAnalyzed(row.explain_value)
          : row.explain_value,
    }));
    out.push({
      name: `EXPLAIN ANALYZE: ${target.name}`,
      input: { sql: target.sql, analyze: true, columns, rows: normalized },
    });
  }
  return out;
}

/** A plan built by hand: the `explain_value` text `EXPLAIN (FORMAT JSON)` would return. */
function synthetic(name: string, plan: Json | string, analyze = false): NamedInput {
  return {
    name: `synthetic: ${name}`,
    input: {
      sql: "(synthetic)",
      analyze,
      columns: ["explain_key", "explain_value"],
      rows: [
        {
          explain_key: analyze ? "analyzed_plan" : "physical_plan",
          explain_value: typeof plan === "string" ? plan : JSON.stringify(plan, null, 4),
        },
      ],
    },
  };
}

/**
 * One per table name: the adapter's table sizes list `fx_aux`'s tables again.
 * `main.fx_items` (only in `fx_aux`) is left out: its count fails, and
 * TsEngineClient keeps such a table uncounted.
 */
const ROW_COUNT_TARGETS = [
  { schema: "main", table: "users" },
  { schema: "main", table: "categories" },
  { schema: "main", table: "orders" },
  { schema: "main", table: "fx_types" },
  { schema: "fx_sales", table: "orders" },
  { schema: "fx_sales", table: "regions" },
];

async function rowCounts(ctx: RecordContext): Promise<NamedInput[]> {
  const count = ctx.adapter.getTableRowCountQuery?.bind(ctx.adapter);
  if (!count) throw new Error("the DuckDB adapter counts rows per table");
  const out: NamedInput[] = [];
  for (const { schema, table } of ROW_COUNT_TARGETS) {
    const { columns, rows } = await ctx.session.select(count(table, schema));
    out.push({
      name: `row count ${schema}.${table}`,
      input: { kind: "rowCount", schema, table, columns, rows },
    });
  }
  return out;
}
