def q(name):
    return "[" + name.replace("]", "]]") + "]"


def lit(text):
    return "N'" + text.replace("'", "''") + "'"


def table(schema, name):
    return f"{q(schema)}.{q(name)}"


# ─── Model of the fixed ALTER generator ─────────────────────────────

def col_type(c):
    if c.get("precision"):
        return f"{c['type']}({c['precision']})"
    if c.get("length"):
        return f"{c['type']}({c['length']})"
    return c["type"]


def sanitize(v):
    t = (v or "").strip()
    if not t or ";" in t or "--" in t:
        return ""
    return t


def drop_default(n, t, column):
    return (
        f"DECLARE @df{n} nvarchar(max) = (SELECT {lit('ALTER TABLE ' + t + ' DROP CONSTRAINT ')} + QUOTENAME(d.name) "
        f"FROM sys.default_constraints d INNER JOIN sys.columns c ON c.object_id = d.parent_object_id AND c.column_id = d.parent_column_id "
        f"WHERE d.parent_object_id = OBJECT_ID({lit(t)}) AND c.name = {lit(column)}); "
        f"IF @df{n} IS NOT NULL EXEC (@df{n});"
    )


def drop_index(t, name):
    return (
        f"IF EXISTS (SELECT 1 FROM sys.key_constraints WHERE parent_object_id = OBJECT_ID({lit(t)}) AND name = {lit(name)}) "
        f"ALTER TABLE {t} DROP CONSTRAINT {q(name)} ELSE DROP INDEX {q(name)} ON {t};"
    )


def alter_sql(frm, to):
    t = table(to["schemaName"], to["tableName"])
    orig_t = table(frm["schemaName"], frm["tableName"])
    stmts = []
    n = [0]

    def var():
        n[0] += 1
        return n[0]

    orig = {c["id"]: c for c in frm["columns"]}
    new = {c["id"]: c for c in to["columns"]}
    new_idx = {i["name"] for i in to["indexes"]}
    orig_idx = {i["name"] for i in frm["indexes"]}
    # Fix 13: dropped indexes first, before the column changes they'd block.
    for i in frm["indexes"]:
        if i["name"] and i["name"] not in new_idx:
            stmts.append(drop_index(orig_t, i["name"]))  # fix 12
    for c in to["columns"]:
        o = orig.get(c["id"])
        if o and o["name"] != c["name"]:
            stmts.append(f"EXEC sp_rename {lit(t + '.' + q(o['name']))}, {lit(c['name'])}, N'COLUMN';")
    for c in to["columns"]:
        if c["id"] not in orig:
            s = f"ALTER TABLE {t} ADD {q(c['name'])} {col_type(c)}"
            s += " NULL" if c["nullable"] else " NOT NULL"  # fix 7: explicit, whatever ANSI_NULL_DFLT says
            if sanitize(c["defaultValue"]):
                s += f" DEFAULT {sanitize(c['defaultValue'])}"
            stmts.append(s + ";")
    for c in frm["columns"]:
        if c["id"] not in new:
            # Fix 13: the default constraint goes first, whatever the definition
            # says about a default (the lookup is guarded by IS NOT NULL).
            stmts.append(drop_default(var(), t, c["name"]) + f" ALTER TABLE {t} DROP COLUMN {q(c['name'])};")
    for c in to["columns"]:
        o = orig.get(c["id"])
        if not o:
            continue
        if col_type(o) != col_type(c) or o["nullable"] != c["nullable"]:
            coll = f" COLLATE {c['collation']}" if c.get("collation") else ""  # fix 15
            stmts.append(f"ALTER TABLE {t} ALTER COLUMN {q(c['name'])} {col_type(c)}{coll} {'NULL' if c['nullable'] else 'NOT NULL'};")
        if o["defaultValue"] != c["defaultValue"]:
            s = drop_default(var(), t, c["name"])
            if sanitize(c["defaultValue"]):
                s += f" ALTER TABLE {t} ADD DEFAULT {sanitize(c['defaultValue'])} FOR {q(c['name'])};"
            stmts.append(s)
    for i in to["indexes"]:
        if i["name"] and i["name"] not in orig_idx:
            u = "UNIQUE " if i["unique"] else ""
            stmts.append(f"CREATE {u}INDEX {q(i['name'])} ON {t} ({', '.join(q(x) for x in i['columns'])});")
    key = lambda f: f"{f['column']}->{f['referencedTable']}.{f['referencedColumn']}"
    orig_fks = {key(f) for f in frm["foreignKeys"]}
    for f in to["foreignKeys"]:
        if key(f) not in orig_fks:
            ref = table(f["referencedSchema"], f["referencedTable"]) if f["referencedSchema"] else q(f["referencedTable"])
            stmts.append(f"ALTER TABLE {t} ADD FOREIGN KEY ({q(f['column'])}) REFERENCES {ref} ({q(f['referencedColumn'])});")
    return "\n".join(stmts) if stmts else "-- No changes detected"

def case(name, fix, kind, output, replaces=None, input=None):
    c = {"name": name, "fix": fix, "kind": kind}

        file, rec = replaces.split(": ", 1)

# CREATE TABLE model: fix 2 (]] quoting) and an explicit NULL for nullable
# columns (fix 7's rule for ADD, applied to CREATE: nullability otherwise
# depends on the session's ANSI_NULL_DFLT settings).
def create_sql(d, explicit_null=True):
    lines = []
    for c in d["columns"]:
        line = f"  {q(c['name'])} {col_type(c)}" + ((" NULL" if explicit_null else "") if c["nullable"] else " NOT NULL")
        if sanitize(c["defaultValue"]):
            line += f" DEFAULT {sanitize(c['defaultValue'])}"
        lines.append(line)
    pk = [q(c["name"]) for c in d["columns"] if c["isPrimaryKey"]]
    if pk:
        lines.append(f"  PRIMARY KEY ({', '.join(pk)})")
    for c in d["columns"]:
        if c["isUnique"] and not c["isPrimaryKey"]:
            lines.append(f"  UNIQUE ({q(c['name'])})")
    for f in d["foreignKeys"]:
        lines.append(f"  FOREIGN KEY ({q(f['column'])}) REFERENCES {table(f['referencedSchema'], f['referencedTable'])} ({q(f['referencedColumn'])})")
    t = table(d["schemaName"], d["tableName"])
    out = f"CREATE TABLE {t} (\n" + ",\n".join(lines) + "\n);"
    for i in d["indexes"]:
        u = "UNIQUE " if i["unique"] else ""
        out += f"\n\nCREATE {u}INDEX {q(i['name'])} ON {t} ({', '.join(q(x) for x in i['columns'])});"
    return out

# Fix 8/14: the pagination model: a small T-SQL tokenizer.
WORD = set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_@#$")


def paginate_sql(sql, limit, offset):
    """Fix 8: ORDER BY counts only at depth 0, outside strings ('..' with '', N'..'),
    [..] (]] escaped) and ".." ("" escaped) names and -- / nested /* */ comments.
    Fix 14: whatever follows the last token that isn't ';' (whitespace, ';', comments)
    is dropped before the clause is appended."""
    i, n, depth = 0, len(sql), 0
    words = []  # top-level words, uppercased
    cut = 0  # end of the last significant token that isn't ';'
    while i < n:
        ch = sql[i]
        if ch.isspace():
            i += 1
        elif sql.startswith("--", i):
            j = sql.find("\n", i)
            i = n if j < 0 else j + 1
        elif sql.startswith("/*", i):
            level, i = 1, i + 2
            while i < n and level:
                if sql.startswith("/*", i):
                    level, i = level + 1, i + 2
                elif sql.startswith("*/", i):
                    level, i = level - 1, i + 2
                else:
                    i += 1
        elif ch in "'\"[":
            close = "]" if ch == "[" else ch
            i += 1
            while i < n:
                if sql[i] == close:
                    if i + 1 < n and sql[i + 1] == close:
                        i += 2
                        continue
                    i += 1
                    break
                i += 1
            cut = i
            if depth == 0:
                words.append(None)
        elif ch in WORD:
            j = i
            while j < n and sql[j] in WORD:
                j += 1
            if depth == 0:
                words.append(sql[i:j].upper())
            i = cut = j
        else:
            if ch == "(":
                depth += 1
            elif ch == ")":
                depth -= 1
            if ch != ";":
                cut = i + 1
            if depth == 0 and ch not in "()":
                words.append(None)
            i += 1
    has_order = any(a == "ORDER" and b == "BY" for a, b in zip(words, words[1:]))
    body = sql[:cut]
    order = "" if has_order else " ORDER BY (SELECT NULL)"
    return f"{body}{order} OFFSET {offset} ROWS FETCH NEXT {limit} ROWS ONLY"

def shape(node, analyze):
    out = {"nodeType": node["nodeType"]}
    for k in ("relationName", "indexName", "filter"):
        if k in node:
            out[k] = node[k].replace("]]", "]") if k != "filter" else node[k]
    if analyze:
        for k in ("actualRows", "actualLoops"):
            if k in node:
                out[k] = node[k]
    out["children"] = [shape(c, analyze) for c in node["children"]]
    return out

def parse_columns(rows):
    out = []
    for c in rows:
        o = {"name": c["column_name"], "type": c["data_type"], "nullable": c["is_nullable"] == "YES"}
        if c["column_default"]:
            o["defaultValue"] = c["column_default"]
        o["isPrimaryKey"] = c["is_primary_key"] == 1
        o["isForeignKey"] = c["is_foreign_key"] == 1
        if c["is_foreign_key"] and c["referenced_table"]:
            o["foreignKeyRef"] = {"referencedSchema": c["referenced_schema"] or "dbo",
                                  "referencedTable": c["referenced_table"],
                                  "referencedColumn": c["referenced_column"] or ""}
        out.append(o)
    return out

def editor_def(schema, tbl, cols):
    """A definition as the table editor builds it from SchemaColumn (create-table-tabs addFromTable)."""
    out = []
    for i, c in enumerate(cols):
        typ, length, precision = c["type"], None, None
        if "(" in typ:
            typ, params = typ[:-1].split("(", 1)
            if "," in params:
                precision = params
            else:
                length = params
        d = {"id": f"c{i + 1}", "name": c["name"], "type": typ, "nullable": c["nullable"],
             "defaultValue": c.get("defaultValue", ""), "isPrimaryKey": c["isPrimaryKey"], "isUnique": False}
        if length:
            d["length"] = length
        if c.get("collation"):
            d["collation"] = c["collation"]
        if precision:
            d["precision"] = precision
        out.append(d)
    return {"tableName": tbl, "schemaName": schema, "columns": out, "indexes": [], "foreignKeys": []}

        for k in ("relationName", "indexName"):
            if k in n:
                n[k] = n[k].replace("]]", "]")
        for ch in n["children"]:
            unescape(ch)

    unescape(out["plan"])
    if out == c["output"]:
        continue
    case(f"fix 11: ]] in plan names is un-escaped ({name})", 11, "parse-explain", out,
         replaces=f"parse-explain.json: {name}")
