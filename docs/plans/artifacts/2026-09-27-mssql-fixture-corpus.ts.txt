/**
 * MSSQL corpus (Rust Core Phase 2, Task 12), recorded into
 * `crates/seaquel-engine-mssql/tests/fixtures/`:
 *
 *   npm run fixtures:record -- mssql    (SEAQUEL_TEST_MSSQL, 127.0.0.1:1433)
 *
 * The scratch objects live in `seaquel_test` (in `dbo` and a `fx_sales`
 * schema), all named `fx…`; their SQL is the `scratch` block of
 * `bugfixes.json`, so the Rust tests build the same ones. Every constraint
 * and index is named, since SQL Server's generated names change per run.
 *
 * Tables whose names fail the adapter's `validateIdentifier` (`fx order
 * items`, `fx]odd name`) and the views (the adapter joined `sys.tables`, so a
 * view had no columns) can't be recorded through the adapter's catalog
 * queries; their expectations are hand-written bug-fix cases, checked live.
 *
 * EXPLAIN: the adapter's `getExplainQuery` puts `SET SHOWPLAN_XML ON`, the
 * query and `OFF` in one batch, which SQL Server rejects, and its STATISTICS
 * XML batch returns the query's rows first, where `seaquel-server` stops. So
 * `parse.explain` isn't used: the plans are captured with the Node `mssql`
 * package (`../mssql-showplan.ts`) through `parse.extra`, and replayed
 * through `parseExplainResult`.
 *
 * `parseExplainResult` uses the browser's `DOMParser`; vitest runs in Node,
 * so jsdom's is installed on `globalThis` below.
 *
 * No CRUD group: the Rust dialect binds parameters (`@P1`, decision 4 of the
 * phase 2 plan) instead of the adapter's inline literals, and is checked with
 * live tests (Task 13). No statistics: the adapter has none.
 */

/** A plan built by hand: branches no captured plan reaches. */
function synthetic(name: string, xml: string | null, analyze = false): NamedInput {
  const column = "Microsoft SQL Server 2005 XML Showplan";
  return {
    name: `synthetic: ${name}`,
    input: {
      sql: "(synthetic)",
      analyze,
      columns: xml === null ? [] : [column],
      rows: xml === null ? [] : [{ [column]: xml }],
    },
  };
}
