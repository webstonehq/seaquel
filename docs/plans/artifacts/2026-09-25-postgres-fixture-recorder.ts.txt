/**
 * Parity fixtures for the TypeScript Postgres dialect (Rust Core Phase 1, Task 3).
 *
 * Record mode (`npm run fixtures:postgres`, i.e. RECORD_FIXTURES=1) runs the
 * adapter against the seeded Docker Postgres and writes one JSON file per
 * group to `crates/seaquel-engine-postgres/tests/fixtures/`. Each file is
 * `{ "cases": [ { "name", "input", "output" } ] }`.
 *
 * Check mode (the default) needs no database: it replays every stored
 * `input` through the TypeScript code and compares with the stored `output`.
 * Parser groups replay the stored rows, so volatile values (EXPLAIN ANALYZE
 * timings, statistics sizes) are only compared against themselves.
 *
 * `bugfixes.json` is written by hand and never by this script. Check mode
 * validates its shape and that TypeScript still shows the old behaviour;
 * record mode additionally proves its catalog expectations (fixes 1 and 4)
 * against the scratch schema.
 */

type Json = null | boolean | number | string | Json[] | { [key: string]: Json };
interface FixtureCase {
  name: string;
  input: Json;
  output: Json;
}
interface FixtureFile {
  cases: FixtureCase[];

/** What the fixture file holds: JSON drops `undefined` properties. */
function toJson(value: unknown): Json {
  return JSON.parse(JSON.stringify(value)) as Json;
}

function fixturePath(file: string): string {
  return `${FIXTURE_DIR}${file}`;
}

function readFixture(file: string): FixtureFile {
  return JSON.parse(readFileSync(fixturePath(file), "utf-8")) as FixtureFile;
}

function writeFixture(file: string, cases: FixtureCase[]): void {
  mkdirSync(FIXTURE_DIR, { recursive: true });
  writeFileSync(fixturePath(file), JSON.stringify({ cases }, null, 2) + "\n");
}

// ─── Replay: fixture input → TypeScript output ──────────────────────

type Obj = Record<string, unknown>;

function castLookupFrom(casts: unknown): CastLookup | undefined {
  if (!casts) return undefined;
  const map = casts as Record<string, string>;
  return (column) => (Object.hasOwn(map, column) ? map[column] : undefined);
}

function replayCrud(input: Obj): unknown {
  const schema = input.schema as string;
  const table = input.table as string;
  const primaryKeys = input.primaryKeys as string[];
  const row = input.row as Obj;
  switch (input.fn) {
    case "update":
      return adapter.buildUpdateSql(
        schema,
        table,
        input.column as string,
        input.value,
        primaryKeys,
        row,
        castLookupFrom(input.casts),
      );
    case "setDefault":
      return adapter.buildSetDefaultSql(schema, table, input.column as string, primaryKeys, row);
    case "insert":
      return adapter.buildInsertSql(
        schema,
        table,
        input.values as Obj,
        castLookupFrom(input.casts),
      );
    case "delete":
      return adapter.buildDeleteSql(schema, table, primaryKeys, row);
    default:
      throw new Error(`unknown crud fn ${String(input.fn)}`);
  }
}

function replaySql(input: Obj): string {
  switch (input.query) {
    case "getSchemaQuery":
      return adapter.getSchemaQuery();
    case "getColumnsQuery":
      return adapter.getColumnsQuery(input.table as string, input.schema as string);
    case "getIndexesQuery":
      return adapter.getIndexesQuery(input.table as string, input.schema as string);
    case "getTableSizesQuery":
      return adapter.getTableSizesQuery();
    case "getIndexUsageQuery":
      return adapter.getIndexUsageQuery();
    case "getDatabaseOverviewQuery":
      return adapter.getDatabaseOverviewQuery();
    case "getSchemasQuery":
      return adapter.getSchemasQuery();
    case "getExplainQuery":
      return adapter.getExplainQuery(input.sql as string, input.analyze as boolean);
    default:
      throw new Error(`unknown query ${String(input.query)}`);
  }
}

function replayStats(input: Obj): unknown {
  const rows = input.rows as unknown[];
  switch (input.kind) {
    case "tableSizes":
      return adapter.parseTableSizesResult(rows);
    case "indexUsage":
      return adapter.parseIndexUsageResult(rows);
    case "overview":
      return adapter.parseDatabaseOverviewResult(rows);
    default:
      throw new Error(`unknown stats kind ${String(input.kind)}`);
  }
}

const replay: Record<string, (input: Obj) => unknown> = {
  "crud.json": replayCrud,
  "quote.json": (i) => adapter.quoteIdentifier(i.id as string),
  "paginate.json": (i) =>
    adapter.paginateQuery(i.sql as string, i.limit as number, i.offset as number),
  "ddl-create.json": (i) => adapter.generateCreateTableSql(i as unknown as CreateTableDefinition),
  "ddl-alter.json": (i) =>
    adapter.generateAlterTableSql(i.from as CreateTableDefinition, i.to as CreateTableDefinition),
  "column-types.json": () => adapter.getColumnTypes(),
  "parse-schema.json": (i) => adapter.parseSchemaResult(i.rows as unknown[]),
  "parse-columns.json": (i) => adapter.parseColumnsResult(i.rows as unknown[]),
  "parse-indexes.json": (i) => adapter.parseIndexesResult(i.rows as unknown[]),
  "parse-stats.json": replayStats,
  "parse-explain.json": (i) =>
    adapter.parseExplainResult(i.rows as unknown[], i.analyze as boolean),
  "sql.json": replaySql,
};

function col(
  id: string,
  name: string,
  type: string,
  extra: Partial<CreateTableColumn> = {},
): CreateTableColumn {
  return {
    id,
    name,
    type,
    nullable: true,
    defaultValue: "",
    isPrimaryKey: false,
    isUnique: false,
    ...extra,
  };
}

function idx(id: string, name: string, columns: string[], unique = false): CreateTableIndex {
  return { id, name, columns, unique, type: "btree" };
}

function fk(
  id: string,
  column: string,
  referencedSchema: string,
  referencedTable: string,
  referencedColumn: string,
): CreateTableForeignKey {
  return { id, column, referencedSchema, referencedTable, referencedColumn };
}

function def(
  tableName: string,
  columns: CreateTableColumn[],
  extra: Partial<CreateTableDefinition> = {},
): CreateTableDefinition {
  return { tableName, schemaName: "public", columns, indexes: [], foreignKeys: [], ...extra };
}

/** Base table for ALTER cases. */
function baseDef(): CreateTableDefinition {
  return def(
    "customers",
    [
      col("c1", "id", "integer", { nullable: false, isPrimaryKey: true }),
      col("c2", "name", "varchar", { length: "100", nullable: false }),
      col("c3", "email", "text"),
      col("c4", "status", "text", { defaultValue: "'active'" }),
      col("c5", "balance", "numeric", { precision: "10,2", nullable: false, defaultValue: "0" }),
    ],
    {
      schemaName: "crm",
      indexes: [idx("i1", "customers_email_idx", ["email"])],
      foreignKeys: [fk("f1", "id", "auth", "users", "id")],
    },
  );
}

function alterCase(name: string, change: (d: CreateTableDefinition) => void): NamedInput {
  const from = baseDef();
  const to = baseDef();
  change(to);
  return { name, input: { from: asJson(from), to: asJson(to) } };
}

function column(d: CreateTableDefinition, id: string): CreateTableColumn {
  const c = d.columns.find((x) => x.id === id);
  if (!c) throw new Error(`no column ${id}`);
  return c;
}

/**

 * plus `order items` / `my-table` for bug fixes 1 and 4. The same DDL lives
 * in bugfixes.json (`scratch.setup`) so Rust tests can rebuild it.
 */
function scratchSetup(): string[] {
  return (readFixture("bugfixes.json") as unknown as { scratch: { setup: string[] } }).scratch
    .setup;
}
function scratchTeardown(): string[] {
  return (readFixture("bugfixes.json") as unknown as { scratch: { teardown: string[] } }).scratch
    .teardown;

let client: pg.Client | undefined;

async function rows(sql: string, params: unknown[] = []): Promise<Obj[]> {
  if (!client) throw new Error("no database client (record mode only)");
  const res = await client.query(sql, params);
  return res.rows as Obj[];
}

/**
 * Mimic what Seaquel's Rust driver produced before Task 2: int8 and numeric
 * as JSON numbers. `pg` already returns bool as boolean, json/jsonb parsed,
 * int2/int4 as numbers and name/text/domain-over-text as strings.
 */
function configureTypeParsers(): void {
  pg.types.setTypeParser(20, (v: string) => Number(v)); // int8
  pg.types.setTypeParser(1700, (v: string) => Number(v)); // numeric
}

const byIndexName = (a: Obj, b: Obj) =>
  String(a.indexname) < String(b.indexname)
    ? -1
    : String(a.indexname) > String(b.indexname)
      ? 1
      : 0;

async function recordDbInputs(file: string): Promise<NamedInput[]> {
  switch (file) {
    case "parse-schema.json": {
      // Only the seeded and scratch schemas: other tests may leave schemas behind.
      const all = await rows(adapter.getSchemaQuery());
      const kept = all.filter((r) => r.schema_name === "public" || r.schema_name === SCRATCH);
      return [{ name: "public and scratch schemas", input: { rows: toJson(kept) } }];
    }
    case "parse-columns.json": {
      const out: NamedInput[] = [];
      for (const [schema, table] of COLUMN_TARGETS) {
        const r = await rows(adapter.getColumnsQuery(table, schema));
        out.push({ name: `${schema}.${table}`, input: { schema, table, rows: toJson(r) } });
      }
      return out;
    }
    case "parse-indexes.json": {
      const out: NamedInput[] = [];
      for (const [schema, table] of INDEX_TARGETS) {
        // pg_indexes has no ORDER BY; sort by name so re-recording is stable.
        const r = (await rows(adapter.getIndexesQuery(table, schema))).sort(byIndexName);
        out.push({ name: `${schema}.${table}`, input: { schema, table, rows: toJson(r) } });
      }
      return out;
    }
    case "parse-stats.json":
      // The TS statistics queries splice `schemaname || '.' || relname` into
      // regclass casts, so any table whose name needs quoting makes them fail
      // ("invalid name syntax"). Drop the bug-fix tables first; the bugfixes
      // checks have already run by now.
      await rows(`DROP TABLE ${SCRATCH}."order items", ${SCRATCH}."my-table"`);
      // Publish this session's pending insert counts (PG 15+) so n_live_tup
      // for the scratch tables doesn't depend on flush timing.
      await rows("SELECT pg_stat_force_next_flush()");
      await rows("SELECT pg_sleep(0.05)");
      return [
        {
          name: "table sizes",
          input: { kind: "tableSizes", rows: toJson(await rows(adapter.getTableSizesQuery())) },
        },
        {
          name: "index usage",
          input: { kind: "indexUsage", rows: toJson(await rows(adapter.getIndexUsageQuery())) },
        },
        {
          name: "database overview",
          input: { kind: "overview", rows: toJson(await rows(adapter.getDatabaseOverviewQuery())) },
        },
        { name: "database overview, no rows (fallbacks)", input: { kind: "overview", rows: [] } },
      ];
    case "parse-explain.json": {
      const out: NamedInput[] = [];
      for (const analyze of [false, true]) {
        for (const target of EXPLAIN_TARGETS) {
          const r = await rows(adapter.getExplainQuery(target.sql, analyze));
          out.push({
            name: `${analyze ? "EXPLAIN ANALYZE" : "EXPLAIN"}: ${target.name}`,
            input: { sql: target.sql, analyze, rows: toJson(r) },
          });
        }
      }
      return out;
    }
    default:
      throw new Error(`no recorder for ${file}`);
  }
}

async function inputsFor(file: string): Promise<NamedInput[]> {
  if (file === "sql.json") return sqlInputs;
  if (file in pureInputs) return pureInputs[file];
  return recordDbInputs(file);
}

function checkCases(file: string, cases: FixtureCase[]): void {
  const run = replay[file];
  expect(cases.length, `${file} has cases`).toBeGreaterThan(0);
  const names = new Set<string>();
  for (const c of cases) {
    expect(names.has(c.name), `${file}: duplicate case name "${c.name}"`).toBe(false);
    names.add(c.name);
    expect(toJson(run(c.input as Obj)), `${file}: ${c.name}`).toStrictEqual(c.output);
  }
}

describe("Postgres dialect parity fixtures", () => {
  if (RECORD) {
    beforeAll(async () => {
      configureTypeParsers();
      client = new pg.Client({ connectionString: DATABASE_URL });
      await client.connect();
      for (const stmt of scratchTeardown()) await client.query(stmt);
      for (const stmt of scratchSetup()) await client.query(stmt);
      // Flush the insert counts before ANALYZE, or they are added on top of
      // ANALYZE's live-tuple count later (n_live_tup would read 10000).
      await client.query("SELECT pg_stat_force_next_flush()");
      await client.query("SELECT pg_sleep(0.05)");
      // Deterministic plans: fresh statistics and no parallel workers.
      await client.query(
        `ANALYZE ${SCRATCH}.customers, ${SCRATCH}.events, public.users, public.orders`,
      );
      await client.query("SET max_parallel_workers_per_gather = 0");
    });

    afterAll(async () => {
      if (!client) return;
      try {
        for (const stmt of scratchTeardown()) await client.query(stmt);
      } finally {
        await client.end();
        client = undefined;
      }
    });
  }

  describe("bugfixes.json (hand-written)", () => {
    interface BugfixCase {
      name: string;
      fix: 1 | 2 | 3 | 4;
      kind: "columns" | "indexes" | "ddl-create" | "ddl-alter";
      input: Obj;
      output: Json;
    }
    const bugfixes = () =>
      readFixture("bugfixes.json") as unknown as { scratch: Obj; cases: BugfixCase[] };

    it("covers all four fixes", () => {
      const fixes = new Set(bugfixes().cases.map((c) => c.fix));
      expect([...fixes].sort((a, b) => a - b)).toEqual([1, 2, 3, 4]);
    });

    it("fixes 2 and 3 differ from what TypeScript generates today", () => {
      for (const c of bugfixes().cases) {
        if (c.kind !== "ddl-create" && c.kind !== "ddl-alter") continue;
        const ts = replay[`${c.kind}.json`](c.input);
        expect(ts, `${c.name}: TypeScript output should still show the bug`).not.toEqual(c.output);
      }
    });

    it("fix 1: TypeScript still rejects the identifiers", () => {
      for (const c of bugfixes().cases) {
        if (c.kind !== "columns" && c.kind !== "indexes") continue;
        const { schema, table } = c.input as { schema: string; table: string };
        if (/^[\p{L}\p{N}_$]+$/u.test(table) && /^[\p{L}\p{N}_$]+$/u.test(schema)) continue;
        expect(() => adapter.getColumnsQuery(table, schema), c.name).toThrow(
          /Invalid SQL identifier/,
        );
      }
    });

    if (RECORD) {
      it("fixes 1 and 4 match the real catalog", async () => {
        for (const c of bugfixes().cases) {
          const { schema, table } = c.input as { schema: string; table: string };
          if (c.kind === "columns") {
            // The TS columns query with the catalog filters bound, as fix 1 does.
            const sql = adapter
              .getColumnsQuery("__table__", "__schema__")
              .replace("'__table__'", "$1")
              .replace("'__schema__'", "$2");
            const got = adapter.parseColumnsResult(await rows(sql, [table, schema]));
            expect(toJson(got), c.name).toStrictEqual(c.output);
          } else if (c.kind === "indexes") {
            expect(toJson(await catalogIndexes(schema, table)), c.name).toStrictEqual(c.output);
          }
        }
      });
    }
  });

  for (const file of [...PURE_FILES, ...DB_FILES, "sql.json"]) {
    it(file, async () => {
      let cases: FixtureCase[];
      if (RECORD) {
        const inputs = await inputsFor(file);
        cases = inputs.map(({ name, input }) => ({
          name,
          input,
          output: toJson(replay[file](input as Obj)),
        }));
        writeFixture(file, cases);
      } else {
        cases = readFixture(file).cases;
      }
      checkCases(file, cases);
    });
  }

  it("has at least 60 cases in total", () => {
    const total = [...PURE_FILES, ...DB_FILES, "sql.json"].reduce(
      (n, file) => n + readFixture(file).cases.length,
      0,
    );
    expect(total).toBeGreaterThanOrEqual(60);
  });
});

/** Reference implementation of fix 4, used only to validate bugfixes.json. */
async function catalogIndexes(schema: string, table: string): Promise<Json[]> {
  const idxRows = await rows(
    `SELECT ic.relname AS name, x.indexrelid::int8 AS oid, x.indnkeyatts AS nkeys,
            x.indisunique AS "unique", am.amname AS type
       FROM pg_index x
       JOIN pg_class ic ON ic.oid = x.indexrelid
       JOIN pg_class t ON t.oid = x.indrelid
       JOIN pg_namespace n ON n.oid = t.relnamespace
       JOIN pg_am am ON am.oid = ic.relam
      WHERE n.nspname = $1 AND t.relname = $2
      ORDER BY ic.relname`,
    [schema, table],
  );
  const out: Json[] = [];
  for (const r of idxRows) {
    const cols = await rows(
      "SELECT pg_get_indexdef($1::oid, k, true) AS col FROM generate_series(1, $2::int) AS k ORDER BY k",
      [r.oid, r.nkeys],
    );
    out.push({
      name: r.name as string,
      columns: cols.map((c) => c.col as string),
      unique: r.unique as boolean,
      type: r.type as string,
    });
  }
  return out;
}
